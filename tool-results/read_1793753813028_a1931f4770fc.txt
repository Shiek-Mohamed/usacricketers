    43→/** Format an ISO date as "Mon DD" (e.g. "Mar 10"). */
    44→function formatShortDate(iso: string): string {
    45→  try {
    46→    const d = new Date(iso)
    47→    return d.toLocaleDateString("en-US", { month: "short", day: "numeric" })
    48→  } catch {
    49→    return ""
    50→  }
    51→}

   100→/** Last innings total for a team in a match, e.g. "217/4". */
   101→function inningsTotal(match: Match, teamId: string): string {
   102→  const sc = match.scorecard
   103→  if (!sc?.innings) return ""
   104→  const inn = sc.innings.filter((i) => i.teamId === teamId)
   105→  if (!inn.length) return ""
   106→  const last = inn[inn.length - 1]
   107→  return `${last.totalRuns}/${last.totalWickets}`
   108→}

   110→/** Compact horizontal live-match bar (cricket.com.au-style live strip).
   111→ *  Mobile-first: everything fits in one row on a 375px (iPhone SE) viewport.
   112→ *  - Badges + scores are `shrink-0` (never compressed).
   113→ *  - Team short-name spans are `min-w-0 truncate` so they clip instead of
   114→ *    pushing siblings past the card boundary.
   115→ *  - Gaps + font sizes scale up at the `sm:` breakpoint.
   116→ */
   117→function LiveMatchStrip({ match }: { match: Match }) {
   118→  const t1 = match.team1
   119→  const t2 = match.team2
   120→  const t1Score = inningsTotal(match, match.team1Id)
   121→  const t2Score = inningsTotal(match, match.team2Id)
   122→  // Render as a real `<a href="/matches/${id}">` so the strip deep-links to
   123→  // the SSR match page (full page navigation — what we want for SEO +
   124→  // shareability). The SPA `?view=match-detail&id=xxx` flow still works for
   125→  // legacy links.
   126→  const href = `/matches/${match.id}`
   127→  return (
   128→    <a
   129→      href={href}
   130→      className="group flex w-full items-center gap-1.5 overflow-hidden rounded-lg border bg-card px-2.5 py-2 text-left shadow-sm transition-shadow hover:shadow-md sm:gap-2 sm:px-4 sm:py-2.5"
   131→    >
   132→      {/* Live indicator */}
   133→      <span className="flex shrink-0 items-center gap-1">
   134→        <span className="relative flex h-2 w-2">
   135→          <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-red-500 opacity-75" />
   136→          <span className="relative inline-flex h-2 w-2 rounded-full bg-red-600" />
   137→        </span>
   138→        <span className="text-[10px] font-bold uppercase tracking-wide text-red-600">
   139→          Live
   140→        </span>
   141→      </span>
   142→      <FormatPill format={match.format} />
   143→      {/* Match metadata — hidden below `sm` to save horizontal real estate */}
   144→      <span className="hidden min-w-0 truncate text-xs text-muted-foreground sm:inline">
   145→        {match.matchNo}
   146→        {match.series?.shortName ? ` · ${match.series.shortName}` : ""}
   147→      </span>
   148→      {/* Teams + scores — pushed right; truncates names instead of overflowing */}
   149→      <div className="ml-auto flex min-w-0 items-center justify-end gap-1.5 sm:gap-3">
   150→        <div className="flex min-w-0 items-center gap-1 sm:gap-1.5">
   151→          {t1 && <TeamBadge team={t1} size="sm" />}
   152→          <span className="min-w-0 truncate text-xs font-bold sm:text-sm">
   153→            {t1?.shortName}
   154→          </span>
   155→          {t1Score && (
   156→            <span className="shrink-0 text-xs font-extrabold tabular-nums sm:text-sm">
   157→              {t1Score}
   158→            </span>
   159→          )}
   160→        </div>
   161→        <span className="shrink-0 text-[10px] font-medium text-muted-foreground sm:text-xs">
   162→          v
   163→        </span>
   164→        <div className="flex min-w-0 items-center gap-1 sm:gap-1.5">
   165→          {t2 && <TeamBadge team={t2} size="sm" />}
   166→          <span className="min-w-0 truncate text-xs font-bold sm:text-sm">
   167→            {t2?.shortName}
   168→          </span>
   169→          {t2Score && (
   170→            <span className="shrink-0 text-xs font-extrabold tabular-nums sm:text-sm">
   171→              {t2Score}
   172→            </span>
   173→          )}
   174→        </div>
   175→      </div>
   176→      <ChevronRight className="h-4 w-4 shrink-0 text-muted-foreground transition-transform group-hover:translate-x-0.5" />
   177→    </a>
   178→  )
   179→}

   770→/** Resolve a display thumbnail for a video (custom → YouTube hqdefault → none). */
   771→function videoThumb(v: Video): string {
   772→  if (v.thumbnailUrl) return v.thumbnailUrl
   773→  if (v.source === "youtube" && v.youtubeId)
   774→    return `https://img.youtube.com/vi/${v.youtubeId}/hqdefault.jpg`
   775→  return ""
   776→}
