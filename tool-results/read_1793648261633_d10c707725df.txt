    43→/** Format an ISO date as "Mon DD" (e.g. "Mar 10"). */
    44→function formatShortDate(iso: string): string {
    45→  try {
    46→    const d = new Date(iso)
    47→    return d.toLocaleDateString("en-US", { month: "short", day: "numeric" })
    48→  } catch {
    49→    return ""
    50→  }
    51→}

   100→/** Last innings total for a team in a match, e.g. "217/4". */
   101→function inningsTotal(match: Match, teamId: string): string {
   102→  const sc = match.scorecard
   103→  if (!sc?.innings) return ""
   104→  const inn = sc.innings.filter((i) => i.teamId === teamId)
   105→  if (!inn.length) return ""
   106→  const last = inn[inn.length - 1]
   107→  return `${last.totalRuns}/${last.totalWickets}`
   108→}

   274→export function HomeView() {
   275→  const navigate = useAppStore((s) => s.navigate)
   276→  const { data: matchesData, loading: matchesLoading } = useMatches()
   277→  const { data: featuredArticles, loading: articlesLoadingFeatured } =
   278→    useArticles({ featured: true, pageSize: 12 })
   279→  const { data: allArticles, loading: articlesLoadingAll } = useArticles({ pageSize: 16 })
   280→  const { data: interviewData } = useArticles({ category: "interview", pageSize: 6 })
   281→
   282→  const matches = matchesData ?? []
   283→  const liveMatch = matches.find((m) => m.status === "live")
   284→
   285→  const completedMatches = useMemo(
   286→    () => matches.filter((m) => m.status === "completed").slice(0, 6),
   287→    [matches]
   288→  )
   289→  const upcomingMatches = useMemo(
   290→    () => matches.filter((m) => m.status === "upcoming").slice(0, 6),
   291→    [matches]
   292→  )
   293→
   294→  const featuredItems = featuredArticles?.items ?? []
   295→  const latestItems = allArticles?.items ?? []
   296→
   297→  // Featured hero = most recent article that has a hero image; secondary = next 3 with images.
   298→  const withImages = useMemo(
   299→    () => latestItems.filter((a) => a.imageUrl),
   300→    [latestItems]
   301→  )
   302→  const heroStory = withImages[0] ?? featuredItems[0] ?? latestItems[0] ?? null
   303→  const secondaryStories = withImages.slice(1, 4)
   304→
   305→  // Exclude the hero + secondary stories from Latest News so they don't
   306→  // appear twice on the homepage.
   307→  const featuredIds = useMemo(
   308→    () =>
   309→      new Set(
   310→        [heroStory?.id, ...secondaryStories.map((a) => a.id)].filter(
   311→          Boolean,
   312→        ) as string[],
   313→      ),
   314→    [heroStory, secondaryStories],
   315→  )
   316→  const latestNews = useMemo(
   317→    () => latestItems.filter((a) => !featuredIds.has(a.id)).slice(0, 8),
   318→    [latestItems, featuredIds],
   319→  )
   320→
   321→  const heroLoading = articlesLoadingFeatured && articlesLoadingAll
   322→
   323→  return (
   324→    <div className="space-y-6 sm:space-y-8">
   325→      {/* Hero section — cricket.com.au-style featured story */}
   326→      <section className="space-y-5">
   327→        {/* Live match strip */}
   328→        {matchesLoading ? (
   329→          <Skeleton className="h-12 w-full rounded-lg" />
   330→        ) : liveMatch ? (
   331→          <LiveMatchStrip match={liveMatch} />
   332→        ) : null}
   333→
   334→        {/* AD SLOT — between Live and the hero (AdSense, non-rotating) */}
   335→        <BannerSlot placement="home-live-hero" />
   336→
   337→        {/* Featured hero story (split layout: text card + large image) */}
   338→        {heroLoading ? (
   339→          <HomeHeroSkeleton />
   340→        ) : heroStory ? (
   341→          <FeaturedHero article={heroStory} />
   342→        ) : (
   343→          <Card className="grid min-h-[200px] place-items-center p-6">
   344→            <p className="text-sm text-muted-foreground">No stories available.</p>
   345→          </Card>
   346→        )}
   347→
   348→        {/* Secondary stories — 3 cards */}
   349→        {secondaryStories.length > 0 && (
   350→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   351→            {secondaryStories.map((a) => (
   352→              <SecondaryStoryCard key={a.id} article={a} />

   356→      </section>
   357→
   358→      {/* AD SLOT — above Latest Results (rotating) */}

   361→      {/* Latest Results + Upcoming Matches — navy panel */}

   747→/** Resolve a display thumbnail for a video (custom → YouTube hqdefault → none). */
   748→function videoThumb(v: Video): string {
   749→  if (v.thumbnailUrl) return v.thumbnailUrl
   750→  if (v.source === "youtube" && v.youtubeId)
   751→    return `https://img.youtube.com/vi/${v.youtubeId}/hqdefault.jpg`
   752→  return ""
   753→}
