    79→  // Site admins don't have a scorer session — they only have the sa-session
    80→  // httpOnly cookie the score-service can't read. Fall back to fetching a
    81→  // god-mode scorer token (POST /api/admin/scorer-token), which the
    82→  // score-service accepts as `tournamentIds: ["*"]` god-mode.
    83→  //
    84→  // NOTE: We don't gate on `mode === "admin"` because navigating directly to
    85→  // ?view=live-scorer&id=... bypasses the admin-login flow that sets the mode.
    86→  // Instead, we try fetching the scorer token unconditionally (if scorerToken
    87→  // isn't already set). If the admin cookie isn't there, the API returns 401
    88→  // and adminToken stays null — the shell shows "Not authenticated".

   102→        if (cancelled) return

   104→      })
   105→      .catch(() => {
   106→        // 401 means the admin cookie expired or isn't there — the shell
   107→        // below will show the "Not authenticated" gate instead.
   108→      })

   110→      cancelled = true
   111→    }

   123→  // Wicket-dialog open state — the Wicket button opens a modal that lets the
   124→  // admin pick dismissal type + fielder(s) involved, then sends the structured
   125→  // wicket BallEvent to applyBall via handleBall.

   128→  /* ------------------ Keyboard shortcuts (power-user scorers) ------------------
   129→   * 0,1,2,3,4,6 → score N runs | . / - → dot | w → wicket dialog |
   130→   * e → swap ends | u → undo | d → declare (Test) | s → save.
   131→   * No deps array → re-subscribes on every render with the latest closures.
   132→   * The function declarations (handleBall, swapEnds, etc.) are hoisted by
   133→   * JS so they're in scope at call-time even though defined later in the
   134→   * component body. The lint rule below is overly cautious about this
   135→   * pattern — it's safe because the effect re-runs every render. */
   136→  /* eslint-disable react-hooks/immutability */
   137→  useEffect(() => {
   138→    function isTyping(el: Element | null): boolean {
   139→      if (!el) return false
   140→      const tag = el.tagName?.toLowerCase()
   141→      return tag === "input" || tag === "textarea" || tag === "select" ||
   142→        el.getAttribute("role") === "combobox" ||
   143→        (el as HTMLElement)?.isContentEditable === true
   144→    }
   145→    function onKey(e: KeyboardEvent) {
   146→      if (e.metaKey || e.ctrlKey || e.altKey) return
   147→      if (isTyping(document.activeElement)) return
   148→      if (wicketOpen) return
   149→      if (!liveState || liveState.status !== "live") return
   150→      if (!liveState.strikerId || !liveState.nonStrikerId || !liveState.bowlerId) return
   151→      const k = e.key.toLowerCase()
   152→      if (["0", "1", "2", "3", "4", "6"].includes(k)) {
   153→        e.preventDefault()
   154→        handleBall({ kind: "run", runs: Number(k) })
   155→      } else if (k === "." || k === "-") {
   156→        e.preventDefault()
   157→        handleBall({ kind: "run", runs: 0 })
   158→      } else if (k === "w") {
   159→        e.preventDefault()
   160→        setWicketOpen(true)
   161→      } else if (k === "e") {
   162→        e.preventDefault()
   163→        swapEnds()
   164→      } else if (k === "u") {
   165→        e.preventDefault()
   166→        handleUndo()
   167→      } else if (k === "d") {
   168→        e.preventDefault()
   169→        handleDeclare()
   170→      } else if (k === "s") {
   171→        e.preventDefault()
   172→        if (liveState) {
   173→          persist(liveState)
   174→          toast.success("Scorecard saved")
   175→        }
   176→      }
   177→    }
   178→    window.addEventListener("keydown", onKey)
   179→    return () => window.removeEventListener("keydown", onKey)
   180→  })

   222→  function persist(state: LiveMatchState) {
   223→    if (saveTimer.current) clearTimeout(saveTimer.current)
   224→    saveTimer.current = setTimeout(async () => {
   225→      try {
   226→        await api.matches.update(match!.id, {
   227→          scorecard: stateToScorecard(state),
   228→          status: state.status,
   229→          result: state.result,
   230→        })
   231→        refetch()
   232→      } catch (e) {
   233→        // Surface persistence errors so the operator knows the DB write failed
   234→        // (was previously silent — caused confusion when "Innings started"
   235→        // toast showed but the match status didn't actually update).
   236→        // The live socket broadcast is still the priority — this toast
   237→        // doesn't block scoring, just informs the operator something's wrong.
   238→        const msg = e instanceof Error ? e.message : String(e).slice(0, 200)
   239→        toast.error("Persist failed — match state not saved", {
   240→          description: msg,
   241→        })
   242→      }
   243→    }, 800)
   244→  }

   246→  function handleBall(event: BallEvent) {
   247→    if (!liveState) return
   248→    setHistory((h) => [...h, liveState].slice(-50))
   249→    const next = applyBall(liveState, event)

   251→    persist(next)

   254→  function handleUndo() {
   255→    if (history.length === 0) {
   256→      toast.info("Nothing to undo")
   257→      return
   258→    }
   259→    const prev = history[history.length - 1]
   260→    setHistory((h) => h.slice(0, -1))
   261→    pushState(prev)
   262→    persist(prev)
   263→    toast.success("Undid last action")
   264→  }

   266→  function startInnings(
   267→    battingFirstId: string,
   268→    strikerId: string,
   269→    nonStrikerId: string,
   270→    bowlerId: string
   271→  ) {
   272→    const battingPlayers = playersByTeam[battingFirstId] ?? []
   273→    const bowlingPlayers =
   274→      playersByTeam[battingFirstId === match!.team1Id ? match!.team2Id : match!.team1Id] ?? []
   275→    const striker = battingPlayers.find((p) => p.id === strikerId)
   276→    const nonStriker = battingPlayers.find((p) => p.id === nonStrikerId)
   277→    const bowler = bowlingPlayers.find((p) => p.id === bowlerId)
   278→    if (!striker || !nonStriker || !bowler) {
   279→      toast.error("Select striker, non-striker and bowler")
   280→      return
   281→    }
   282→    let state = initState(match!, battingFirstId)
   283→    // Pre-populate the squad with the full batting-team roster so the
   284→    // "Did not bat" section of the public scorecard lists bench players.
   285→    // Captain/keeper tags come from the explicit Player model fields
   286→    // (was previously a brittle regex on the role string).
   287→    for (const p of battingPlayers) {
   288→      state = addSquadMember(state, p.id, p.name, {
   289→        isCaptain: !!p.isCaptain,
   290→        isKeeper: !!p.isKeeper,
   291→      })
   292→    }
   293→    state = addBatsman(state, striker.id, striker.name, {
   294→      isCaptain: !!striker.isCaptain,
   295→      isKeeper: !!striker.isKeeper,
   296→    })
   297→    state = addBatsman(state, nonStriker.id, nonStriker.name, {
   298→      isCaptain: !!nonStriker.isCaptain,
   299→      isKeeper: !!nonStriker.isKeeper,
   300→    })
   301→    state = setBatsmen(state, striker.id, nonStriker.id)
   302→    state = setBowler(state, bowler.id, bowler.name)
   303→    setHistory([])
   304→    pushState(state)
   305→    persist(state)
   306→    toast.success("Innings started — status set to LIVE")
   307→  }

   309→  function bringInBatsman(playerId: string) {
   310→    if (!liveState) return
   311→    const battingPlayers = playersByTeam[liveState.innings[liveState.currentInnings].battingTeamId] ?? []
   312→    const p = battingPlayers.find((x) => x.id === playerId)
   313→    if (!p) return
   314→    let state = addBatsman(liveState, p.id, p.name, {
   315→      isCaptain: !!p.isCaptain,
   316→      isKeeper: !!p.isKeeper,
   317→    })
   318→    // Fill whichever slot is currently empty — the striker's end if a normal
   319→    // wicket fell (strikerId === null), or the non-striker's end if a run-out
   320→    // happened at the non-striker's end (nonStrikerId === null). If both are
   321→    // empty (innings restart, edge case), default to striker.
   322→    const fillStriker = liveState.strikerId === null
   323→    const fillNonStriker = !fillStriker && liveState.nonStrikerId === null
   324→    state = {
   325→      ...state,
   326→      strikerId: fillStriker ? p.id : state.strikerId,
   327→      nonStrikerId: fillNonStriker ? p.id : state.nonStrikerId,
   328→    }
   329→    setHistory((h) => [...h, liveState].slice(-50))
   330→    pushState(state)
   331→    persist(state)
   332→    toast.success(`${p.name} in`)
   333→  }

   335→  function pickNextBowler(playerId: string) {
   336→    if (!liveState) return
   337→    const bowlingPlayers = playersByTeam[liveState.innings[liveState.currentInnings].bowlingTeamId] ?? []
   338→    const p = bowlingPlayers.find((x) => x.id === playerId)
   339→    if (!p) return

   342→    pushState(state)
   343→    persist(state)
   344→    toast.success(`${p.name} to bowl`)

   347→  /** "I picked the wrong striker — let me re-pick." Pushes the current state
   348→   *  to the history stack (so Undo can recover it), then clears the strikerId
   349→   *  slot. The Picker re-shows automatically because `needStriker` becomes
   350→   *  true when strikerId === null. */
   351→  function changeStriker() {
   352→    if (!liveState || !liveState.strikerId) return
   353→    setHistory((h) => [...h, liveState].slice(-50))
   354→    const next = { ...liveState, strikerId: null as string | null }
   355→    pushState(next)
   356→    persist(next)
   357→    toast.info("Re-pick the striker")
   358→  }

   362→    if (!liveState || !liveState.nonStrikerId) return

   364→    const next = { ...liveState, nonStrikerId: null as string | null }
   365→    pushState(next)
   366→    persist(next)
   367→    toast.info("Re-pick the non-striker")
   368→  }

   380→  /** Swap the striker and non-striker ends. Used when the auto strike-rotation
   381→   *  logic disagrees with reality (e.g. batsmen crossed on a caught-and-
   382→   *  bowled, odd run on a wide, runner substituted, etc.). Allowed at any
   383→   *  time — doesn't affect stats, just relabels who's on strike. */
   384→  function swapEnds() {
   385→    if (!liveState) return
   386→    if (!liveState.strikerId || !liveState.nonStrikerId) {
   387→      toast.error("Need both batsmen at the crease to swap ends")
   388→      return
   389→    }
   390→    setHistory((h) => [...h, liveState].slice(-50))
   391→    const next = {
   392→      ...liveState,
   393→      strikerId: liveState.nonStrikerId,
   394→      nonStrikerId: liveState.strikerId,
   395→    }
   396→    pushState(next)
   397→    persist(next)
   398→    toast.success("Ends swapped")
   399→  }

   401→  function reset() {
   402→    setHistory([])
   403→    // clear live cache
   404→    // (clearState not destructured here; use pushState(null) approach via a clear)
   405→    pushState({ ...liveState!, status: "live", result: "" })

   408→  /** Test-match declaration: captain ends the current innings without being
   409→   *  all out. No-op for limited-overs matches (those don't have declarations).
   410→   *  Pushes the prev state to history (so Undo can recover) then runs the
   411→   *  engine's declareInnings which marks `declared: true` + sets up the
   412→   *  next innings / computes the result. */
   413→  function handleDeclare() {
   414→    if (!liveState) return
   415→    if (liveState.maxOvers > 0) {
   416→      toast.error("Declarations are only available in Test matches")
   417→      return
   418→    }
   419→    if (liveState.status !== "live") {
   420→      toast.error("Can't declare — innings already ended")
   421→      return
   422→    }
   423→    setHistory((h) => [...h, liveState].slice(-50))
   424→    const next = declareInnings(liveState)
   425→    pushState(next)
   426→    persist(next)
   427→    toast.success("Innings declared")
   428→  }

   430→  /** Enforce the follow-on in a Test match. Only valid when Team A leads by
   431→   *  200+ runs after Team B's 1st innings. Team B bats again immediately. */
   432→  function handleFollowOn() {
   433→    if (!liveState) return
   434→    setHistory((h) => [...h, liveState].slice(-50))
   435→    const next = enforceFollowOn(liveState)
   436→    if (next === liveState) {
   437→      toast.error("Follow-on requires a 200+ run lead in a Test after innings 1")
   438→      return
   439→    }
   440→    pushState(next)
   441→    persist(next)
   442→    toast.success("Follow-on enforced")
   443→  }

   445→  /** Start a super-over after a tied limited-overs match. Creates 2 new
   446→   *  1-over innings (one per team). */
   447→  function handleSuperOver() {
   448→    if (!liveState) return

   452→      toast.error("Super-over only available for tied limited-overs matches")
   453→      return
   454→    }
   455→    pushState(next)
   456→    persist(next)
   457→    toast.success("Super Over started")
   458→  }

   874→function LiveScoreboard({
   875→  state,
   876→  onChangeStriker,
   877→  onChangeNonStriker,
   878→  onChangeBowler,
   879→  onSwapEnds,
   880→}: {
   881→  state: LiveMatchState
   882→  onChangeStriker?: () => void
   883→  onChangeNonStriker?: () => void
   884→  onChangeBowler?: () => void
   885→  /** Swap striker ↔ non-striker — override the auto strike-rotation logic
   886→   *  when it disagrees with reality (caught-and-bowled crossing, etc.).
   887→   *  Allowed at any time (not over-boundary-gated) — doesn't affect stats. */
   888→  onSwapEnds?: () => void
   889→}) {
   890→  const inn = state.innings[state.currentInnings]
   891→  const striker = inn.batsmen.find((b) => b.id === state.strikerId)
   892→  const nonStriker = inn.batsmen.find((b) => b.id === state.nonStrikerId)
   893→  const bowler = inn.bowlers.find((b) => b.id === state.bowlerId)
   894→  const overs = oversString(inn.balls)
   895→  const runRate = inn.balls > 0 ? (inn.runs / inn.balls * 6).toFixed(2) : "0.00"
   896→  const target = inn.target
   897→  const ballsLeft = state.maxOvers > 0 ? state.maxOvers * 6 - inn.balls : null
   898→  const runsNeeded = target ? target - inn.runs : null
   899→  const reqRate =
   900→    target && ballsLeft && ballsLeft > 0
   901→      ? ((runsNeeded! / ballsLeft) * 6).toFixed(2)
   902→      : null
   903→
   904→  // A slot can be safely "Changed" only at over-boundaries — mid-over swaps
   905→  // would mess up the bowler's figures + the batsman's balls-faced count.
   906→  // Disable the Change button once any ball of the current over has been
   907→  // bowled (i.e. balls % 6 !== 0). The exception is the bowler slot after a
   908→  // wicket (still mid-over) — the bowler can be subbed off due to injury in
   909→  // real cricket, but for our MVP we keep the rule simple: only at over
   910→  // boundaries. Undo is still available mid-over for fixing mistakes.
   911→  const atOverBoundary = inn.balls % 6 === 0
   912→  const canChangeBatsmen = atOverBoundary && state.status === "live"
   913→  const canChangeBowler = atOverBoundary && state.status === "live"
   914→
   915→  return (
   916→    <div className="rounded-lg border bg-gradient-to-br from-slate-900 to-slate-800 p-4 text-white sm:p-5">
   917→      <div className="flex flex-wrap items-end justify-between gap-3">
   918→        <div>
   919→          <div className="text-xs uppercase tracking-wide text-slate-400">
   920→            {inn.battingTeamShortName} · Innings {state.currentInnings + 1}
   921→            {target ? ` · need ${runsNeeded} runs` : ""}
   922→          </div>
   923→          <div className="mt-1 text-4xl font-extrabold tabular-nums">
   924→            {inn.runs}/{inn.wickets}
   925→            <span className="ml-2 text-lg font-normal text-slate-300">
   926→              ({overs} ov)
   927→            </span>
   928→            {inn.freeHit && (
   929→              <span className="ml-3 inline-flex items-center gap-1 rounded-full bg-amber-500 px-2.5 py-0.5 text-xs font-extrabold uppercase tracking-wide text-amber-950 animate-pulse">
   930→                <span className="h-1.5 w-1.5 rounded-full bg-amber-950" />
   931→                Free hit
   932→              </span>
   933→            )}
   934→          </div>
   935→          <div className="mt-1 text-xs text-slate-300">
   936→            CRR {runRate}
   937→            {reqRate && ` · RRR ${reqRate}`}
   938→            {ballsLeft !== null && ` · ${ballsLeft} balls left`}
   939→          </div>
   940→        </div>
   941→      </div>
   942→      <div className="mt-3 grid grid-cols-2 gap-3 text-xs">
   943→        <div className="rounded-md bg-white/10 p-2.5">
   944→          <div className="flex items-center justify-between">
   945→            <div className="text-[10px] uppercase text-slate-300">Striker</div>
   946→            {striker && onChangeStriker && (
   947→              <ChangeButton
   948→                onClick={onChangeStriker}
   949→                disabled={!canChangeBatsmen}
   950→                hint={
   951→                  canChangeBatsmen
   952→                    ? "Re-pick striker"
   953→                    : "Wait for end of over (or use Undo)"
   954→                }
   955→              />
   956→            )}
   957→          </div>
   958→          <div className="font-bold">{striker?.name ?? "—"}</div>
   959→          <div className="text-slate-300">
   960→            {striker ? `${striker.runs} (${striker.balls})` : "—"}
   961→          </div>
   962→        </div>
   963→        <div className="relative rounded-md bg-white/10 p-2.5">
   964→          <div className="flex items-center justify-between">
   965→            <div className="text-[10px] uppercase text-slate-300">Non-striker</div>
   966→            {nonStriker && onChangeNonStriker && (
   967→              <ChangeButton
   968→                onClick={onChangeNonStriker}
   969→                disabled={!canChangeBatsmen}
   970→                hint={
   971→                  canChangeBatsmen
   972→                    ? "Re-pick non-striker"
   973→                    : "Wait for end of over (or use Undo)"
   974→                }
   975→              />
   976→            )}
   977→          </div>
   978→          <div className="font-bold">{nonStriker?.name ?? "—"}</div>
   979→          <div className="text-slate-300">
   980→            {nonStriker ? `${nonStriker.runs} (${nonStriker.balls})` : "—"}
   981→          </div>
   982→          {/* Swap-ends button — pinned to the inner edge between the two
   983→              batsman cells. Visible only when both batsmen are at the crease.
   984→              Always enabled (not over-boundary-gated) because swapping ends
   985→              doesn't affect any stats — it's just a relabel. */}
   986→          {striker && nonStriker && onSwapEnds && (
   987→            <button
   988→              type="button"
   989→              onClick={onSwapEnds}
   990→              title="Swap striker ↔ non-striker (override the auto strike-rotation)"
   991→              aria-label="Swap striker and non-striker ends"
   992→              className="absolute -left-[14px] top-1/2 grid h-7 w-7 -translate-y-1/2 place-items-center rounded-full border border-white/20 bg-slate-700 text-white shadow-md transition-colors hover:bg-slate-600"
   993→            >
   994→              <ArrowLeftRight className="h-3.5 w-3.5" />
   995→            </button>
   996→          )}
   997→        </div>
   998→        <div className="col-span-2 rounded-md bg-white/10 p-2.5">
   999→          <div className="flex items-center justify-between">
  1000→            <div className="text-[10px] uppercase text-slate-300">Bowler</div>
  1001→            {bowler && onChangeBowler && (
  1002→              <ChangeButton
  1003→                onClick={onChangeBowler}
  1004→                disabled={!canChangeBowler}
  1005→                hint={
  1006→                  canChangeBowler
  1007→                    ? "Re-pick bowler"
  1008→                    : "Wait for end of over (or use Undo)"
  1009→                }
  1010→              />
  1011→            )}
  1012→          </div>
  1013→          <div className="flex items-center justify-between">
  1014→            <span className="font-bold">{bowler?.name ?? "—"}</span>
  1015→            <span className="text-slate-300">
  1016→              {bowler
  1017→                ? `${oversString(bowler.balls)} - ${bowler.maidens} - ${bowler.runsConceded} - ${bowler.wickets}`
  1018→                : "—"}
  1019→            </span>
  1020→          </div>
  1021→        </div>
  1022→      </div>
  1023→    </div>
  1024→  )
  1025→}

  1065→  onWicket: () => void

  1127→const WICKET_TYPE_OPTIONS: { value: WicketType; label: string; hint: string }[] = [
  1128→  { value: "bowled", label: "Bowled", hint: "b Bowler" },
  1129→  { value: "caught", label: "Caught", hint: "c Fielder b Bowler" },
  1130→  { value: "lbw", label: "LBW", hint: "lbw b Bowler" },
  1131→  { value: "runout", label: "Run Out", hint: "run out (Fielder)" },
  1132→  { value: "stumped", label: "Stumped", hint: "st Keeper b Bowler" },
  1133→  { value: "hitwkt", label: "Hit Wicket", hint: "hit wkt b Bowler" },
  1134→  { value: "retiredHurt", label: "Retired Hurt", hint: "not out — may return" },

  1138→/** Build the dismissal summary string from the structured inputs (this is
  1139→ *  what the public scorecard renders next to the batsman's name). */
  1140→function buildDismissalText(args: {
  1141→  wicketType: WicketType
  1142→  bowlerName?: string
  1143→  fielderName?: string
  1144→}): string {
  1145→  const bowler = args.bowlerName ?? "?"
  1146→  const fielder = args.fielderName
  1147→  switch (args.wicketType) {
  1148→    case "bowled":
  1149→      return `b ${bowler}`
  1150→    case "caught":
  1151→      // Caught & bowled when no fielder is selected OR the fielder is the bowler.
  1152→      if (!fielder || fielder === bowler) return `c &b ${bowler}`
  1153→      return `c ${fielder} b ${bowler}`
  1154→    case "lbw":
  1155→      return `lbw b ${bowler}`
  1156→    case "stumped":
  1157→      return fielder ? `st ${fielder} b ${bowler}` : `st ${bowler}`
  1158→    case "runout":
  1159→      return fielder ? `run out (${fielder})` : `run out`
  1160→    case "hitwkt":
  1161→      return `hit wkt b ${bowler}`
  1162→    case "retiredHurt":
  1163→      return `retired hurt`
  1164→    case "retiredOut":
  1165→      return `retired out`
  1166→  }
  1167→}

  1169→function WicketDialog({
  1170→  open,
  1171→  onOpenChange,
  1172→  striker,
  1173→  nonStriker,
  1174→  bowler,
  1175→  fieldingPlayers,
  1176→  freeHit,
  1177→  onConfirm,
  1178→}: {
  1179→  open: boolean
  1180→  onOpenChange: (v: boolean) => void
  1181→  striker: { id: string; name: string }
  1182→  nonStriker: { id: string; name: string }
  1183→  bowler: { id: string; name: string }
  1184→  fieldingPlayers: { id: string; name: string; role?: string }[]
  1185→  /** True when the current ball is a free hit — bowler-credited dismissals
  1186→   *  (bowled/caught/lbw/stumped/hit-wkt) will be REJECTED by the engine.
  1187→   *  The dialog shows a hint banner so the admin knows to pick Run Out or
  1188→   *  Retired Out instead. */
  1189→  freeHit?: boolean
  1190→  onConfirm: (ev: {
  1191→    kind: "wicket"
  1192→    wicketType: WicketType
  1193→    outBatsmanId: string
  1194→    fielderIds: string[]
  1195→    dismissal: string
  1196→  }) => void
  1197→}) {
  1198→  const [wicketType, setWicketType] = useState<WicketType>("bowled")
  1199→  const [outBatsmanId, setOutBatsmanId] = useState<string>(striker.id)
  1200→  const [fielderId, setFielderId] = useState<string>("")
  1201→  // Note: this component is conditionally rendered by the parent only when
  1202→  // `wicketOpen` is true, so it remounts fresh each time the user opens the
  1203→  // dialog — useState initial values act as the "reset on open".
  1204→
  1205→  // The wicket-keeper on the fielding side (auto-prefilled for stumped).
  1206→  // Uses the explicit `isKeeper` flag on the Player model — was previously
  1207→  // a brittle regex on the role string.
  1208→  const keeperId = useMemo(() => {
  1209→    const k = fieldingPlayers.find((p) => p.isKeeper)
  1210→    return k?.id ?? ""
  1211→  }, [fieldingPlayers])
  1212→
  1213→  // Derived state — avoid setState-in-effect by computing the effective
  1214→  // selections during render. The picker's `value` reads from these, so the
  1215→  // UI stays in sync without needing an effect to "correct" the user's
  1216→  // last selection when the dismissal type changes.
  1217→  // • For any dismissal type other than run-out, the batsman out is always
  1218→  //   the striker (run-out can happen at the non-striker's end).
  1219→  // • For stumped with no fielder selected, auto-prefill the keeper.
  1220→  const effectiveOutBatsmanId =
  1221→    wicketType === "runout" ? outBatsmanId : striker.id
  1222→  const effectiveFielderId =
  1223→    wicketType === "stumped" && !fielderId && keeperId ? keeperId : fielderId
  1224→
  1225→  const needsFielder =
  1226→    wicketType === "caught" ||
  1227→    wicketType === "stumped" ||
  1228→    wicketType === "runout"
  1229→  const showBatsmanSelect = wicketType === "runout"
  1230→
  1231→  const fielderName = fieldingPlayers.find(
  1232→    (p) => p.id === effectiveFielderId
  1233→  )?.name
  1234→  const dismissal = buildDismissalText({
  1235→    wicketType,
  1236→    bowlerName: bowler.name,
  1237→    fielderName,
  1238→  })
  1239→
  1240→  const outBatsmanName =
  1241→    effectiveOutBatsmanId === striker.id
  1242→      ? striker.name
  1243→      : effectiveOutBatsmanId === nonStriker.id
  1244→        ? nonStriker.name
  1245→        : "?"
  1246→
  1247→  // Free-hit-derived warning: when the current ball is a free hit AND the
  1248→  // selected dismissal type is bowler-credited, the engine will REJECT the
  1249→  // wicket (batsman survives). Show a warning so the admin knows.
  1250→  const freeHitWillReject =
  1251→    !!freeHit &&
  1252→    BOWLER_CREDITED_WICKETS.has(wicketType) &&
  1253→    wicketType !== "retired"
  1254→
  1255→  function handleConfirm() {
  1256→    onConfirm({
  1257→      kind: "wicket",
  1258→      wicketType,
  1259→      outBatsmanId: effectiveOutBatsmanId,
  1260→      fielderIds: effectiveFielderId ? [effectiveFielderId] : [],
  1261→      dismissal,
  1262→    })
  1263→  }
  1264→
  1265→  return (
  1266→    <Dialog open={open} onOpenChange={onOpenChange}>
  1267→      <DialogContent className="sm:max-w-lg">
  1268→        <DialogHeader>
  1269→          <DialogTitle className="flex items-center gap-2">
  1270→            <span className="grid h-7 w-7 place-items-center rounded-md bg-red-600 text-white">
  1271→              <Activity className="h-4 w-4" />
  1272→            </span>
  1273→            Dismiss {outBatsmanName}
  1274→          </DialogTitle>
  1275→          <DialogDescription>
  1276→            Pick the dismissal type. If caught, stumped or run out, choose the
  1277→            fielder(s) involved. The dismissal text updates live in the preview.
  1278→          </DialogDescription>
  1279→        </DialogHeader>
  1280→
  1281→        {/* Free-hit banner — appears when the current ball is a free hit. */}
  1282→        {freeHit && (
  1283→          <div className="rounded-md border border-amber-400 bg-amber-50 px-3 py-2 text-xs dark:bg-amber-950/40">
  1284→            <div className="flex items-center gap-1.5 font-bold uppercase tracking-wide text-amber-800 dark:text-amber-300">
  1285→              <span className="h-1.5 w-1.5 animate-pulse rounded-full bg-amber-600" />
  1286→              Free hit in effect
  1287→            </div>
  1288→            <p className="mt-1 text-amber-700 dark:text-amber-400">
  1289→              Bowler-credited dismissals (bowled, caught, lbw, stumped, hit
  1290→              wicket) are <strong>not allowed</strong> on a free hit — the
  1291→              batsman survives and only run-out / retired-out stand.
  1292→              {freeHitWillReject
  1293→                ? " Picking this dismissal will be rejected by the engine — pick Run Out instead."
  1294→                : ""}
  1295→            </p>
  1296→          </div>
  1297→        )}
  1298→
  1299→        <div className="grid gap-3 py-2">
  1300→          {/* Dismissal type */}
  1301→          <div className="grid gap-1.5">
  1302→            <Label>Dismissal type</Label>
  1303→            <Select value={wicketType} onValueChange={(v) => setWicketType(v as WicketType)}>
  1304→              <SelectTrigger className="w-full">
  1305→                <SelectValue />
  1306→              </SelectTrigger>
  1307→              <SelectContent>
  1308→                {WICKET_TYPE_OPTIONS.map((opt) => (
  1309→                  <SelectItem key={opt.value} value={opt.value}>
  1310→                    <span className="font-medium">{opt.label}</span>
  1311→                    <span className="ml-2 text-[10px] text-muted-foreground">{opt.hint}</span>
  1312→                  </SelectItem>
  1313→                ))}
  1314→              </SelectContent>
  1315→            </Select>
  1316→          </div>
  1317→
  1318→          {/* Batsman out — only shown for run-out */}
  1319→          {showBatsmanSelect && (
  1320→            <div className="grid gap-1.5">
  1321→              <Label>Which batsman is out?</Label>
  1322→              <Select
  1323→                value={effectiveOutBatsmanId}
  1324→                onValueChange={setOutBatsmanId}
  1325→              >
  1326→                <SelectTrigger className="w-full">
  1327→                  <SelectValue />
  1328→                </SelectTrigger>
  1329→                <SelectContent>
  1330→                  <SelectItem value={striker.id}>
  1331→                    Striker — {striker.name}
  1332→                  </SelectItem>
  1333→                  <SelectItem value={nonStriker.id}>
  1334→                    Non-striker — {nonStriker.name}
  1335→                  </SelectItem>
  1336→                </SelectContent>
  1337→              </Select>
  1338→            </div>
  1339→          )}
  1340→
  1341→          {/* Fielder(s) involved — for caught, stumped, run out */}
  1342→          {needsFielder && (
  1343→            <div className="grid gap-1.5">
  1344→              <Label>
  1345→                {wicketType === "caught"
  1346→                  ? "Catcher"
  1347→                  : wicketType === "stumped"
  1348→                    ? "Wicket-keeper"
  1349→                    : "Run-out fielder"}
  1350→              </Label>
  1351→              <Select value={effectiveFielderId} onValueChange={setFielderId}>
  1352→                <SelectTrigger className="w-full">
  1353→                  <SelectValue placeholder="(direct hit / no fielder)" />
  1354→                </SelectTrigger>
  1355→                <SelectContent>
  1356→                  <SelectItem value="(direct-hit)">(direct hit / no fielder)</SelectItem>
  1357→                  {fieldingPlayers.map((p) => (
  1358→                    <SelectItem key={p.id} value={p.id}>
  1359→                      {p.name}
  1360→                      {p.isKeeper && " †"}
  1361→                      {p.isCaptain && " (c)"}
  1362→                    </SelectItem>
  1363→                  ))}
  1364→                </SelectContent>
  1365→              </Select>
  1366→              {wicketType === "stumped" && keeperId && !fielderId && (
  1367→                <p className="text-[11px] text-muted-foreground">
  1368→                  Wicket-keeper auto-prefilled (you can change it).
  1369→                </p>

  1371→            </div>

  1380→              <span className="font-semibold">{outBatsmanName}</span>{" "}
  1381→              <span className="text-muted-foreground">{dismissal}</span>

  1383→          </div>
  1384→        </div>
  1385→
  1386→        <DialogFooter>

  1403→function Picker({
  1404→  title,
  1405→  players,
  1406→  placeholder,
  1407→  onPick,
  1408→  confirmLabel = "Confirm",
  1409→}: {
  1410→  title: string
  1411→  players: { id: string; name: string }[]
  1412→  placeholder: string
  1413→  onPick: (id: string) => void
  1414→  /** Label for the commit button. Use "Add batsman" / "Set bowler" etc.
  1415→   *  Defaults to "Confirm". */
  1416→  confirmLabel?: string
  1417→}) {
  1418→  const [val, setVal] = useState("")
  1419→  const selected = players.find((p) => p.id === val)
  1420→  // Decouple selection from commit: changing the dropdown only updates the
  1421→  // local `val` state. The commit (calling onPick) requires an explicit
  1422→  // button click — so a misclick in the dropdown doesn't write to the live
  1423→  // state. If the user picks the wrong name in the dropdown, they can just
  1424→  // pick a different one before pressing Confirm.
  1425→  return (
  1426→    <div className="rounded-lg border-2 border-amber-400 bg-amber-50 p-4 dark:bg-amber-950/40">
  1427→      <div className="mb-2 flex items-center gap-2 text-amber-800 dark:text-amber-300">
  1428→        <Plus className="h-4 w-4" />
  1429→        <h3 className="text-sm font-bold uppercase tracking-wide">{title}</h3>
  1430→      </div>
  1431→      {players.length === 0 ? (
  1432→        // Empty-state — the team has no bench players available (rare, but
  1433→        // happens if a team was created without enough players seeded, or
  1434→        // everyone has already batted). Tell the admin how to fix it instead
  1435→        // of opening a broken dropdown.
  1436→        <div className="rounded-md border border-dashed bg-amber-100/60 px-3 py-4 text-center text-xs text-amber-800 dark:bg-amber-950/60 dark:text-amber-300">
  1437→          No eligible players left in the squad.
  1438→          <br />
  1439→          Add more players to the team via{" "}
  1440→          <span className="font-semibold">Admin → Teams → bulk-add players</span>{" "}
  1441→          (the green Users icon on the team row), then come back here.
  1442→        </div>
  1443→      ) : (
  1444→        <>
  1445→          <Select value={val} onValueChange={setVal}>
  1446→            <SelectTrigger className="w-full">
  1447→              <SelectValue placeholder={placeholder} />
  1448→            </SelectTrigger>
  1449→            <SelectContent>
  1450→              {players.map((p) => (
  1451→                <SelectItem key={p.id} value={p.id}>
  1452→                  {p.name}
  1453→                </SelectItem>
  1454→              ))}
  1455→            </SelectContent>
  1456→          </Select>
  1457→          <div className="mt-2 flex items-center justify-between gap-2">
  1458→            <span className="text-xs text-amber-700 dark:text-amber-400">
  1459→              {selected ? (
  1460→                <>
  1461→                  Selected: <span className="font-semibold">{selected.name}</span>
  1462→                </>
  1463→              ) : (
  1464→                "Pick a player above, then confirm."
  1465→              )}
  1466→            </span>
  1467→            <Button
  1468→              size="sm"
  1469→              disabled={!val}
  1470→              onClick={() => val && onPick(val)}
  1471→              className="gap-1.5"
  1472→            >
  1473→              <Plus className="h-3.5 w-3.5" />
  1474→              {confirmLabel}
  1475→            </Button>
  1476→          </div>
  1477→        </>
  1478→      )}
  1479→    </div>
  1480→  )
  1481→}
