   355→function StatusBadge({ status, label }: { status: "live" | "final" | "upcoming"; label: string }) {
   356→  const color =
   357→    status === "live"
   358→      ? "text-cricket-red"
   359→      : status === "final"
   360→        ? "text-league-green"
   361→        : "text-league-muted"
   362→  return (
   363→    <Badge
   364→      variant="outline"
   365→      className={cn(
   366→        "border-0 bg-transparent px-0 text-[10px] font-black uppercase tracking-[.7px] p-0",
   367→        color,
   368→      )}
   369→    >
   370→      {label}
   371→    </Badge>
   372→  )
