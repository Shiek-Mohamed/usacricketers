   190→   190→/** A small colored pill for one ball in the recent-balls strip.
   191→   191→ *  Color coding (ESPNcricinfo-ish):
   192→   192→ *    dot=gray · 1-3 runs=green · 4=blue · 6=purple · W=red · wide/noball=amber
   193→   193→ */
   194→   194→function ballPillStyle(entry: CommentaryItem): {
   195→   195→  cls: string
   196→   196→  label: string
   197→   197→} {
   198→   198→  const k = entry.kind
   199→   199→  if (k === "wicket") return { cls: "bg-red-600 text-white", label: "W" }
   200→   200→  if (k === "six") return { cls: "bg-purple-600 text-white", label: "6" }
   201→   201→  if (k === "boundary") return { cls: "bg-blue-600 text-white", label: "4" }
   202→   202→  if (k === "extra") {
   203→   203→    // Distinguish wide vs noball from the text when possible.
   204→   204→    const t = entry.text.toLowerCase()
   205→   205→    if (t.includes("wide")) return { cls: "bg-amber-500 text-amber-950", label: "wd" }
   206→   206→    if (t.includes("no ball")) return { cls: "bg-amber-500 text-amber-950", label: "nb" }
   207→   207→    return { cls: "bg-amber-500 text-amber-950", label: "+" }
   208→   208→  }
   209→   209→  if (k === "run") {
   210→   210→    // Extract the run count from text like "3 runs — Smith"
   211→   211→    const m = entry.text.match(/^(\d+)/)
   212→   212→    const r = m ? Number(m[1]) : 0
   213→   213→    return { cls: "bg-emerald-600 text-white", label: String(r) }
   214→   214→  }
   215→   215→  if (k === "dot") return { cls: "bg-muted text-muted-foreground", label: "•" }
   216→   216→  // Manual / info / over end / unknown — render as a subtle gray dot.
   217→   217→  return { cls: "bg-muted text-muted-foreground", label: "•" }
   218→   218→}

   220→   220→/** Recent balls strip — shows the last 6 ball-events as colored pills.
   221→   221→ *  Uses the liveState.commentary feed (which is already newest-first) so
   222→   222→ *  we can pick out the most recent deliveries with their proper kinds. */
   223→   223→function RecentBallsStrip({
   224→   224→  commentary,
   225→   225→}: {
   226→   226→  commentary: CommentaryItem[]
   227→   227→}) {
   228→   228→  // Pick only ball-event kinds — skip "info", "over" (end-of-over markers)
   229→   229→  // and "manual" (commentator notes). These 6 kinds cover every legal
   230→   230→  // delivery + extras.
   231→   231→  const ballKinds = new Set(["run", "boundary", "six", "wicket", "extra", "dot"])
   232→   232→  const recent = commentary.filter((c) => ballKinds.has(c.kind)).slice(0, 6)
   233→   233→
   234→   234→  if (recent.length === 0) {
   235→   235→    return (
   236→   236→      <div className="flex items-center gap-2 rounded-lg border bg-card px-3 py-2 text-xs text-muted-foreground">
   237→   237→        <span className="text-[10px] font-bold uppercase tracking-wide">This over</span>
   238→   238→        <span>No balls bowled yet</span>
   239→   239→      </div>
   240→   240→    )
   241→   241→  }
   242→   242→
   243→   243→  return (
   244→   244→    <div className="flex items-center gap-2 rounded-lg border bg-card px-3 py-2">
   245→   245→      <span className="text-[10px] font-bold uppercase tracking-wide text-muted-foreground">
   246→   246→        Recent
   247→   247→      </span>
   248→   248→      <div className="flex items-center gap-1.5">
   249→   249→        {/* Reverse so the most-recent ball sits on the right (reading order
   250→   250→            matches the natural "left→right = older→newer" of an over). */}
   251→   251→        {[...recent].reverse().map((entry, i) => {
   252→   252→          const { cls, label } = ballPillStyle(entry)
   253→   253→          return (
   254→   254→            <span
   255→   255→              key={entry.id + i}
   256→   256→              className={cn(
   257→   257→                "grid h-7 w-7 place-items-center rounded-full text-xs font-extrabold tabular-nums",
   258→   258→                cls
   259→   259→              )}
   260→   260→              title={`${entry.over} · ${entry.text}`}
   261→   261→            >
   262→   262→              {label}
   263→   263→            </span>
   264→   264→          )
   265→   265→        })}
   266→   266→      </div>
   267→   267→    </div>
   268→   268→  )
   269→   269→}

   383→   383→/** Compact score + partnership + RRR summary bar at the top of the Live tab. */
   384→   384→function LiveSummaryBar({ state }: { state: LiveMatchState }) {
   385→   385→  const inn = state.innings[state.currentInnings]
   386→   386→  if (!inn) return null
   387→   387→  const overs = oversString(inn.balls)
   388→   388→  const crr = inn.balls > 0 ? (inn.runs / inn.balls * 6).toFixed(2) : "0.00"
   389→   389→  // Partnership runs = current total − runs on the board when this
   390→   390→  // partnership began. Balls faced = current balls − balls then.
   391→   391→  const pRuns = Math.max(0, inn.runs - inn.partnershipStartRuns)
   392→   392→  const pBalls = Math.max(0, inn.balls - inn.partnershipStartBalls)
   393→   393→  const target = inn.target
   394→   394→  const ballsLeft = state.maxOvers > 0 ? state.maxOvers * 6 - inn.balls : null
   395→   395→  const runsNeeded = target ? target - inn.runs : null
   396→   396→  const rrr =
   397→   397→    target && ballsLeft && ballsLeft > 0
   398→   398→      ? ((runsNeeded! / ballsLeft) * 6).toFixed(2)
   399→   399→      : null
   400→   400→
   401→   401→  return (
   402→   402→    <div className="rounded-lg border bg-gradient-to-br from-slate-900 to-slate-800 p-3 text-white sm:p-4">
   403→   403→      <div className="flex flex-wrap items-end justify-between gap-3">
   404→   404→        <div className="min-w-0">
   405→   405→          <div className="text-[11px] uppercase tracking-wide text-slate-400">
   406→   406→            {inn.battingTeamShortName} · Innings {state.currentInnings + 1}
   407→   407→            {inn.target ? ` · target ${inn.target}` : ""}
   408→   408→          </div>
   409→   409→          <div className="mt-1 text-3xl font-extrabold tabular-nums sm:text-4xl">
   410→   410→            {inn.runs}/{inn.wickets}
   411→   411→            <span className="ml-2 text-base font-normal text-slate-300">
   412→   412→              ({overs} ov)
   413→   413→            </span>
   414→   414→            {inn.freeHit && (
   415→   415→              <span className="ml-3 inline-flex items-center gap-1 rounded-full bg-amber-500 px-2.5 py-0.5 text-[10px] font-extrabold uppercase tracking-wide text-amber-950 animate-pulse">
   416→   416→                Free hit
   417→   417→              </span>
   418→   418→            )}
   419→   419→          </div>
   420→   420→        </div>
   421→   421→        <div className="text-right text-xs text-slate-300">
   422→   422→          <div>CRR <span className="font-bold text-white tabular-nums">{crr}</span></div>
   423→   423→          {rrr && (
   424→   424→            <div>
   425→   425→              RRR <span className="font-bold text-white tabular-nums">{rrr}</span>
   426→   426→            </div>
   427→   427→          )}

   490→   490→/** Group commentary by over, preserving the newest-first order inside
   491→   491→ *  each over. Returns an array of { over, entries } blocks sorted so
   492→   492→ *  the most recent over appears first. */
   493→   493→function groupByOver(commentary: CommentaryItem[]): {
   494→   494→  over: string
   495→   495→  entries: CommentaryItem[]
   496→   496→}[] {
   497→   497→  const groups = new Map<string, CommentaryItem[]>()
   498→   498→  for (const entry of commentary) {
   499→   499→    // Use the integer part of the over string ("12.3" → "12", "12" → "12").
   500→   500→    // The "over" entry at end-of-over has over like "12" (no decimal).
   501→   501→    const overNum = entry.over.split(".")[0] ?? "0"
   502→   502→    const key = overNum
   503→   503→    if (!groups.has(key)) groups.set(key, [])
   504→   504→    groups.get(key)!.push(entry)
   505→   505→  }
   506→   506→  // Sort group keys numerically descending (newest over first).
   507→   507→  const sorted = [...groups.entries()].sort((a, b) => {
   508→   508→    const an = parseInt(a[0], 10) || 0
   509→   509→    const bn = parseInt(b[0], 10) || 0
   510→   510→    return bn - an
   511→   511→  })
   512→   512→  return sorted.map(([over, entries]) => ({ over, entries }))
   513→   513→}
