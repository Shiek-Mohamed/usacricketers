    68→function emptyForm(): FormState {
    69→  const year = new Date().getFullYear().toString()

    71→  const end = new Date(Date.now() + 1000 * 60 * 60 * 24 * 60)
    72→    .toISOString()
    73→    .slice(0, 10)
    74→  return {
    75→    name: "",
    76→    shortName: "",
    77→    format: "T20",
    78→    season: year,
    79→    status: "upcoming",
    80→    startDate: today,
    81→    endDate: end,
    82→    description: "",
    83→  }
    84→}

    86→function toDateInput(iso: string): string {
    87→  try {
    88→    return new Date(iso).toISOString().slice(0, 10)
    89→  } catch {
    90→    return ""
    91→  }
    92→}

    94→async function jsonOrThrow<T>(res: Response): Promise<T> {
    95→  if (!res.ok) {
    96→    let msg = `Request failed: ${res.status}`
    97→    try {
    98→      const body = await res.json()
    99→      if (body?.error) msg = body.error
   100→    } catch {
   101→      // ignore JSON parse errors
   102→    }
   103→    throw new Error(msg)
   104→  }
   105→  return (await res.json()) as T
   106→}
