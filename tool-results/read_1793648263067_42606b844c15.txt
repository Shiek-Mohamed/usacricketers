   100→   100→/** Last innings total for a team in a match, e.g. "217/4". */
   101→   101→function inningsTotal(match: Match, teamId: string): string {
   102→   102→  const sc = match.scorecard
   103→   103→  if (!sc?.innings) return ""
   104→   104→  const inn = sc.innings.filter((i) => i.teamId === teamId)
   105→   105→  if (!inn.length) return ""
   106→   106→  const last = inn[inn.length - 1]
   107→   107→  return `${last.totalRuns}/${last.totalWickets}`
   108→   108→}
