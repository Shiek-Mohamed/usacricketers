    76→async function jsonOrThrow<T>(res: Response): Promise<T> {
    77→  if (!res.ok) {
    78→    let msg = `Request failed: ${res.status}`
    79→    try {
    80→      const body = await res.json()
    81→      if (body?.error) msg = body.error
    82→    } catch {
    83→      // ignore JSON parse errors
    84→    }
    85→    throw new Error(msg)
    86→  }
    87→  return (await res.json()) as T
    88→}

   101→function emptyMatchForm(): MatchForm {
   102→  const now = new Date()
   103→  // local datetime string YYYY-MM-DDTHH:MM
   104→  const pad = (n: number) => String(n).padStart(2, "0")
   105→  const local = `${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(
   106→    now.getDate()
   107→  )}T${pad(now.getHours())}:${pad(now.getMinutes())}`
   108→  return {
   109→    team1Id: "",
   110→    team2Id: "",
   111→    format: "T20",
   112→    matchNo: "",
   113→    status: "upcoming",
   114→    venue: "",
   115→    city: "",
   116→    startDate: local,
   117→  }
   118→}
   119→
   120→function toDatetimeLocal(iso: string): string {
   121→  try {
   122→    const d = new Date(iso)
   123→    const pad = (n: number) => String(n).padStart(2, "0")
   124→    return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(
   125→      d.getDate()
   126→    )}T${pad(d.getHours())}:${pad(d.getMinutes())}`
   127→  } catch {
   128→    return ""
   129→  }
   130→}
