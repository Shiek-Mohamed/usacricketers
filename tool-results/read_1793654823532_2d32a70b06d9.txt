   141→   141→      return tag === "input" || tag === "textarea" || tag === "select" ||
   142→   142→        el.getAttribute("role") === "combobox" ||

   145→   145→    function onKey(e: KeyboardEvent) {

   150→   150→      if (!liveState.strikerId || !liveState.nonStrikerId || !liveState.bowlerId) return
   151→   151→      const k = e.key.toLowerCase()

   254→   254→  function handleUndo() {
   255→   255→    if (history.length === 0) {
   256→   256→      toast.info("Nothing to undo")
   257→   257→      return
   258→   258→    }

   262→   262→    persist(prev)
   263→   263→    toast.success("Undid last action")

   266→   266→  function startInnings(
   267→   267→    battingFirstId: string,
   268→   268→    strikerId: string,
   269→   269→    nonStrikerId: string,
   270→   270→    bowlerId: string
   271→   271→  ) {
   272→   272→    const battingPlayers = playersByTeam[battingFirstId] ?? []
   273→   273→    const bowlingPlayers =
   274→   274→      playersByTeam[battingFirstId === match!.team1Id ? match!.team2Id : match!.team1Id] ?? []

   276→   276→    const nonStriker = battingPlayers.find((p) => p.id === nonStrikerId)
   277→   277→    const bowler = bowlingPlayers.find((p) => p.id === bowlerId)
   278→   278→    if (!striker || !nonStriker || !bowler) {
   279→   279→      toast.error("Select striker, non-striker and bowler")
   280→   280→      return
   281→   281→    }
   282→   282→    let state = initState(match!, battingFirstId)
   283→   283→    // Pre-populate the squad with the full batting-team roster so the
   284→   284→    // "Did not bat" section of the public scorecard lists bench players.
   285→   285→    // Captain/keeper tags come from the explicit Player model fields
   286→   286→    // (was previously a brittle regex on the role string).
   287→   287→    for (const p of battingPlayers) {
   288→   288→      state = addSquadMember(state, p.id, p.name, {
   289→   289→        isCaptain: !!p.isCaptain,
   290→   290→        isKeeper: !!p.isKeeper,
   291→   291→      })
   292→   292→    }
   293→   293→    state = addBatsman(state, striker.id, striker.name, {
   294→   294→      isCaptain: !!striker.isCaptain,
   295→   295→      isKeeper: !!striker.isKeeper,
   296→   296→    })
   297→   297→    state = addBatsman(state, nonStriker.id, nonStriker.name, {
   298→   298→      isCaptain: !!nonStriker.isCaptain,
   299→   299→      isKeeper: !!nonStriker.isKeeper,
   300→   300→    })
   301→   301→    state = setBatsmen(state, striker.id, nonStriker.id)
   302→   302→    state = setBowler(state, bowler.id, bowler.name)
   303→   303→    setHistory([])
   304→   304→    pushState(state)
   305→   305→    persist(state)
   306→   306→    toast.success("Innings started — status set to LIVE")
   307→   307→  }

   309→   309→  function bringInBatsman(playerId: string) {
   310→   310→    if (!liveState) return
   311→   311→    const battingPlayers = playersByTeam[liveState.innings[liveState.currentInnings].battingTeamId] ?? []
   312→   312→    const p = battingPlayers.find((x) => x.id === playerId)
   313→   313→    if (!p) return
   314→   314→    let state = addBatsman(liveState, p.id, p.name, {
   315→   315→      isCaptain: !!p.isCaptain,
   316→   316→      isKeeper: !!p.isKeeper,
   317→   317→    })
   318→   318→    // Fill whichever slot is currently empty — the striker's end if a normal
   319→   319→    // wicket fell (strikerId === null), or the non-striker's end if a run-out
   320→   320→    // happened at the non-striker's end (nonStrikerId === null). If both are
   321→   321→    // empty (innings restart, edge case), default to striker.
   322→   322→    const fillStriker = liveState.strikerId === null
   323→   323→    const fillNonStriker = !fillStriker && liveState.nonStrikerId === null
   324→   324→    state = {
   325→   325→      ...state,
   326→   326→      strikerId: fillStriker ? p.id : state.strikerId,
   327→   327→      nonStrikerId: fillNonStriker ? p.id : state.nonStrikerId,
   328→   328→    }
   329→   329→    setHistory((h) => [...h, liveState].slice(-50))
   330→   330→    pushState(state)
   331→   331→    persist(state)
   332→   332→    toast.success(`${p.name} in`)
   333→   333→  }

   380→   380→  /** Swap the striker and non-striker ends. Used when the auto strike-rotation
   381→   381→   *  logic disagrees with reality (e.g. batsmen crossed on a caught-and-
   382→   382→   *  bowled, odd run on a wide, runner substituted, etc.). Allowed at any
   383→   383→   *  time — doesn't affect stats, just relabels who's on strike. */
   384→   384→  function swapEnds() {
   385→   385→    if (!liveState) return
   386→   386→    if (!liveState.strikerId || !liveState.nonStrikerId) {
   387→   387→      toast.error("Need both batsmen at the crease to swap ends")
   388→   388→      return
   389→   389→    }
   390→   390→    setHistory((h) => [...h, liveState].slice(-50))
   391→   391→    const next = {
   392→   392→      ...liveState,
   393→   393→      strikerId: liveState.nonStrikerId,
   394→   394→      nonStrikerId: liveState.strikerId,
   395→   395→    }
   396→   396→    pushState(next)
   397→   397→    persist(next)
   398→   398→    toast.success("Ends swapped")
   399→   399→  }

   408→   408→  /** Test-match declaration: captain ends the current innings without being
   409→   409→   *  all out. No-op for limited-overs matches (those don't have declarations).
   410→   410→   *  Pushes the prev state to history (so Undo can recover) then runs the
   411→   411→   *  engine's declareInnings which marks `declared: true` + sets up the
   412→   412→   *  next innings / computes the result. */
   413→   413→  function handleDeclare() {
   414→   414→    if (!liveState) return
   415→   415→    if (liveState.maxOvers > 0) {
   416→   416→      toast.error("Declarations are only available in Test matches")
   417→   417→      return
   418→   418→    }
   419→   419→    if (liveState.status !== "live") {
   420→   420→      toast.error("Can't declare — innings already ended")
   421→   421→      return
   422→   422→    }
   423→   423→    setHistory((h) => [...h, liveState].slice(-50))
   424→   424→    const next = declareInnings(liveState)
   425→   425→    pushState(next)
   426→   426→    persist(next)
   427→   427→    toast.success("Innings declared")
   428→   428→  }

   430→   430→  /** Enforce the follow-on in a Test match. Only valid when Team A leads by
   431→   431→   *  200+ runs after Team B's 1st innings. Team B bats again immediately. */
   432→   432→  function handleFollowOn() {
   433→   433→    if (!liveState) return
   434→   434→    setHistory((h) => [...h, liveState].slice(-50))
   435→   435→    const next = enforceFollowOn(liveState)
   436→   436→    if (next === liveState) {
   437→   437→      toast.error("Follow-on requires a 200+ run lead in a Test after innings 1")
   438→   438→      return
   439→   439→    }
   440→   440→    pushState(next)
   441→   441→    persist(next)
   442→   442→    toast.success("Follow-on enforced")
   443→   443→  }

   445→   445→  /** Start a super-over after a tied limited-overs match. Creates 2 new
   446→   446→   *  1-over innings (one per team). */
   447→   447→  function handleSuperOver() {
   448→   448→    if (!liveState) return
   449→   449→    setHistory((h) => [...h, liveState].slice(-50))

   874→   874→function LiveScoreboard({
   875→   875→  state,
   876→   876→  onChangeStriker,
   877→   877→  onChangeNonStriker,
   878→   878→  onChangeBowler,
   879→   879→  onSwapEnds,
   880→   880→}: {
   881→   881→  state: LiveMatchState
   882→   882→  onChangeStriker?: () => void
   883→   883→  onChangeNonStriker?: () => void
   884→   884→  onChangeBowler?: () => void
   885→   885→  /** Swap striker ↔ non-striker — override the auto strike-rotation logic
   886→   886→   *  when it disagrees with reality (caught-and-bowled crossing, etc.).
   887→   887→   *  Allowed at any time (not over-boundary-gated) — doesn't affect stats. */
   888→   888→  onSwapEnds?: () => void
   889→   889→}) {
   890→   890→  const inn = state.innings[state.currentInnings]
   891→   891→  const striker = inn.batsmen.find((b) => b.id === state.strikerId)
   892→   892→  const nonStriker = inn.batsmen.find((b) => b.id === state.nonStrikerId)
   893→   893→  const bowler = inn.bowlers.find((b) => b.id === state.bowlerId)
   894→   894→  const overs = oversString(inn.balls)
   895→   895→  const runRate = inn.balls > 0 ? (inn.runs / inn.balls * 6).toFixed(2) : "0.00"
   896→   896→  const target = inn.target
   897→   897→  const ballsLeft = state.maxOvers > 0 ? state.maxOvers * 6 - inn.balls : null
   898→   898→  const runsNeeded = target ? target - inn.runs : null
   899→   899→  const reqRate =
   900→   900→    target && ballsLeft && ballsLeft > 0
   901→   901→      ? ((runsNeeded! / ballsLeft) * 6).toFixed(2)
   902→   902→      : null
   903→   903→
   904→   904→  // A slot can be safely "Changed" only at over-boundaries — mid-over swaps

  1138→  1138→/** Build the dismissal summary string from the structured inputs (this is
  1139→  1139→ *  what the public scorecard renders next to the batsman's name). */
  1140→  1140→function buildDismissalText(args: {
  1141→  1141→  wicketType: WicketType
  1142→  1142→  bowlerName?: string
  1143→  1143→  fielderName?: string
  1144→  1144→}): string {
  1145→  1145→  const bowler = args.bowlerName ?? "?"
  1146→  1146→  const fielder = args.fielderName
  1147→  1147→  switch (args.wicketType) {
  1148→  1148→    case "bowled":
  1149→  1149→      return `b ${bowler}`
  1150→  1150→    case "caught":
  1151→  1151→      // Caught & bowled when no fielder is selected OR the fielder is the bowler.
  1152→  1152→      if (!fielder || fielder === bowler) return `c &b ${bowler}`
  1153→  1153→      return `c ${fielder} b ${bowler}`
  1154→  1154→    case "lbw":
  1155→  1155→      return `lbw b ${bowler}`
  1156→  1156→    case "stumped":
  1157→  1157→      return fielder ? `st ${fielder} b ${bowler}` : `st ${bowler}`
  1158→  1158→    case "runout":
  1159→  1159→      return fielder ? `run out (${fielder})` : `run out`
  1160→  1160→    case "hitwkt":
  1161→  1161→      return `hit wkt b ${bowler}`
  1162→  1162→    case "retiredHurt":
  1163→  1163→      return `retired hurt`
  1164→  1164→    case "retiredOut":
  1165→  1165→      return `retired out`
  1166→  1166→  }
  1167→  1167→}

  1403→  1403→function Picker({
  1404→  1404→  title,
  1405→  1405→  players,
  1406→  1406→  placeholder,
  1407→  1407→  onPick,
  1408→  1408→  confirmLabel = "Confirm",
  1409→  1409→}: {
  1410→  1410→  title: string
  1411→  1411→  players: { id: string; name: string }[]
  1412→  1412→  placeholder: string
  1413→  1413→  onPick: (id: string) => void

  1416→  1416→  confirmLabel?: string
