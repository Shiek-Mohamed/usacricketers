   190→/** A small colored pill for one ball in the recent-balls strip.
   191→ *  Color coding (ESPNcricinfo-ish):
   192→ *    dot=gray · 1-3 runs=green · 4=blue · 6=purple · W=red · wide/noball=amber
   193→ */
   194→function ballPillStyle(entry: CommentaryItem): {
   195→  cls: string
   196→  label: string
   197→} {
   198→  const k = entry.kind
   199→  if (k === "wicket") return { cls: "bg-red-600 text-white", label: "W" }
   200→  if (k === "six") return { cls: "bg-purple-600 text-white", label: "6" }
   201→  if (k === "boundary") return { cls: "bg-blue-600 text-white", label: "4" }
   202→  if (k === "extra") {
   203→    // Distinguish wide vs noball from the text when possible.
   204→    const t = entry.text.toLowerCase()
   205→    if (t.includes("wide")) return { cls: "bg-amber-500 text-amber-950", label: "wd" }
   206→    if (t.includes("no ball")) return { cls: "bg-amber-500 text-amber-950", label: "nb" }
   207→    return { cls: "bg-amber-500 text-amber-950", label: "+" }
   208→  }
   209→  if (k === "run") {
   210→    // Extract the run count from text like "3 runs — Smith"
   211→    const m = entry.text.match(/^(\d+)/)
   212→    const r = m ? Number(m[1]) : 0
   213→    return { cls: "bg-emerald-600 text-white", label: String(r) }
   214→  }
   215→  if (k === "dot") return { cls: "bg-muted text-muted-foreground", label: "•" }
   216→  // Manual / info / over end / unknown — render as a subtle gray dot.
   217→  return { cls: "bg-muted text-muted-foreground", label: "•" }
   218→}

   220→/** Recent balls strip — shows the last 6 ball-events as colored pills.
   221→ *  Uses the liveState.commentary feed (which is already newest-first) so
   222→ *  we can pick out the most recent deliveries with their proper kinds. */
   223→function RecentBallsStrip({
   224→  commentary,
   225→}: {
   226→  commentary: CommentaryItem[]
   227→}) {
   228→  // Pick only ball-event kinds — skip "info", "over" (end-of-over markers)
   229→  // and "manual" (commentator notes). These 6 kinds cover every legal
   230→  // delivery + extras.
   231→  const ballKinds = new Set(["run", "boundary", "six", "wicket", "extra", "dot"])
   232→  const recent = commentary.filter((c) => ballKinds.has(c.kind)).slice(0, 6)
   233→
   234→  if (recent.length === 0) {
   235→    return (
   236→      <div className="flex items-center gap-2 rounded-lg border bg-card px-3 py-2 text-xs text-muted-foreground">
   237→        <span className="text-[10px] font-bold uppercase tracking-wide">This over</span>
   238→        <span>No balls bowled yet</span>
   239→      </div>
   240→    )
   241→  }
   242→
   243→  return (
   244→    <div className="flex items-center gap-2 rounded-lg border bg-card px-3 py-2">
   245→      <span className="text-[10px] font-bold uppercase tracking-wide text-muted-foreground">
   246→        Recent
   247→      </span>
   248→      <div className="flex items-center gap-1.5">
   249→        {/* Reverse so the most-recent ball sits on the right (reading order
   250→            matches the natural "left→right = older→newer" of an over). */}
   251→        {[...recent].reverse().map((entry, i) => {
   252→          const { cls, label } = ballPillStyle(entry)
   253→          return (
   254→            <span
   255→              key={entry.id + i}
   256→              className={cn(
   257→                "grid h-7 w-7 place-items-center rounded-full text-xs font-extrabold tabular-nums",
   258→                cls
   259→              )}
   260→              title={`${entry.over} · ${entry.text}`}
   261→            >
   262→              {label}
   263→            </span>
   264→          )
   265→        })}
   266→      </div>
   267→    </div>
   268→  )
   269→}

   383→/** Compact score + partnership + RRR summary bar at the top of the Live tab. */
   384→function LiveSummaryBar({ state }: { state: LiveMatchState }) {
   385→  const inn = state.innings[state.currentInnings]
   386→  if (!inn) return null
   387→  const overs = oversString(inn.balls)
   388→  const crr = inn.balls > 0 ? (inn.runs / inn.balls * 6).toFixed(2) : "0.00"
   389→  // Partnership runs = current total − runs on the board when this
   390→  // partnership began. Balls faced = current balls − balls then.
   391→  const pRuns = Math.max(0, inn.runs - inn.partnershipStartRuns)
   392→  const pBalls = Math.max(0, inn.balls - inn.partnershipStartBalls)
   393→  const target = inn.target
   394→  const ballsLeft = state.maxOvers > 0 ? state.maxOvers * 6 - inn.balls : null
   395→  const runsNeeded = target ? target - inn.runs : null
   396→  const rrr =
   397→    target && ballsLeft && ballsLeft > 0
   398→      ? ((runsNeeded! / ballsLeft) * 6).toFixed(2)
   399→      : null
   400→
   401→  return (
   402→    <div className="rounded-lg border bg-gradient-to-br from-slate-900 to-slate-800 p-3 text-white sm:p-4">
   403→      <div className="flex flex-wrap items-end justify-between gap-3">
   404→        <div className="min-w-0">
   405→          <div className="text-[11px] uppercase tracking-wide text-slate-400">
   406→            {inn.battingTeamShortName} · Innings {state.currentInnings + 1}
   407→            {inn.target ? ` · target ${inn.target}` : ""}
   408→          </div>
   409→          <div className="mt-1 text-3xl font-extrabold tabular-nums sm:text-4xl">
   410→            {inn.runs}/{inn.wickets}
   411→            <span className="ml-2 text-base font-normal text-slate-300">
   412→              ({overs} ov)
   413→            </span>
   414→            {inn.freeHit && (
   415→              <span className="ml-3 inline-flex items-center gap-1 rounded-full bg-amber-500 px-2.5 py-0.5 text-[10px] font-extrabold uppercase tracking-wide text-amber-950 animate-pulse">
   416→                Free hit
   417→              </span>
   418→            )}
   419→          </div>
   420→        </div>
   421→        <div className="text-right text-xs text-slate-300">
   422→          <div>CRR <span className="font-bold text-white tabular-nums">{crr}</span></div>
   423→          {rrr && (
   424→            <div>
   425→              RRR <span className="font-bold text-white tabular-nums">{rrr}</span>
   426→            </div>
   427→          )}
   428→        </div>
   429→      </div>
   430→      <div className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 border-t border-white/10 pt-2 text-[11px] text-slate-300">
   431→        <span>
   432→          Partnership:{" "}
   433→          <span className="font-bold text-white tabular-nums">{pRuns} ({pBalls})</span>
   434→        </span>
   435→        {target && runsNeeded !== null && ballsLeft !== null && ballsLeft > 0 ? (
   436→          <span>
   437→            Need <span className="font-bold text-amber-300 tabular-nums">{runsNeeded}</span> from{" "}
   438→            <span className="font-bold text-white tabular-nums">{ballsLeft}</span> balls · RRR{" "}
   439→            <span className="font-bold text-white tabular-nums">{rrr}</span>
   440→          </span>
   441→        ) : target && runsNeeded !== null && runsNeeded <= 0 ? (
   442→          <span className="font-bold text-emerald-300">Target reached</span>
   443→        ) : null}
   444→      </div>
   445→    </div>
   446→  )
   447→}

   490→/** Group commentary by over, preserving the newest-first order inside
   491→ *  each over. Returns an array of { over, entries } blocks sorted so
   492→ *  the most recent over appears first. */
   493→function groupByOver(commentary: CommentaryItem[]): {
   494→  over: string
   495→  entries: CommentaryItem[]
   496→}[] {
   497→  const groups = new Map<string, CommentaryItem[]>()
   498→  for (const entry of commentary) {
   499→    // Use the integer part of the over string ("12.3" → "12", "12" → "12").
   500→    // The "over" entry at end-of-over has over like "12" (no decimal).
   501→    const overNum = entry.over.split(".")[0] ?? "0"
   502→    const key = overNum
   503→    if (!groups.has(key)) groups.set(key, [])
   504→    groups.get(key)!.push(entry)
   505→  }
   506→  // Sort group keys numerically descending (newest over first).
   507→  const sorted = [...groups.entries()].sort((a, b) => {
   508→    const an = parseInt(a[0], 10) || 0
   509→    const bn = parseInt(b[0], 10) || 0
   510→    return bn - an
   511→  })
   512→  return sorted.map(([over, entries]) => ({ over, entries }))
   513→}
   514→
   515→function CommentaryFeed({
   516→  commentary,
   517→}: {
   518→  commentary: CommentaryItem[]
   519→}) {
   520→  const scrollRef = useRef<HTMLDivElement | null>(null)
   521→  // Track the latest entry id so we know when "new" entries arrive.
   522→  const latestId = commentary[0]?.id ?? ""
   523→
   524→  // Auto-scroll to top (newest entry) when new entries arrive. The feed
   525→  // renders newest-first, so "scroll to top" means showing the freshest
   526→  // commentary — exactly what ESPNcricinfo does.
   527→  useEffect(() => {
   528→    const el = scrollRef.current
   529→    if (el) el.scrollTop = 0
   530→  }, [latestId])
   531→
   532→  if (commentary.length === 0) {
   533→    return (
   534→      <div className="rounded-lg border bg-card">
   535→        <div className="flex items-center gap-2 border-b bg-muted/40 px-3 py-2">
   536→          <MessageSquare className="h-3.5 w-3.5 text-muted-foreground" />
   537→          <h3 className="text-[11px] font-extrabold uppercase tracking-wide">
   538→            Commentary
   539→          </h3>
   540→        </div>
   541→        <div className="flex flex-col items-center justify-center px-3 py-10 text-center">
   542→          <MessageSquare className="h-8 w-8 text-muted-foreground/60" />
   543→          <p className="mt-2 text-sm text-muted-foreground">No commentary yet</p>
   544→          <p className="text-[11px] text-muted-foreground/80">
   545→            Ball-by-ball updates will appear here once scoring begins.
   546→          </p>
   547→        </div>
   548→      </div>
   549→    )
   550→  }
   551→
   552→  const grouped = groupByOver(commentary)
   553→
   554→  return (
   555→    <div className="rounded-lg border bg-card overflow-hidden">
   556→      <div className="flex items-center justify-between gap-2 border-b bg-muted/40 px-3 py-2">
   557→        <div className="flex items-center gap-2">
   558→          <Radio className="h-3.5 w-3.5 text-red-600" />
   559→          <h3 className="text-[11px] font-extrabold uppercase tracking-wide">
   560→            Live Commentary
   561→          </h3>
   562→        </div>
   563→        <span className="text-[10px] text-muted-foreground">
   564→          {commentary.length} entries · newest first
   565→        </span>
   566→      </div>
   567→      <div
   568→        ref={scrollRef}
   569→        className="max-h-[500px] overflow-y-auto p-3 [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-border"
   570→      >
   571→        {grouped.map((group) => (
   572→          <div key={group.over} className="mb-3 last:mb-0">
   573→            <div className="mb-1.5 flex items-center gap-2">
   574→              <span className="text-[10px] font-extrabold uppercase tracking-wide text-muted-foreground">
   575→                Over {group.over}
   576→              </span>
   577→              <span className="h-px flex-1 bg-border/60" />
   578→            </div>
   579→            <div className="space-y-1.5">
   580→              {group.entries.map((entry) => {
   581→                const colors = commentaryColors(entry.kind)
   582→                const isManual = entry.kind === "manual"
   583→                return (
   584→                  <div
   585→                    key={entry.id}
   586→                    className={cn(
   587→                      "flex gap-2 text-sm",
   588→                      colors.border
   589→                    )}
   590→                  >
   591→                    <span
   592→                      className={cn(
   593→                        "inline-flex h-6 min-w-[2.75rem] shrink-0 items-center justify-center rounded-md px-1.5 text-[11px] font-bold tabular-nums",
   594→                        colors.badge
   595→                      )}
   596→                    >
   597→                      {entry.over}
   598→                    </span>
   599→                    <span className={cn("min-w-0 flex-1 leading-snug", colors.text)}>
   600→                      {isManual && (
   601→                        <span className="mr-1 inline-block rounded bg-blue-600 px-1 py-0.5 text-[9px] font-bold uppercase tracking-wide text-white align-middle">
   602→                          Commentator
   603→                        </span>
   604→                      )}
   605→                      {entry.text.startsWith("[audio:") ? (
   606→                        <span className="flex flex-wrap items-center gap-2">
   607→                          <span className="inline-flex items-center gap-1 text-xs font-bold text-blue-600 dark:text-blue-400">
   608→                            <Mic className="h-3 w-3" />
   609→                            Voice commentary
   610→                          </span>
   611→                          <audio
   612→                            src={entry.text.match(/\[audio:(.+?)\]/)?.[1]}
   613→                            controls
   614→                            className="h-8"
   615→                          />
   616→                        </span>
   617→                      ) : (
   618→                        entry.text
   619→                      )}
   620→                    </span>
   621→                  </div>
   622→                )
   623→              })}
   624→            </div>
   625→          </div>
   626→        ))}
   627→      </div>
   628→    </div>
   629→  )
   630→}
