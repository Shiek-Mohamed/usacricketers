    45→  battingTeamShortName: string
    46→  bowlingTeamId: string
    47→  bowlingTeamShortName: string
    48→  runs: number

   169→export function maxOversForFormat(format: string): number {
   170→  if (format === "ODI") return 50
   171→  if (format === "T20I" || format === "T20") return 20
   172→  if (format === "T10") return 10
   173→  if (format === "T25") return 25
   174→  if (format === "T30") return 30
   175→  if (format === "T35") return 35
   176→  // Custom formats: parse "T8" → 8, "T15" → 15, "T40" → 40, etc.
   177→  const customMatch = format.match(/^T(\d+)$/i)
   178→  if (customMatch) return parseInt(customMatch[1], 10)
   179→  return 0 // Test = unlimited
   180→}

   182→export function oversString(balls: number): string {
   183→  const o = Math.floor(balls / 6)
   184→  const b = balls % 6
   185→  return b === 0 ? `${o}` : `${o}.${b}`
   186→}

   188→export function initState(match: Match, battingFirstTeamId: string): LiveMatchState {
   189→  const battingTeam =
   190→    match.team1Id === battingFirstTeamId ? match.team1 : match.team2
   191→  const bowlingTeam =
   192→    match.team1Id === battingFirstTeamId ? match.team2 : match.team1
   193→  const innings: ScorerInnings = {
   194→    battingTeamId: battingFirstTeamId,
   195→    battingTeamShortName: battingTeam?.shortName ?? "BAT",
   196→    bowlingTeamId: battingTeam === match.team1 ? match.team2Id : match.team1Id,
   197→    bowlingTeamShortName: bowlingTeam?.shortName ?? "BWL",
   198→    runs: 0,
   199→    wickets: 0,
   200→    balls: 0,
   201→    batsmen: [],
   202→    bowlers: [],
   203→    fallOfWickets: [],
   204→    declared: false,
   205→    target: null,
   206→    extrasByes: 0,

   208→    extrasWides: 0,

   243→export function addBatsman(
   244→  state: LiveMatchState,
   245→  id: string,
   246→  name: string,
   247→  opts?: { isCaptain?: boolean; isKeeper?: boolean }
   248→): LiveMatchState {
   249→  const inn = state.innings[state.currentInnings]
   250→  const batsman: ScorerBatsman = {
   251→    id,
   252→    name,
   253→    runs: 0,
   254→    balls: 0,
   255→    fours: 0,
   256→    sixes: 0,
   257→    out: false,
   258→    dismissal: "",
   259→    isCaptain: opts?.isCaptain,
   260→    isKeeper: opts?.isKeeper,
   261→  }
   262→  const newInnings = [...state.innings]
   263→  // ensure batsman list (dedup by id) and squad list (dedup by id)
   264→  const batsmen = inn.batsmen.some((b) => b.id === id)
   265→    ? inn.batsmen
   266→    : [...inn.batsmen, batsman]
   267→  const squad = inn.squad.some((s) => s.id === id)
   268→    ? inn.squad
   269→    : [
   270→        ...inn.squad,
   271→        { id, name, isCaptain: opts?.isCaptain, isKeeper: opts?.isKeeper },
   272→      ]
   273→  newInnings[state.currentInnings] = { ...inn, batsmen, squad }
   274→  return { ...state, innings: newInnings }
   275→}

   277→/** Pre-populate the squad with the full batting-team roster (so 'Did not bat'
   278→ * works). Players already added as batsmen are skipped to preserve any
   279→ * captain/keeper tags set on the batsman entry. */
   280→export function addSquadMember(
   281→  state: LiveMatchState,
   282→  id: string,
   283→  name: string,
   284→  opts?: { isCaptain?: boolean; isKeeper?: boolean }
   285→): LiveMatchState {
   286→  const inn = state.innings[state.currentInnings]
   287→  if (inn.squad.some((s) => s.id === id)) return state
   288→  const newInnings = [...state.innings]
   289→  newInnings[state.currentInnings] = {
   290→    ...inn,
   291→    squad: [
   292→      ...inn.squad,
   293→      { id, name, isCaptain: opts?.isCaptain, isKeeper: opts?.isKeeper },
   294→    ],
   295→  }
   296→  return { ...state, innings: newInnings }
   297→}

   364→export function applyBall(state: LiveMatchState, event: BallEvent): LiveMatchState {
   365→  if (state.status !== "live") return state
   366→  if (!state.strikerId || !state.nonStrikerId || !state.bowlerId) return state
   367→
   368→  const inn = state.innings[state.currentInnings]
   369→  const strikerIdx = inn.batsmen.findIndex((b) => b.id === state.strikerId)
   370→  const bowlerIdx = inn.bowlers.findIndex((b) => b.id === state.bowlerId)
   371→  if (strikerIdx < 0 || bowlerIdx < 0) return state
   372→
   373→  let batsmen = inn.batsmen.map((b) => ({ ...b }))
   374→  let bowlers = inn.bowlers.map((b) => ({ ...b }))
   375→  let runs = inn.runs
   376→  let wickets = inn.wickets
   377→  let balls = inn.balls
   378→  let extrasByes = inn.extrasByes
   379→  let extrasLegByes = inn.extrasLegByes
   380→  let extrasWides = inn.extrasWides
   381→  let extrasNoBalls = inn.extrasNoBalls
   382→  // Partnership tracking — start runs/balls for the current partnership.
   383→  let partnerships = inn.partnerships ?? []
   384→  let partnershipStartRuns = inn.partnershipStartRuns ?? 0
   385→  let partnershipStartBalls = inn.partnershipStartBalls ?? 0
   386→  // Free-hit tracking. `isFreeHit` = the CURRENT ball is a free hit (only
   387→  // applies in limited-overs matches — maxOvers > 0). `freeHit` is the value
   388→  // that will be carried into the NEXT ball's state.
   389→  const limitedOvers = state.maxOvers > 0
   390→  const isFreeHit = limitedOvers && inn.freeHit
   391→  let freeHit = inn.freeHit
   392→  let fallOfWickets = inn.fallOfWickets
   393→  let strikerId = state.strikerId
   394→  let nonStrikerId = state.nonStrikerId
   395→  let bowlerId = state.bowlerId
   396→  let note = state.note
   397→  let commentary = state.commentary
   398→
   399→  const striker = batsmen[strikerIdx]
   400→  const bowler = bowlers[bowlerIdx]
   401→  let isLegal = true
   402→  let eventText = ""
   403→  let eventKind: CommentaryEntry["kind"] = "run"
   404→
   405→  switch (event.kind) {
   406→    case "run": {
   407→      const r = event.runs
   408→      runs += r
   409→      striker.runs += r
   410→      striker.balls += 1
   411→      bowler.runsConceded += r
   412→      bowler.overRuns += r
   413→      if (r === 0) {
   414→        bowler.dots += 1
   415→        // Don't leave the commentary text blank for a dot ball — show a
   416→        // visible "0 · dot ball" entry so the commentary feed reads
   417→        // ball-by-ball without gaps.
   418→        eventText = "0 · dot ball"
   419→        eventKind = "dot"
   420→      } else if (r === 4) {
   421→        striker.fours += 1
   422→        bowler.foursConceded += 1
   423→        eventText = `FOUR! ${striker.name} ${r}`
   424→        eventKind = "boundary"
   425→      } else if (r === 6) {
   426→        striker.sixes += 1
   427→        bowler.sixesConceded += 1
   428→        eventText = `SIX! ${striker.name} ${r}`
   429→        eventKind = "six"
   430→      } else {
   431→        eventText = `${r} run${r !== 1 ? "s" : ""} — ${striker.name}`
   432→        eventKind = "run"
   433→      }
   434→      if (r % 2 === 1) {
   435→        const tmp = strikerId
   436→        strikerId = nonStrikerId
   437→        nonStrikerId = tmp
   438→      }
   439→      break
   440→    }
   441→    case "wide": {
   442→      runs += 1
   443→      bowler.runsConceded += 1
   444→      bowler.overRuns += 1
   445→      bowler.wides += 1
   446→      extrasWides += 1
   447→      isLegal = false
   448→      eventText = "Wide (+1)"
   449→      eventKind = "extra"
   450→      break
   451→    }
   452→    case "noball": {
   453→      runs += 1
   454→      bowler.runsConceded += 1
   455→      bowler.overRuns += 1
   456→      bowler.noballs += 1
   457→      extrasNoBalls += 1
   458→      isLegal = false
   459→      // Free-hit policy (limited-overs only): a no-ball makes the next
   460→      // LEGAL delivery a free hit. Wides + no-balls that follow keep it
   461→      // pending until a legal ball is bowled. See the post-switch block
   462→      // for the carry-over logic.
   463→      if (limitedOvers) {
   464→        freeHit = true
   465→        eventText = "No ball (+1) — FREE HIT next ball"
   466→      } else {
   467→        eventText = "No ball (+1)"
   468→      }
   469→      eventKind = "extra"
   470→      break
   471→    }
   472→    case "bye":
   473→    case "legbye": {
   474→      const r = event.runs
   475→      runs += r
   476→      striker.balls += 1
   477→      if (r === 0) bowler.dots += 1
   478→      // byes/legbyes: not credited to batsman runs or bowler runs
   479→      if (event.kind === "bye") {
   480→        extrasByes += r
   481→      } else {
   482→        extrasLegByes += r
   483→      }
   484→      eventText = `${event.kind === "bye" ? "Bye" : "Leg bye"} ${r}`
   485→      eventKind = "extra"
   486→      if (r % 2 === 1) {
   487→        const tmp = strikerId
   488→        strikerId = nonStrikerId
   489→        nonStrikerId = tmp
   490→      }
   491→      break
   492→    }
   493→    case "wicket": {
   494→      // Resolve which batsman got out — must be the striker or the
   495→      // non-striker (run-out can happen at the non-striker's end).
   496→      const isStrikerOut = event.outBatsmanId === strikerId
   497→      const isNonStrikerOut =
   498→        event.outBatsmanId === nonStrikerId && event.outBatsmanId !== strikerId
   499→      if (!isStrikerOut && !isNonStrikerOut) {
   500→        // Unknown batsman id — bail out without changing state.
   501→        return state
   502→      }
   503→      const outBatsman = batsmen.find((b) => b.id === event.outBatsmanId)
   504→      if (!outBatsman) return state
   505→
   506→      // Free-hit policy (limited-overs only): bowler-credited dismissals
   507→      // (bowled / caught / lbw / stumped / hit-wkt) are NOT allowed on a
   508→      // free hit — only run-out, retired-hurt, and retired-out stand.
   509→      // The striker still faces the ball, the ball still counts, but the
   510→      // dismissal is void and the batsman survives. Log a commentary line
   511→      // + an "info"-styled entry so the admin sees the rejection.
   512→      if (
   513→        isFreeHit &&
   514→        BOWLER_CREDITED_WICKETS.has(event.wicketType)
   515→      ) {
   516→        // The striker faced this delivery — credit the ball.
   517→        if (isStrikerOut) {
   518→          outBatsman.balls += 1
   519→        }
   520→        // Consume the free hit — next ball is normal.
   521→        freeHit = false
   522→        eventText = `FREE HIT! ${outBatsman.name} survives (${event.wicketType} not allowed on free hit)`
   523→        eventKind = "info"
   524→        break
   525→      }
   526→
   527→      // Retired-hurt: NOT a dismissal. The batsman left the field injured
   528→      // and may return. Doesn't bowl a delivery, doesn't count as a wicket,
   529→      // doesn't credit the bowler, doesn't add to fall-of-wickets. The
   530→      // batsman's `out` flag stays false. Only the slot is cleared so a
   531→      // replacement batsman can come in (the original can return later
   532→      // via the standard bringInBatsman flow).
   533→      if (event.wicketType === "retiredHurt") {
   534→        isLegal = false  // no delivery was bowled
   535→        outBatsman.dismissal = event.dismissal  // "retired hurt" — stored but `out` stays false
   536→        // Clear the slot so the picker re-shows for the replacement batsman.
   537→        if (isStrikerOut) {
   538→          strikerId = null
   539→        } else {
   540→          nonStrikerId = null
   541→        }
   542→        eventText = `RETIRED HURT — ${outBatsman.name} left the field (may return)`
   543→        eventKind = "info"
   544→        break
   545→      }
   546→
   547→      // Retired-out: a dismissal that doesn't bowl a delivery, so the
   548→      // ball count for the over and the bowler's figures stay untouched.
   549→      // Unlike retired-hurt, this counts as a wicket — the batsman is
   550→      // permanently out and cannot return.
   551→      if (event.wicketType === "retiredOut") {
   552→        isLegal = false
   553→      }
   554→
   555→      outBatsman.out = true
   556→      outBatsman.dismissal = event.dismissal
   557→      // Only the striker has faced this delivery — the non-striker didn't
   558→      // (and on a retired-out, no delivery was bowled at all).
   559→      if (isStrikerOut && event.wicketType !== "retiredOut") {
   560→        outBatsman.balls += 1
   561→      }
   562→
   563→      wickets += 1
   564→
   565→      // Credit the bowler only for these dismissal types (not retired-out
   566→      // or retired-hurt — but retired-hurt already broke above).
   567→      if (BOWLER_CREDITED_WICKETS.has(event.wicketType)) {
   568→        bowler.wickets += 1
   569→        bowler.overWickets += 1
   570→      }
   571→
   572→      fallOfWickets = [
   573→        ...fallOfWickets,
   574→        {
   575→          wktNo: wickets,
   576→          score: runs,
   577→          over: oversString(balls),
   578→          batsman: outBatsman.name,
   579→        },
   580→      ]
   581→
   582→      // Record the completed partnership (runs + balls since the last
   583→      // wicket / innings start). The other batsman is the one NOT out.
   584→      const otherBatsman = batsmen.find(
   585→        (b) => b.id === (isStrikerOut ? nonStrikerId : strikerId)
   586→      )
   587→      partnerships = [
   588→        ...partnerships,
   589→        {
   590→          wicket: wickets,
   591→          runs: runs - partnershipStartRuns,
   592→          balls: balls - partnershipStartBalls,
   593→          batsman1: outBatsman.name,
   594→          batsman2: otherBatsman?.name ?? "?",
   595→        },
   596→      ]
   597→      // Reset the start for the next partnership.
   598→      partnershipStartRuns = runs
   599→      partnershipStartBalls = balls
   600→
   601→      // Set the right slot to null so the UI prompts for the replacement at
   602→      // the correct end.
   603→      if (isStrikerOut) {
   604→        strikerId = null
   605→      } else {
   606→        nonStrikerId = null
   607→      }
   608→      eventText = `WICKET! ${outBatsman.name} ${event.dismissal}`
   609→      eventKind = "wicket"
   610→      break
   611→    }
   612→  }
   613→
   614→  if (isLegal) {
   615→    balls += 1
   616→    bowler.balls += 1
   617→    bowler.overBalls += 1
   618→    state = { ...state, ballNo: state.ballNo + 1 }
   619→  }
   620→
   621→  // Free-hit consumption: a legal delivery (run/bye/legbye, or an allowed
   622→  // non-retired wicket) consumes the free hit — the NEXT ball is normal
   623→  // play. No-balls already set `freeHit = true` above (the next legal ball
   624→  // will be a free hit too). Wides carry the prior status forward (the wide
   625→  // case doesn't touch `freeHit`). Retired-out (no delivery) doesn't consume
   626→  // the free hit either. The wicket-rejection branch already set
   627→  // `freeHit = false` itself; the wicket-allowed branch needs the explicit
   628→  // consume below.
   629→  if (isLegal) {
   630→    if (
   631→      event.kind === "run" ||
   632→      event.kind === "bye" ||
   633→      event.kind === "legbye" ||
   634→      (event.kind === "wicket" &&
   635→        event.wicketType !== "retiredOut" &&
   636→        event.wicketType !== "retiredHurt")
   637→    ) {
   638→      freeHit = false
   639→    }
   640→  }
   641→
   642→  // Over complete?
   643→  const ballsThisOver = balls % 6
   644→  let overComplete = false
   645→  if (isLegal && ballsThisOver === 0 && balls > 0) {
   646→    overComplete = true
   647→    // maiden detection for this bowler's just-finished over
   648→    if (bowler.overRuns === 0 && bowler.overWickets === 0) {
   649→      bowler.maidens += 1
   650→    }
   651→    bowler.overRuns = 0
   652→    bowler.overWickets = 0
   653→    bowler.overBalls = 0
   654→    // end of over: swap ends
   655→    const tmp = strikerId
   656→    strikerId = nonStrikerId
   657→    nonStrikerId = tmp
   658→    bowlerId = null // awaiting new bowler
   659→  }
   660→
   661→  const newInnings: ScorerInnings = {
   662→    ...inn,
   663→    runs,
   664→    wickets,
   665→    balls,
   666→    batsmen,
   667→    bowlers,
   668→    fallOfWickets,
   669→    extrasByes,
   670→    extrasLegByes,
   671→    extrasWides,
   672→    extrasNoBalls,
   673→    freeHit,
   674→    partnerships,
   675→    partnershipStartRuns,
   676→    partnershipStartBalls,
   677→    followOn: inn.followOn ?? false,
   678→    isSuperOver: inn.isSuperOver ?? false,
   679→  }
   680→  const innings = [...state.innings]
   681→  innings[state.currentInnings] = newInnings
   682→
   683→  let status = state.status
   684→  let result = state.result
   685→  let target = inn.target
   686→  let next: LiveMatchState = {
   687→    ...state,
   688→    innings,
   689→    strikerId,
   690→    nonStrikerId,
   691→    bowlerId,
   692→    note,
   693→  }
   694→
   695→  // Commentary
   696→  const overLabel = oversString(balls)
   697→  commentary = pushCommentary(next, eventText, eventKind, overLabel)
   698→  if (overComplete) {
   699→    commentary = [
   700→      {
   701→        id: uid(),
   702→        ballNo: next.ballNo,
   703→        over: oversString(balls),
   704→        text: `End of over ${Math.floor(balls / 6)} — ${inn.battingTeamShortName} ${runs}/${wickets} (${oversString(balls)})`,
   705→        kind: "over",
   706→      },
   707→      ...commentary,
   708→    ].slice(0, 120)
   709→  }
   710→  next.commentary = commentary
   711→
   712→  // Innings end checks
   713→  const allOut = wickets >= 10
   714→  const oversDone = next.maxOvers > 0 && balls >= next.maxOvers * 6
   715→  const targetReached = target !== null && runs >= target
   716→
   717→  if (allOut || oversDone || targetReached) {
   718→    next = endInnings(next, allOut ? "all out" : oversDone ? "overs complete" : "target reached")
   719→  }
   720→
   721→  return next
   722→}
   723→
   724→export function endInnings(state: LiveMatchState, reason: string): LiveMatchState {
   725→  const inn = state.innings[state.currentInnings]
   726→  const inningsCount = state.innings.length
   727→  const innings = [...state.innings]
   728→  // Mark the current innings as ended (and `declared` if the captain declared).
   729→  innings[state.currentInnings] = {
   730→    ...inn,
   731→    declared: reason === "declared" ? true : inn.declared,
   732→  }
   733→
   734→  let commentary = state.commentary
   735→  commentary = [
   736→    {
   737→      id: uid(),
   738→      ballNo: state.ballNo,
   739→      over: oversString(inn.balls),
   740→      text: `Innings end — ${inn.battingTeamShortName} ${inn.runs}/${inn.wickets} (${reason})`,
   741→      kind: "info",
   742→    },
   743→    ...commentary,
   744→  ].slice(0, 120)
   745→
   746→  // Test matches (maxOvers === 0) play up to 4 innings; limited-overs matches
   747→  // (maxOvers > 0) play 2 innings.
   748→  const isTest = state.maxOvers === 0
   749→  const maxInnings = isTest ? 4 : 2
   750→
   751→  // Super-over special case: if the current innings was a super-over, the
   752→  // normal maxInnings check doesn't apply (we already had 2 innings before
   753→  // the super-over started). Instead, count super-over innings: if this is
   754→  // the 1st, create the 2nd (chase target); if this is the 2nd, compute
   755→  // the super-over result.
   756→  if (inn.isSuperOver) {
   757→    const soInnings = innings.filter((i) => i.isSuperOver)
   758→    if (soInnings.length < 2) {
   759→      // This was the 1st super-over innings → create the 2nd with chase.
   760→      const target = inn.runs + 1
   761→      const newSO: ScorerInnings = {
   762→        battingTeamId: inn.bowlingTeamId,
   763→        battingTeamShortName: inn.bowlingTeamShortName,
   764→        bowlingTeamId: inn.battingTeamId,
   765→        bowlingTeamShortName: inn.battingTeamShortName,
   766→        runs: 0, wickets: 0, balls: 0,
   767→        batsmen: [], bowlers: [], fallOfWickets: [],
   768→        declared: false, target,
   769→        extrasByes: 0, extrasLegByes: 0, extrasWides: 0, extrasNoBalls: 0,
   770→        squad: [], freeHit: false,
   771→        partnerships: [], partnershipStartRuns: 0, partnershipStartBalls: 0,
   772→        followOn: false,
   773→        isSuperOver: true,
   774→      }
   775→      innings.push(newSO)
   776→      return {
   777→        ...state,
   778→        innings,
   779→        currentInnings: inningsCount,
   780→        strikerId: null, nonStrikerId: null, bowlerId: null,
   781→        note: `Super Over: ${inn.battingTeamShortName} scored ${inn.runs}. ${inn.bowlingTeamShortName} needs ${target} to win.`,
   782→        commentary: [
   783→          {
   784→            id: uid(),
   785→            ballNo: state.ballNo,
   786→            over: "0",
   787→            text: `Super Over: ${inn.battingTeamShortName} ${inn.runs}/${inn.wickets}. ${inn.bowlingTeamShortName} needs ${target}.`,
   788→            kind: "info" as const,
   789→          },
   790→          ...commentary,
   791→        ].slice(0, 120),
   792→      }
   793→    }
   794→    // Both super-over innings done → compute the result.
   795→    return computeSuperOverResult(state, innings, commentary)
   796→  }
   797→
   798→  // Follow-on special case: if the current innings was a follow-on (Team B
   799→  // batting again in a Test), and Team B's overall total is still less than
   800→  // Team A's 1st innings total, Team A wins by an innings + runs (no need
   801→  // for Team A's 2nd innings). Otherwise, Team A bats their 2nd innings
   802→  // (innings 3) chasing Team B's overall total + 1.
   803→  if (inn.followOn) {
   804→    const teamATotal = innings[0].runs
   805→    const teamBOverall = innings[1].runs + innings[2].runs
   806→    if (teamBOverall < teamATotal) {
   807→      // Team A wins by an innings + runs. Match ends.
   808→      const margin = teamATotal - teamBOverall
   809→      const result = `${innings[0].battingTeamShortName} won by an innings and ${margin} run${margin !== 1 ? "s" : ""}`
   810→      return {
   811→        ...state,
   812→        innings,
   813→        status: "completed",
   814→        result,
   815→        note: result,
   816→        strikerId: null, nonStrikerId: null, bowlerId: null,
   817→        commentary: [
   818→          {
   819→            id: uid(),
   820→            ballNo: state.ballNo,
   821→            over: oversString(inn.balls),
   822→            text: `Result: ${result}`,
   823→            kind: "info" as const,
   824→          },
   825→          ...commentary,
   826→        ].slice(0, 120),
   827→      }
   828→    }
   829→    // Team B surpassed Team A → Team A bats innings 3, chasing.
   830→    const target = teamBOverall - teamATotal + 1
   831→    const newInnings: ScorerInnings = {
   832→      battingTeamId: innings[0].battingTeamId,
   833→      battingTeamShortName: innings[0].battingTeamShortName,
   834→      bowlingTeamId: innings[1].battingTeamId,
   835→      bowlingTeamShortName: innings[1].battingTeamShortName,
   836→      runs: 0, wickets: 0, balls: 0,
   837→      batsmen: [], bowlers: [], fallOfWickets: [],
   838→      declared: false, target,
   839→      extrasByes: 0, extrasLegByes: 0, extrasWides: 0, extrasNoBalls: 0,
   840→      squad: [], freeHit: false,
   841→      partnerships: [], partnershipStartRuns: 0, partnershipStartBalls: 0,
   842→      followOn: false,
   843→      isSuperOver: false,
   844→    }
   845→    innings.push(newInnings)
   846→    return {
   847→      ...state,
   848→      innings,
   849→      currentInnings: inningsCount,
   850→      strikerId: null, nonStrikerId: null, bowlerId: null,
   851→      note: `Team A needs ${target} runs to win.`,
   852→      commentary,
   853→    }
   854→  }
   855→
   856→  // If this was the final innings of the match, compute the result.
   857→  if (inningsCount >= maxInnings) {
   858→    return computeMatchResult(state, innings, commentary)
   859→  }
   860→
   861→  // Otherwise, set up the next innings. Swap batting/bowling sides — in a
   862→  // 4-innings Test the same team bats in innings 0 + 2 (and the other in
   863→  // 1 + 3), which is exactly what swapping from the previous innings gives:
   864→  //   innings 0: A bats, B bowls   →  innings 1: B bats, A bowls
   865→  //   innings 1: B bats, A bowls   →  innings 2: A bats, B bowls
   866→  //   innings 2: A bats, B bowls  →  innings 3: B bats, A bowls
   867→  const newInnings: ScorerInnings = {
   868→    battingTeamId: inn.bowlingTeamId,
   869→    battingTeamShortName: inn.bowlingTeamShortName,
   870→    bowlingTeamId: inn.battingTeamId,
   871→    bowlingTeamShortName: inn.battingTeamShortName,
   872→    runs: 0,
   873→    wickets: 0,
   874→    balls: 0,
   875→    batsmen: [],
   876→    bowlers: [],
   877→    fallOfWickets: [],
   878→    declared: false,
   879→    target: null,
   880→    extrasByes: 0,
   881→    extrasLegByes: 0,
   882→    extrasWides: 0,
   883→    extrasNoBalls: 0,
   884→    squad: [],
   885→    freeHit: false,
   886→    partnerships: [],
   887→    partnershipStartRuns: 0,
   888→    partnershipStartBalls: 0,
   889→    followOn: false,
   890→    isSuperOver: false,
   891→  }
   892→
   893→  // Chase target: only set on the FINAL innings of the match (the team
   894→  // batting last needs to know how many to win).
   895→  //   • Limited-overs (2 innings): innings 1 target = innings[0].runs + 1
   896→  //   • Test (4 innings): innings 3 target =
   897→  //     (innings[0].runs + innings[2].runs) - innings[1].runs + 1
   898→  //     (Team A's two innings totals, minus Team B's first innings, plus 1 to win)
   899→  let targetNote = ""
   900→  if (!isTest && inningsCount === 1) {
   901→    // Limited-overs 2nd innings chase.
   902→    newInnings.target = innings[0].runs + 1
   903→    targetNote = `${inn.battingTeamShortName} set ${newInnings.target} to win. ${inn.bowlingTeamShortName} to chase.`
   904→  } else if (isTest && inningsCount === 3) {
   905→    // Test 4th innings chase.
   906→    // innings[0] = Team A 1st innings
   907→    // innings[1] = Team B 1st innings
   908→    // innings[2] = Team A 2nd innings (just ended — A's total)
   909→    // innings[3] = Team B 2nd innings (about to start — chasing)
   910→    const teamATotal = innings[0].runs + innings[2].runs
   911→    const teamBFirstInnings = innings[1].runs
   912→    const target = teamATotal - teamBFirstInnings + 1
   913→    newInnings.target = target
   914→    targetNote = `${inn.battingTeamShortName} declared at ${inn.runs}/${inn.wickets}. ${inn.bowlingTeamShortName} needs ${target} runs to win.`
   915→  } else {
   916→    targetNote = `${inn.battingTeamShortName} innings ended (${reason}). ${inn.bowlingTeamShortName} to bat next.`
   917→  }
   918→
   919→  innings.push(newInnings)
   920→  return {
   921→    ...state,
   922→    innings,
   923→    currentInnings: inningsCount,
   924→    strikerId: null,
   925→    nonStrikerId: null,
   926→    bowlerId: null,
   927→    note: targetNote,
   928→    commentary,
   929→  }
   930→}

   932→/**
   933→ * Compute the final match result from the completed innings.
   934→ *
   935→ * Limited-overs (2 innings): the chasing team won / tied / lost based on
   936→ * whether they reached the target.
   937→ *
   938→ * Test (4 innings): compare Team A's total (innings 0 + 2) vs Team B's total
   939→ * (innings 1 + 3). If the chasing team (B in innings 3) reached the target,
   940→ * they won by wickets. If they were bowled out short, Team A won by runs.
   941→ * If time ran out (innings 3 ended without all-out or target reached), draw.
   942→ */
   943→function computeMatchResult(
   944→  state: LiveMatchState,
   945→  innings: ScorerInnings[],
   946→  commentary: CommentaryEntry[]
   947→): LiveMatchState {
   948→  const isTest = state.maxOvers === 0
   949→
   950→  let result = ""
   951→  let note = ""
   952→
   953→  if (!isTest) {
   954→    // Limited-overs: innings[1] is the chase.
   955→    const chase = innings[1]
   956→    const target = chase.target ?? 0
   957→    const chaser = chase.battingTeamShortName
   958→    const setter = chase.bowlingTeamShortName
   959→    if (chase.runs >= target) {
   960→      const wktsLeft = 10 - chase.wickets
   961→      result = `${chaser} won by ${wktsLeft} wicket${wktsLeft !== 1 ? "s" : ""}`
   962→    } else if (chase.runs === target - 1) {
   963→      result = "Match tied"
   964→    } else {
   965→      const margin = target - 1 - chase.runs
   966→      result = `${setter} won by ${margin} run${margin !== 1 ? "s" : ""}`
   967→    }
   968→    note = result
   969→  } else {
   970→    // Test: compare Team A (innings 0 + 2) vs Team B (innings 1 + 3).
   971→    const teamA = innings[0].battingTeamShortName
   972→    const teamB = innings[1].battingTeamShortName
   973→    const teamATotal = innings[0].runs + innings[2].runs
   974→    const teamBTotal = innings[1].runs + innings[3].runs
   975→    const chase = innings[3]
   976→    const target = chase.target ?? Infinity
   977→    if (chase.runs >= target) {
   978→      // Team B chased it down — won by wickets.
   979→      const wktsLeft = 10 - chase.wickets
   980→      result = `${teamB} won by ${wktsLeft} wicket${wktsLeft !== 1 ? "s" : ""}`
   981→    } else if (chase.wickets >= 10) {
   982→      // Team B all out short — Team A won by runs (margin = A's total - B's total).
   983→      const margin = teamATotal - teamBTotal
   984→      result = `${teamA} won by ${Math.abs(margin)} run${Math.abs(margin) !== 1 ? "s" : ""}`
   985→    } else {
   986→      // Innings ended (declaration or time) without a result — Test draw.
   987→      result = "Match drawn"
   988→    }
   989→    note = result
   990→  }
   991→
   992→  return {
   993→    ...state,
   994→    innings,
   995→    status: "completed",
   996→    result,
   997→    note,
   998→    strikerId: null,
   999→    nonStrikerId: null,
  1000→    bowlerId: null,
  1001→    commentary: [
  1002→      {
  1003→        id: uid(),
  1004→        ballNo: state.ballNo,
  1005→        over: oversString(innings[innings.length - 1].balls),
  1006→        text: `Result: ${result}`,
  1007→        kind: "info",
  1008→      },
  1009→      ...commentary,
  1010→    ].slice(0, 120),
  1011→  }
  1012→}
  1013→
  1014→/**
  1015→ * Compute the super-over result. The team with more runs in their super-over
  1016→ * innings wins. If equal, it's a tie (no further super-overs in our MVP —
  1017→ * the admin can manually start another one via the Super Over button if
  1018→ * the match is still tied).
  1019→ */
  1020→function computeSuperOverResult(
  1021→  state: LiveMatchState,
  1022→  innings: ScorerInnings[],
  1023→  commentary: CommentaryEntry[]
  1024→): LiveMatchState {
  1025→  const soInnings = innings.filter((i) => i.isSuperOver)
  1026→  const teamA = soInnings[0]
  1027→  const teamB = soInnings[1]
  1028→  let result = ""
  1029→  if (teamA.runs > teamB.runs) {
  1030→    result = `${teamA.battingTeamShortName} won the Super Over`
  1031→  } else if (teamB.runs > teamA.runs) {
  1032→    result = `${teamB.battingTeamShortName} won the Super Over`
  1033→  } else {
  1034→    result = "Super Over tied"
  1035→  }
  1036→  return {
  1037→    ...state,
  1038→    innings,
  1039→    status: "completed",
  1040→    result,
  1041→    note: result,
  1042→    strikerId: null,
  1043→    nonStrikerId: null,
  1044→    bowlerId: null,
  1045→    commentary: [
  1046→      {
  1047→        id: uid(),
  1048→        ballNo: state.ballNo,
  1049→        over: "0",
  1050→        text: `Super Over Result: ${result}`,
  1051→        kind: "info" as const,
  1052→      },
  1053→      ...commentary,
  1054→    ].slice(0, 120),
  1055→  }
  1056→}
  1057→
  1058→/**
  1059→ * Captain declares the current innings closed. Only valid for Test matches
  1060→ * (limited-overs matches don't have declarations). Sets `declared: true`
  1061→ * on the current innings, then triggers the standard endInnings flow with
  1062→ * reason "declared" so the next innings is set up correctly.
  1063→ *
  1064→ * The declaration can happen at any point — the captain doesn't need to be
  1065→ * all out or out of overs. Common reasons: setting a chase target for the
  1066→ * opposition, time preservation, etc.
  1067→ */
  1068→export function declareInnings(state: LiveMatchState): LiveMatchState {
  1069→  if (state.maxOvers > 0) {
  1070→    // Limited-overs — declarations aren't allowed.
  1071→    return state
  1072→  }
  1073→  // Add a commentary note before calling endInnings so the declaration is
  1074→  // visible in the feed.
  1075→  const inn = state.innings[state.currentInnings]
  1076→  const declaredCommentary = [
  1077→    {
  1078→      id: uid(),
  1079→      ballNo: state.ballNo,
  1080→      over: oversString(inn.balls),
  1081→      text: `${inn.battingTeamShortName} declared at ${inn.runs}/${inn.wickets}`,
  1082→      kind: "info" as const,
  1083→    },
  1084→    ...state.commentary,
  1085→  ].slice(0, 120)
  1086→  return endInnings(
  1087→    { ...state, commentary: declaredCommentary },
  1088→    "declared"
  1089→  )
  1090→}
  1091→
  1092→/**
  1093→ * Enforce the follow-on in a Test match. Only valid when:
  1094→ *   • It's a Test (maxOvers === 0)
  1095→ *   • We're at the end of innings 1 (Team B's first innings just finished)
  1096→ *   • Team A's lead (innings[0].runs - innings[1].runs) is >= 200 runs
  1097→ *
  1098→ * When enforced, Team B bats again immediately (their 2nd innings becomes
  1099→ * innings 2, and if Team B is bowled out short of Team A's total, Team A
  1100→ * wins by an innings + runs — no need for Team A's 2nd innings at all).
  1101→ *
  1102→ * Without the follow-on, the standard endInnings flow would create innings
  1103→ * 2 with Team A batting (their 2nd innings). With the follow-on, we create
  1104→ * innings 2 with Team B batting again (same batting/bowling sides as
  1105→ * innings 1, not swapped).
  1106→ */
  1107→export function enforceFollowOn(state: LiveMatchState): LiveMatchState {
  1108→  if (state.maxOvers > 0) return state  // Tests only
  1109→  if (state.innings.length !== 2) return state  // must be after innings 1 (0-indexed)
  1110→  const lead = state.innings[0].runs - state.innings[1].runs
  1111→  if (lead < 200) return state  // need 200+ run lead
  1112→
  1113→  const inn = state.innings[1]  // Team B's 1st innings (just ended)
  1114→  // Create Team B's 2nd innings — same batting side, same bowling side
  1115→  // (no swap — that's the whole point of the follow-on).
  1116→  const newInnings: ScorerInnings = {
  1117→    battingTeamId: inn.battingTeamId,
  1118→    battingTeamShortName: inn.battingTeamShortName,
  1119→    bowlingTeamId: inn.bowlingTeamId,
  1120→    bowlingTeamShortName: inn.bowlingTeamShortName,
  1121→    runs: 0, wickets: 0, balls: 0,
  1122→    batsmen: [], bowlers: [], fallOfWickets: [],

  1153→/**
  1154→ * Start a super-over after a tied limited-overs match. Creates 2 new
  1155→ * innings (1 over each):
  1156→ *   • innings N:   Team 1 bats (1 over, no target)
  1157→ *   • innings N+1: Team 2 bats (1 over, target = Team 1's super-over score + 1)
  1158→ *
  1159→ * Only valid for limited-overs matches (maxOvers > 0) where the match
  1160→ * result is "Match tied". The super-over has its own maxOvers = 1.
  1161→ */
  1162→export function startSuperOver(state: LiveMatchState): LiveMatchState {
  1163→  if (state.maxOvers <= 0) return state  // limited-overs only
  1164→  if (state.status !== "completed") return state
  1165→  if (state.result !== "Match tied") return state
  1166→
  1167→  const inn1 = state.innings[0]  // Team A's innings
  1168→  // Super-over innings 1: Team A bats again (same batting side)
  1169→  const so1: ScorerInnings = {
  1170→    battingTeamId: inn1.battingTeamId,
  1171→    battingTeamShortName: inn1.battingTeamShortName,
  1172→    bowlingTeamId: inn1.bowlingTeamId,
  1173→    bowlingTeamShortName: inn1.bowlingTeamShortName,
  1174→    runs: 0, wickets: 0, balls: 0,
  1175→    batsmen: [], bowlers: [], fallOfWickets: [],
  1176→    declared: false, target: null,
  1177→    extrasByes: 0, extrasLegByes: 0, extrasWides: 0, extrasNoBalls: 0,
  1178→    squad: [], freeHit: false,
  1179→    partnerships: [], partnershipStartRuns: 0, partnershipStartBalls: 0,
  1180→    followOn: false,
  1181→    isSuperOver: true,
  1182→  }
  1183→  const innings = [...state.innings, so1]
  1184→  const commentary = [
  1185→    {
  1186→      id: uid(),
  1187→      ballNo: state.ballNo,
  1188→      over: "0",
  1189→      text: `SUPER OVER — ${inn1.battingTeamShortName} to bat first`,
  1190→      kind: "info" as const,
  1191→    },
  1192→    ...state.commentary,
  1193→  ].slice(0, 120)
  1194→  return {
  1195→    ...state,
  1196→    innings,
  1197→    currentInnings: innings.length - 1,
  1198→    maxOvers: 1,  // super-over = 1 over per side
  1199→    status: "live",
  1200→    result: "",
  1201→    strikerId: null,
  1202→    nonStrikerId: null,
  1203→    bowlerId: null,
  1204→    note: `Super Over — ${inn1.battingTeamShortName} bats first (1 over)`,
  1205→    commentary,
  1206→  }
  1207→}

  1209→/** Convert the live state to the public Scorecard format (read by ScorecardView). */
  1210→export function stateToScorecard(state: LiveMatchState): Scorecard {
  1211→  const innings: Innings[] = state.innings.map((inn) => {
  1212→    const batting: BattingLine[] = inn.batsmen.map((b) => ({
  1213→      name: b.name,
  1214→      playerId: b.id,
  1215→      runs: b.runs,
  1216→      balls: b.balls,
  1217→      fours: b.fours,
  1218→      sixes: b.sixes,
  1219→      strikeRate: b.balls > 0 ? Number(((b.runs / b.balls) * 100).toFixed(1)) : 0,
  1220→      dismissal: b.out ? b.dismissal : "not out",
  1221→      isBatting: !b.out && (b.id === state.strikerId || b.id === state.nonStrikerId),
  1222→      isCaptain: b.isCaptain,
  1223→      isKeeper: b.isKeeper,
  1224→    }))
  1225→    // order: striker first, non-striker second, then out batsmen in order
  1226→    batting.sort((a, b) => {
  1227→      const aLive = a.isBatting ? 0 : 1
  1228→      const bLive = b.isBatting ? 0 : 1
  1229→      return aLive - bLive
  1230→    })
  1231→    // add Extras line with breakdown
  1232→    const extrasRuns = Math.max(
  1233→      0,
  1234→      inn.runs - inn.batsmen.reduce((s, b) => s + b.runs, 0)
  1235→    )
  1236→    batting.push({
  1237→      name: "Extras",
  1238→      runs: extrasRuns,
  1239→      balls: 0,
  1240→      fours: 0,
  1241→      sixes: 0,
  1242→      strikeRate: 0,
  1243→      dismissal: `(b ${inn.extrasByes || 0}, lb ${inn.extrasLegByes || 0}, w ${inn.extrasWides || 0}, nb ${inn.extrasNoBalls || 0})`,
  1244→      isBatting: false,
  1245→    })
  1246→    const bowling: BowlingLine[] = inn.bowlers.map((b) => ({
  1247→      name: b.name,
  1248→      playerId: b.id,
  1249→      overs: oversString(b.balls),
  1250→      maidens: b.maidens,
  1251→      runs: b.runsConceded,
  1252→      wickets: b.wickets,
  1253→      economy: b.balls > 0 ? Number(((b.runsConceded / b.balls) * 6).toFixed(2)) : 0,
  1254→      wides: b.wides || 0,
  1255→      noballs: b.noballs || 0,
  1256→      dots: b.dots || 0,
  1257→      foursConceded: b.foursConceded || 0,
  1258→      sixesConceded: b.sixesConceded || 0,
  1259→      isBowling: b.id === state.bowlerId,
  1260→    }))
  1261→    return {
  1262→      teamId: inn.battingTeamId,
  1263→      teamShortName: inn.battingTeamShortName,
  1264→      total: `${inn.runs}/${inn.wickets}`,
  1265→      totalRuns: inn.runs,
  1266→      totalWickets: inn.wickets,
  1267→      totalOvers: oversString(inn.balls),
  1268→      batting,
  1269→      bowling,
  1270→      fallOfWickets: inn.fallOfWickets.map(
  1271→        (f) => `${f.score}-${f.wktNo} (${f.batsman}, ${f.over} ov)`
  1272→      ),
  1273→      declared: inn.declared,
  1274→      extrasBreakdown: {
  1275→        byes: inn.extrasByes || 0,
  1276→        legByes: inn.extrasLegByes || 0,
  1277→        wides: inn.extrasWides || 0,
  1278→        noBalls: inn.extrasNoBalls || 0,
  1279→      },
  1280→      didNotBat: (inn.squad || []).
  1281→        filter((s) => !inn.batsmen.some((b) => b.id === s.id))
  1282→        .map((s) => ({ name: s.name, isCaptain: s.isCaptain, isKeeper: s.isKeeper })),
  1283→      partnerships: (inn.partnerships ?? []).map((p) =>
  1284→        `${p.wicket}-${p.runs} (${p.batsman1} & ${p.batsman2}, ${oversString(p.balls)} balls)`
  1285→      ),
  1286→      isFollowOn: inn.followOn ?? false,
  1287→      isSuperOver: inn.isSuperOver ?? false,
  1288→    }
  1289→  })
  1290→  return {
  1291→    innings,
  1292→    note: state.note,
  1293→    commentary: state.commentary.map((c) => ({
  1294→      id: c.id,
  1295→      over: c.over,
  1296→      text: c.text,
  1297→      kind: c.kind,
  1298→    })),
  1299→  }
  1300→}
