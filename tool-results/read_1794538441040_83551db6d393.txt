    84→    if (match.status === "completed" && match.result) return match.result

   229→function BallChip({ kind, text, size = "sm" }: { kind: string; text: string; size?: "sm" | "md" }) {
   230→  // All chips use the SAME dimensions regardless of ball type — consistency
   231→  const dim = size === "md" ? "h-9 w-9 text-sm" : "h-7 w-7 text-xs"
   232→  const base = `grid ${dim} shrink-0 place-items-center rounded-full font-bold`
   233→  if (kind === "wicket") return <span className={`${base} bg-red-600 text-white`}>W</span>
   234→  if (kind === "six") return <span className={`${base} bg-fuchsia-600 text-white`}>6</span>
   235→  if (kind === "boundary") return <span className={`${base} bg-sky-600 text-white`}>4</span>
   236→  if (kind === "extra") {
   237→    const t = text.toLowerCase()
   238→    const label = t.includes("wide") ? "wd" : t.includes("no ball") ? "nb" : "+"
   239→    return <span className={`${base} bg-amber-500 text-amber-950`}>{label}</span>
   240→  }
   241→  if (kind === "run") {
   242→    const m = text.match(/^(\d+)/)
   243→    const r = m ? m[1] : "0"
   244→    return <span className={`${base} border border-gray-300 bg-white text-gray-700`}>{r}</span>
   245→  }
   246→  // Dot ball — same size, gray fill with center dot
   247→  return <span className={`${base} bg-gray-200`}><span className="h-1.5 w-1.5 rounded-full bg-gray-500" /></span>
   248→}

   254→function CommentaryAccordion({ scorecard, match }: { scorecard: Scorecard; match: Match }) {
   255→  const commentary = scorecard.commentary ?? []
   256→  if (commentary.length === 0) {
   257→    return (
   258→      <Card className="p-6 text-center text-sm text-muted-foreground">
   259→        No commentary available for this match.
   260→      </Card>
   261→    )
   262→  }
   263→
   264→  // Build innings selector pills (if multiple innings)
   265→  const inningsLabels = scorecard.innings.map((inn, i) => ({
   266→    key: `inn-${i}`,
   267→    label: `${inn.teamShortName} Inns ${scorecard.innings.filter((x) => x.teamId === inn.teamId).indexOf(inn) + 1}`,
   268→    teamId: inn.teamId,
   269→  }))
   270→
   271→  // Group commentary by over number. Newest first.
   272→  // Skip over 0 — those are pre-match info messages ("innings begins"),
   273→  // not actual overs. Cricket overs start from 1.
   274→  const ballKinds = new Set(["run", "boundary", "six", "wicket", "extra", "dot"])
   275→  const overMap = new Map<number, typeof commentary>()
   276→  for (const c of commentary) {
   277→    const overNum = parseInt(c.over?.split(".")[0] || "0", 10)
   278→    if (overNum < 1) continue // skip info/pre-match entries
   279→    if (!overMap.has(overNum)) overMap.set(overNum, [])
   280→    overMap.get(overNum)!.push(c)
   281→  }
   282→  const overs = Array.from(overMap.keys()).sort((a, b) => b - a)
   283→
   284→  return (
   285→    <div className="space-y-3">
   286→      {/* Innings selector pills */}
   287→      {inningsLabels.length > 1 && (
   288→        <div className="flex gap-1 rounded-full bg-muted p-1">
   289→          {inningsLabels.map((inn, i) => (
   290→            <button
   291→              key={inn.key}
   292→              className={`flex-1 rounded-full px-3 py-1.5 text-sm font-bold transition-colors ${
   293→                i === 0 ? "bg-[#1e2a5e] text-white" : "text-muted-foreground hover:bg-accent"
   294→              }`}
   295→            >
   296→              {inn.label}
   297→            </button>
   298→          ))}
   299→        </div>
   300→      )}
   301→
   302→      {/* Over blocks */}
   303→      <div className="overflow-hidden rounded-lg border border-border/60">
   304→        {overs.map((overNum) => {
   305→          const entries = overMap.get(overNum)!
   306→          const balls = entries.filter((e) => ballKinds.has(e.kind))
   307→          const runsInOver = balls.reduce((sum, b) => {
   308→            const m = b.text.match(/^(\d+)/)
   309→            return sum + (m ? Number(m[1]) : 0)
   310→          }, 0)
   311→          const hasWicket = balls.some((b) => b.kind === "wicket")
   312→          const bowler = entries.find((e) => e.text.includes("—"))?.text.split("—")[1]?.trim()?.split("  ")[0] ?? ""
   313→
   314→          return (
   315→            <OverBlock
   316→              key={overNum}
   317→              overNum={overNum}
   318→              entries={entries}
   319→              balls={balls}
   320→              runsInOver={runsInOver}
   321→              hasWicket={hasWicket}
   322→              bowler={bowler}
   323→              isLast={overNum === overs[overs.length - 1]}
   324→            />
   325→          )
   326→        })}
   327→      </div>
   328→    </div>
   329→  )
   330→}

   332→function OverBlock({
   333→  overNum,
   334→  entries,
   335→  balls,
   336→  runsInOver,
   337→  hasWicket,
   338→  bowler,
   339→  isLast,
   340→}: {
   341→  overNum: number
   342→  entries: { id: string; over: string; text: string; kind: string }[]
   343→  balls: { id: string; over: string; text: string; kind: string }[]
   344→  runsInOver: number
   345→  hasWicket: boolean
   346→  bowler: string
   347→  isLast: boolean
   348→}) {
   349→  const [expanded, setExpanded] = useState(false)
   350→  return (
   351→    <div className={`bg-card ${!isLast ? "border-b border-border/40" : ""}`}>
   352→      {/* Over header — title row */}
   353→      <button
   354→        onClick={() => setExpanded(!expanded)}
   355→        className="flex w-full items-center justify-between gap-2 px-4 pt-3 text-left"
   356→      >
   357→        <span className="font-display text-base font-bold tabular-nums text-foreground">Over {overNum}</span>
   358→        <div className="flex items-center gap-2">
   359→          {/* Summary text */}
   360→          <span className="text-sm text-muted-foreground">
   361→            {runsInOver > 0 && `${runsInOver} run${runsInOver > 1 ? "s" : ""}`}
   362→            {hasWicket && runsInOver > 0 && " · "}
   363→            {hasWicket && <span className="font-bold text-red-600">W</span>}
   364→            {bowler && ` · ${bowler}`}
   365→          </span>
   366→          <ChevronDown className={`h-4 w-4 shrink-0 text-muted-foreground transition-transform ${expanded ? "rotate-180" : ""}`} />
   367→        </div>
   368→      </button>
   369→
   370→      {/* Ball chips row — below the title */}
   371→      <button
   372→        onClick={() => setExpanded(!expanded)}
   373→        className="flex w-full items-center gap-2 px-4 pb-3 pt-2 text-left"
   374→      >
   375→        <div className="flex flex-1 items-center gap-2 overflow-hidden">
   376→          {balls.slice(0, 8).map((b) => (
   377→            <BallChip key={b.id} kind={b.kind} text={b.text} />
   378→          ))}
   379→        </div>
   380→      </button>
   381→
   382→      {/* Expanded — per-ball commentary cards (Wisden style) */}
   383→      {expanded && (
   384→        <div className="space-y-2 px-4 pb-4">
   385→          {entries.map((e) => (
   386→            <BallCommentaryCard key={e.id} entry={e} />
   387→          ))}
   388→        </div>
   389→      )}
   390→    </div>
   391→  )
   392→}

   394→function BallCommentaryCard({ entry }: { entry: { id: string; over: string; text: string; kind: string } }) {
   395→  const ballKinds = new Set(["run", "boundary", "six", "wicket", "extra", "dot"])
   396→  const isBall = ballKinds.has(entry.kind)
   397→  // Parse "Bowler — description" format, or use the text as-is
   398→  const parts = entry.text.split("—")
   399→  const heading = parts[0]?.trim() || ""
   400→  const description = parts.slice(1).join("—").trim() || entry.text
   401→
   402→  return (
   403→    <div className="flex items-start gap-3 rounded-lg bg-muted/50 p-3">
   404→      {/* Ball number circle + over.ball notation */}
   405→      <div className="flex shrink-0 flex-col items-center gap-1">
   406→        {isBall ? (
   407→          <BallChip kind={entry.kind} text={entry.text} size="md" />
   408→        ) : (
   409→          <div className="h-9 w-9 shrink-0" />
   410→        )}
   411→        <span className="text-xs font-medium text-muted-foreground">{entry.over}</span>
   412→      </div>
   413→      {/* Commentary text */}
   414→      <div className="min-w-0 flex-1 pt-1">
   415→        {heading && (
   416→          <p className="text-sm font-bold text-foreground">{heading}</p>
   417→        )}
   418→        <p className="mt-0.5 text-sm text-muted-foreground">{description}</p>
   419→      </div>
   420→    </div>
   421→  )
   422→}

   440→function SquadCard({
   441→  title,
   442→  shortName,
   443→  innings,
   444→}: {
   445→  title: string
   446→  shortName?: string
   447→  innings: Scorecard["innings"]
   448→}) {
   449→  const played = innings.flatMap((inn) => inn.batting.map((b) => b.name))
   450→  const didNotBat = innings.flatMap((inn) => inn.didNotBat ?? [])
   451→  const all = [...played, ...didNotBat.map((d) => d.name)]
   452→  const deduped = [...new Set(all)]
   453→
   454→  return (
   455→    <Card className="overflow-hidden p-0">
   456→      <div className="border-b bg-muted/30 px-4 py-2.5">
   457→        <h3 className="font-display text-sm font-bold uppercase tracking-wide">
   458→          {shortName && <span className="text-emerald-600">{shortName} </span>}
   459→          {title}
   460→        </h3>
   461→      </div>
   462→      <div className="divide-y divide-border/40">
   463→        {deduped.length > 0 ? (
   464→          deduped.map((name, i) => (
   465→            <div key={i} className="flex items-center gap-2 px-4 py-2">
   466→              <span className="grid h-6 w-6 shrink-0 place-items-center rounded-full bg-muted text-xs font-bold text-muted-foreground">{i + 1}</span>
   467→              <span className="text-sm font-medium text-foreground">{name}</span>

   721→/** Simplified win probability model based on match state. */
   722→function computeWinProbability(match: Match, sc: Scorecard | null):
   723→  | { t1: number; t2: number; draw: number }
   724→  | null {
   725→  if (!sc || sc.innings.length === 0) return null
   726→  if (match.status === "completed" && match.result) {
   727→    // Completed match — assign 100% to the winner
   728→    if (match.result.includes(match.team1?.shortName ?? "___") && !match.result.includes("tie")) {
   729→      return { t1: 100, t2: 0, draw: 0 }
   730→    }
   731→    if (match.result.includes(match.team2?.shortName ?? "___") && !match.result.includes("tie")) {
   732→      return { t1: 0, t2: 100, draw: 0 }
   733→    }
   734→    return { t1: 50, t2: 50, draw: 0 } // tie/no result
   735→  }
   736→
   737→  const isTest = match.format === "Test"
   738→  const innings = sc.innings
   739→  const lastInn = innings[innings.length - 1]
   740→  const firstInn = innings[0]
   741→
   742→  if (innings.length === 1) {
   743→    // 1st innings in progress — roughly 50/50 with slight edge to batting team
   744→    const runs = lastInn.totalRuns
   745→    const wickets = lastInn.totalWickets
   746→    const overs = parseFloat(lastInn.totalOvers) || 0
   747→    const maxOvers = match.format === "T20" || match.format === "T20I" ? 20 : match.format === "ODI" || match.format === "ODI" ? 50 : 90
   748→    // Simple heuristic: more runs + fewer wickets = better position
   749→    const progress = overs / maxOvers // 0 to 1
   750→    const wicketPenalty = wickets * 8 // each wicket costs ~8% confidence
   751→    const runBonus = Math.min(runs / (maxOvers * 5), 1) * 20 // strong total gives bonus
   752→    const battingAdvantage = Math.max(0, Math.min(25, runBonus - wicketPenalty * progress))
   753→    const drawChance = isTest ? Math.max(5, 40 - overs * 0.4) : 0
   754→    const remaining = 100 - drawChance
   755→    const battingTeam = lastInn.teamId === match.team1Id ? "t1" : "t2"
   756→    const battingProb = remaining / 2 + battingAdvantage
   757→    const bowlingProb = remaining - battingProb
   758→    if (battingTeam === "t1") return { t1: Math.round(battingProb), t2: Math.round(bowlingProb), draw: Math.round(drawChance) }
   759→    return { t1: Math.round(bowlingProb), t2: Math.round(battingProb), draw: Math.round(drawChance) }
   760→  }
   761→
   762→  if (innings.length >= 2) {
   763→    // 2nd innings — chasing team's probability based on target + wickets + balls left
   764→    const target = (firstInn.totalRuns ?? 0) + 1
   765→    const chasingRuns = lastInn.totalRuns
   766→    const chasingWickets = lastInn.totalWickets
   767→    const chasingOvers = parseFloat(lastInn.totalOvers) || 0
   768→    const maxOvers = match.format === "T20" || match.format === "T20I" ? 20 : match.format === "ODI" || match.format === "ODI" ? 50 : 90
   769→    const ballsRemaining = Math.max(0, (maxOvers - chasingOvers) * 6)
   770→    const runsNeeded = target - chasingRuns
   771→    const wicketsInHand = 10 - chasingWickets
   772→
   773→    if (runsNeeded <= 0) {
   774→      // Chase complete — chasing team wins
   775→      const chaser = lastInn.teamId === match.team1Id ? "t1" : "t2"
   776→      return chaser === "t1" ? { t1: 100, t2: 0, draw: 0 } : { t1: 0, t2: 100, draw: 0 }
   777→    }
   778→    if (ballsRemaining <= 0 || wicketsInHand <= 0) {
   779→      // Can't chase — batting team (first innings) wins
   780→      const firstTeam = firstInn.teamId === match.team1Id ? "t1" : "t2"
   781→      return firstTeam === "t1" ? { t1: 100, t2: 0, draw: isTest ? 0 : 0 } : { t1: 0, t2: 100, draw: 0 }
   782→    }
   783→
   784→    const requiredRR = (runsNeeded / ballsRemaining) * 6
   785→    const currentRR = chasingOvers > 0 ? chasingRuns / chasingOvers : 0
   786→    // Win probability: based on RRR vs CRR + wickets in hand
   787→    const rrRatio = currentRR > 0 ? requiredRR / currentRR : 2
   788→    const chaseProb = Math.max(0, Math.min(100,
   789→      (wicketsInHand / 10) * 60 + (1 / rrRatio) * 40
   790→    ))
   791→    const drawChance = isTest ? Math.max(0, 30 - innings.length * 10) : 0
   792→    const remaining = 100 - drawChance
   793→    const chaser = lastInn.teamId === match.team1Id ? "t1" : "t2"
   794→    const chaserProb = Math.round(chaseProb * remaining / 100)
   795→    const setterProb = remaining - chaserProb
   796→    if (chaser === "t1") return { t1: chaserProb, t2: setterProb, draw: Math.round(drawChance) }
   797→    return { t1: setterProb, t2: chaserProb, draw: Math.round(drawChance) }
   798→  }
   799→
   800→  return null
   801→}

   847→/* -------------------------------------------------------------------------- */
   848→/*  Cumulative Run Worm — both teams' progress on one chart                    */
   849→/* --------------------------------------------------------------------------- */
   850→
   851→function CumulativeRunWorm({ match, scorecard }: { match: Match; scorecard: Scorecard }) {
   852→  const t1 = match.team1
   853→  const t2 = match.team2
   854→  const t1Color = t1?.primaryColor ?? "#1e2a5e"
   855→  const t2Color = t2?.primaryColor ?? "#94a3b8"
   856→
   857→  // Compute cumulative runs per over for each team
   858→  const ballKinds = new Set(["run", "boundary", "six", "wicket", "extra", "dot"])
   859→
   860→  function cumulativeForTeam(teamId: string): Map<number, number> {
   861→    const overRuns = new Map<number, number>()
   862→    for (const c of scorecard.commentary ?? []) {
   863→      const overNum = parseInt(c.over?.split(".")[0] || "0", 10)
   864→      if (overNum < 1) continue
   865→      if (!ballKinds.has(c.kind)) continue
   866→      // Check if this ball belongs to this team's innings
   867→      const inn = scorecard.innings.find((i) => i.teamId === teamId)
   868→      if (!inn) continue
   869→      const m = c.text.match(/^(\d+)/)
   870→      const runs = m ? Number(m[1]) : 0
   871→      overRuns.set(overNum, (overRuns.get(overNum) ?? 0) + runs)
   872→    }
   873→    // Convert to cumulative
   874→    let cum = 0
   875→    const cumulative = new Map<number, number>()
   876→    const sorted = Array.from(overRuns.keys()).sort((a, b) => a - b)
   877→    for (const over of sorted) {
   878→      cum += overRuns.get(over) ?? 0
   879→      cumulative.set(over, cum)
   880→    }
   881→    return cumulative
   882→  }
   883→
   884→  const t1Data = cumulativeForTeam(match.team1Id)
   885→  const t2Data = cumulativeForTeam(match.team2Id)
   886→
   887→  // Merge into a single dataset keyed by over number
   888→  const allOvers = Array.from(new Set([...t1Data.keys(), ...t2Data.keys()])).sort((a, b) => a - b)
   889→  const data = allOvers.map((over) => ({
   890→    over: `${over}`,
   891→    [match.team1Id]: t1Data.get(over) ?? null,
   892→    [match.team2Id]: t2Data.get(over) ?? null,
   893→  }))
   894→
   895→  if (data.length === 0) {
   896→    return null
   897→  }
   898→
   899→  const chartConfig: ChartConfig = {
   900→    [match.team1Id]: { label: t1?.shortName ?? "Team 1", color: t1Color },
   901→    [match.team2Id]: { label: t2?.shortName ?? "Team 2", color: t2Color },
   902→  }
   903→
   904→  return (
   905→    <Card className="overflow-hidden p-0">
   906→      <div className="border-b bg-muted/30 px-4 py-2.5">
   907→        <h3 className="flex items-center gap-1.5 font-display text-sm font-bold uppercase tracking-wide">
   908→          <BarChart3 className="h-4 w-4 text-emerald-600" />
   909→          Cumulative Run Worm
   910→        </h3>
   911→        <p className="mt-0.5 text-xs text-muted-foreground">
   912→          {t1?.shortName} vs {t2?.shortName} — cumulative runs per over
   913→        </p>
   914→      </div>
   915→      <div className="p-4">
   916→        {/* Legend */}
   917→        <div className="mb-3 flex items-center gap-4">
   918→          <div className="flex items-center gap-1.5">
   919→            <span className="h-3 w-6 rounded-full" style={{ background: t1Color }} />
   920→            <span className="text-sm font-bold">{t1?.shortName}</span>
   921→          </div>
   922→          <div className="flex items-center gap-1.5">
   923→            <span className="h-3 w-6 rounded-full" style={{ background: t2Color }} />
   924→            <span className="text-sm font-bold">{t2?.shortName}</span>
   925→          </div>
   926→        </div>

   966→function RunRateWorm({
   967→  innings,
   968→  teamName,
   969→  teamShort,
   970→  teamColor,
   971→  commentary,
   972→  teamId,
   973→}: {
   974→  innings: Scorecard["innings"][0]
   975→  teamName: string
   976→  teamShort: string
   977→  teamColor: string
   978→  commentary: { id: string; over: string; text: string; kind: string }[]
   979→  teamId: string
   980→}) {
   981→  // Compute runs per over from commentary data
   982→  const ballKinds = new Set(["run", "boundary", "six", "wicket", "extra", "dot"])
   983→  const overRuns = new Map<number, number>()
   984→
   985→  for (const c of commentary) {
   986→    const overNum = parseInt(c.over?.split(".")[0] || "0", 10)
   987→    if (overNum < 1) continue
   988→    if (!ballKinds.has(c.kind)) continue
   989→    const m = c.text.match(/^(\d+)/)
   990→    const runs = m ? Number(m[1]) : 0
   991→    overRuns.set(overNum, (overRuns.get(overNum) ?? 0) + runs)
   992→  }
   993→
   994→  const data = Array.from(overRuns.entries())
   995→    .sort((a, b) => a[0] - b[0])
   996→    .map(([over, runs]) => ({ over: `Ov ${over}`, runs }))
   997→
   998→  if (data.length === 0) {
   999→    return (
  1000→      <Card className="p-6 text-center text-sm text-muted-foreground">
  1001→        No over-by-over data available for {teamName} yet.
  1002→      </Card>
  1003→    )
  1004→  }
  1005→
  1006→  const chartConfig: ChartConfig = {
  1007→    runs: { label: "Runs", color: teamColor },
  1008→  }
  1009→
  1010→  // Color bars: wicket overs in red, maiden (0 runs) in gray, else team color
  1011→  const maxRuns = Math.max(...data.map((d) => d.runs), 6)
  1012→
  1013→  return (
  1014→    <Card className="overflow-hidden p-0">
  1015→      <div className="border-b bg-muted/30 px-4 py-2.5">
  1016→        <h3 className="flex items-center gap-1.5 font-display text-sm font-bold uppercase tracking-wide">
  1017→          <BarChart3 className="h-4 w-4 text-emerald-600" />
  1018→          {teamShort} — Run Rate Worm
  1019→        </h3>
  1020→        <p className="mt-0.5 text-xs text-muted-foreground">
  1021→          {teamName} · {innings.totalRuns}/{innings.totalWickets} ({innings.totalOvers} ov)
  1022→        </p>
  1023→      </div>
  1024→      <div className="p-4">
  1025→        <ChartContainer config={chartConfig} className="h-[300px] w-full">
  1026→          <BarChart data={data} margin={{ top: 16, right: 16, bottom: 16, left: 0 }} barCategoryGap="20%" barGap={4}>
  1027→            <XAxis
  1028→              dataKey="over"
  1029→              tickLine={false}
  1030→              axisLine={false}
  1031→              tick={{ fontSize: 10 }}
  1032→              interval="preserveStartEnd"
  1033→            />
  1034→            <YAxis
  1035→              tickLine={false}
  1036→              axisLine={false}
  1037→              tick={{ fontSize: 10 }}
  1038→              domain={[0, maxRuns]}
  1039→              width={32}
  1040→            />
  1041→            <Bar dataKey="runs" radius={[4, 4, 0, 0]} maxBarSize={48}>
  1042→              {data.map((entry, index) => (
  1043→                <Cell
  1044→                  key={`cell-${index}`}
  1045→                  fill={entry.runs === 0 ? "#d1d5db" : teamColor}
  1046→                />
  1047→              ))}
  1048→            </Bar>
  1049→          </BarChart>
  1050→        </ChartContainer>
  1051→      </div>
  1052→    </Card>
  1053→  )
  1054→}
