   189→      t1.runs,

   192→      t2.runs,
   193→      Math.floor(t2.balls / 6),
   194→      t2.wickets,

   233→  /* ------------------ Voice clip (🎙 mic) recording ------------------
   234→   * The scorer clicks "Record voice clip", the browser captures audio from

   239→   * and render an inline <audio> player instead of the raw text.
   240→   *
   241→   * Refs hold the MediaRecorder + stream + chunk list (no re-render needed

   243→   * reflects what's happening. */
   244→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   245→  const streamRef = useRef<MediaStream | null>(null)

   250→  /** Begin a voice clip recording. Requests microphone permission — the
   251→   *  browser shows a permission prompt the first time, then remembers the
   252→   *  decision for subsequent recordings on this origin. */
   253→  async function startRecording() {
   254→    if (!liveState || !id) return
   255→    if (typeof navigator === "undefined" || !navigator.mediaDevices?.getUserMedia) {
   256→      toast.error("Voice recording is not supported in this browser")
   257→      return
   258→    }
   259→    try {
   260→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   261→      streamRef.current = stream
   262→      chunksRef.current = []
   263→      // Pick the most likely mime type the browser supports — Chrome/Firefox
   264→      // emit audio/webm, Safari emits audio/mp4. We pass this through to the
   265→      // upload so the API saves the file with the correct extension.
   266→      const mimeType = MediaRecorder.isTypeSupported("audio/webm")
   267→        ? "audio/webm"
   268→        : MediaRecorder.isTypeSupported("audio/mp4")
   269→          ? "audio/mp4"
   270→          : ""
   271→      const recorder = mimeType
   272→        ? new MediaRecorder(stream, { mimeType })
   273→        : new MediaRecorder(stream)
   274→      recorder.ondataavailable = (e) => {
   275→        if (e.data && e.data.size > 0) chunksRef.current.push(e.data)
   276→      }
   277→      recorder.onstop = async () => {
   278→        // Tear down the mic stream first so the browser's red dot disappears.
   279→        if (streamRef.current) {
   280→          streamRef.current.getTracks().forEach((t) => t.stop())
   281→          streamRef.current = null
   282→        }
   283→        const chunks = chunksRef.current
   284→        chunksRef.current = []
   285→        if (chunks.length === 0) return
   286→        const blob = new Blob(chunks, {
   287→          type: recorder.mimeType || "audio/webm",
   288→        })
   289→        // Upload the blob.
   290→        setIsUploading(true)
   291→        try {
   292→          const formData = new FormData()
   293→          const ext = (recorder.mimeType || "").includes("mp4")
   294→            ? "mp4"
   295→            : (recorder.mimeType || "").includes("ogg")
   296→              ? "ogg"
   297→              : "webm"
   298→          formData.append("audio", blob, `voice-${Date.now()}.${ext}`)
   299→          const headers: Record<string, string> = {}
   300→          // Always send the bearer scorer token if we have one (real scorer
   301→          // session OR the god-mode token minted for site admins). The API
   302→          // also accepts site-admin cookie auth as a fallback.
   303→          if (token) headers["Authorization"] = `Bearer ${token}`
   304→          // CSRF double-submit for the cookie-auth path (site admin).
   305→          if (typeof document !== "undefined") {
   306→            const csrf = document.cookie.match(/csrf-token=([^;]+)/)?.[1] || ""
   307→            if (csrf) headers["X-CSRF-Token"] = csrf
   308→          }
   309→          const res = await fetch(
   310→            `/api/voice-clip?matchId=${encodeURIComponent(id)}`,
   311→            {
   312→              method: "POST",
   313→              headers,
   314→              credentials: "include",
   315→              body: formData,
   316→            }
   317→          )
   318→          if (!res.ok) {
   319→            const err = await res.json().catch(() => ({ error: "upload failed" }))
   320→            toast.error(`Voice clip upload failed: ${err.error || res.statusText}`)
   321→            return
   322→          }
   323→          const body = (await res.json()) as { url?: string; id?: string }
   324→          if (body?.url) {
   325→            // Insert as a commentary entry — the feed renders an audio
   326→            // player when the text starts with `[audio:`.
   327→            addCommentary(`[audio:${body.url}]`)
   328→            toast.success("Voice clip added to commentary")
   329→          }
   330→        } catch (e) {
   331→          toast.error(`Voice clip upload failed: ${String(e).slice(0, 100)}`)
   332→        } finally {
   333→          setIsUploading(false)
   334→        }
   335→      }
   336→      recorder.start()
   337→      mediaRecorderRef.current = recorder
   338→      setIsRecording(true)
   339→    } catch (e) {
   340→      // Most common: user denied the mic permission, or no mic attached.
   341→      toast.error(
   342→        `Microphone unavailable: ${String(e).slice(0, 120)}. Allow microphone access in your browser and try again.`
   343→      )
   344→      setIsRecording(false)
   345→    }
   346→  }

   349→  function stopRecording() {

   362→    setIsRecording(false)

   365→  /* ------------------ Keyboard shortcuts (power-user scorers) ------------------
   366→   * 0,1,2,3,4,6 → score N runs | . / - → dot | w → wicket dialog |
   367→   * e → swap ends | u → undo | d → declare (Test) | s → save.
   368→   * No deps array → re-subscribes on every render with the latest closures.
   369→   * The function declarations (handleBall, swapEnds, etc.) are hoisted by
   370→   * JS so they're in scope at call-time even though defined later in the
   371→   * component body. The lint rule below is overly cautious about this
   372→   * pattern — it's safe because the effect re-runs every render. */
   373→  /* eslint-disable react-hooks/immutability */
   374→  useEffect(() => {
   375→    function isTyping(el: Element | null): boolean {
   376→      if (!el) return false
   377→      const tag = el.tagName?.toLowerCase()
   378→      return tag === "input" || tag === "textarea" || tag === "select" ||
   379→        el.getAttribute("role") === "combobox" ||
   380→        (el as HTMLElement)?.isContentEditable === true
   381→    }
   382→    function onKey(e: KeyboardEvent) {
   383→      if (e.metaKey || e.ctrlKey || e.altKey) return
   384→      if (isTyping(document.activeElement)) return
   385→      if (wicketOpen) return
   386→      if (!liveState || liveState.status !== "live") return
   387→      if (!liveState.strikerId || !liveState.nonStrikerId || !liveState.bowlerId) return
   388→      const k = e.key.toLowerCase()
   389→      if (["0", "1", "2", "3", "4", "6"].includes(k)) {
   390→        e.preventDefault()
   391→        handleBall({ kind: "run", runs: Number(k) })
   392→      } else if (k === "." || k === "-") {
   393→        e.preventDefault()
   394→        handleBall({ kind: "run", runs: 0 })
   395→      } else if (k === "w") {
   396→        e.preventDefault()
   397→        setWicketOpen(true)
   398→      } else if (k === "e") {
   399→        e.preventDefault()
   400→        swapEnds()
   401→      } else if (k === "u") {
   402→        e.preventDefault()
   403→        handleUndo()
   404→      } else if (k === "d") {
   405→        e.preventDefault()
   406→        handleDeclare()
   407→      } else if (k === "s") {
   408→        e.preventDefault()
   409→        if (liveState) {
   410→          persist(liveState)
   411→          toast.success("Scorecard saved")
   412→        }
   413→      }
   414→    }
   415→    window.addEventListener("keydown", onKey)
   416→    return () => window.removeEventListener("keydown", onKey)
   417→  })

   483→  function persist(state: LiveMatchState) {
   484→    if (saveTimer.current) clearTimeout(saveTimer.current)
   485→    saveTimer.current = setTimeout(async () => {
   486→      try {
   487→        await api.matches.update(match!.id, {
   488→          scorecard: stateToScorecard(state),
   489→          status: state.status,
   490→          result: state.result,
   491→          // Save the FULL LiveMatchState to the DB so scoring can be resumed
   492→          // even after the score-service restarts AND localStorage is cleared
   493→          // (e.g. different browser, overnight, weekend). The liveStateJson
   494→          // field stores the complete state including striker/bowler IDs.
   495→          liveStateJson: JSON.stringify(state),
   496→        })
   497→        refetch()
   498→      } catch (e) {
   499→        // Surface persistence errors so the operator knows the DB write failed
   500→        // (was previously silent — caused confusion when "Innings started"
   501→        // toast showed but the match status didn't actually update).
   502→        // The live socket broadcast is still the priority — this toast
   503→        // doesn't block scoring, just informs the operator something's wrong.
   504→        const msg = e instanceof Error ? e.message : String(e).slice(0, 200)
   505→        toast.error("Persist failed — match state not saved", {
   506→          description: msg,
   507→        })
   508→      }
   509→    }, 800)
   510→  }

   516→   *  visually distinct from the auto-generated ball-by-ball commentary. */
   517→  function addCommentary(text: string) {
   518→    const trimmed = text.trim()
   519→    if (!trimmed || !liveState) return
   520→    const inn = liveState.innings[liveState.currentInnings]
   521→    const overLabel = inn ? oversString(inn.balls) : "0.0"
   522→    const entry = {
   523→      id: Math.random().toString(36).slice(2),
   524→      ballNo: liveState.ballNo,
   525→      over: overLabel,
   526→      text: trimmed,
   527→      kind: "manual" as const,
   528→      scorerId: liveState.scorerId ?? undefined,
   529→      timestamp: new Date().toISOString(),
   530→    }
   531→    const next: LiveMatchState = {
   532→      ...liveState,
   533→      // Newest-first (matches the engine's pushCommentary pattern).
   534→      commentary: [entry, ...liveState.commentary].slice(0, 120),
   535→    }
   536→    setHistory((h) => [...h, liveState].slice(-50))
   537→    pushState(next)
   538→    persist(next)
   539→    setCommentaryText("")
   540→  }

   542→  function handleBall(event: BallEvent) {
   543→    if (!liveState) return
   544→    setHistory((h) => [...h, liveState].slice(-50))
   545→    const next = applyBall(liveState, event)
   546→    pushState(next)
   547→    persist(next)
   548→    // Audit trail — log every ball scored
   549→    if (id) logAudit(id, liveState.scorerId ?? null, "ball_scored", next.ballNo, `${event.kind}${'runs' in event ? ` ${event.runs}` : ''}`)
   550→  }

   552→  function handleUndo() {
   553→    if (history.length === 0) {
   554→      toast.info("Nothing to undo")
   555→      return
   556→    }
   557→    const prev = history[history.length - 1]
   558→    setHistory((h) => h.slice(0, -1))
   559→    pushState(prev)
   560→    persist(prev)
   561→    toast.success("Undid last action")
   562→    // Audit trail — log every undo
   563→    if (id) logAudit(id, prev.scorerId ?? null, "undo", prev.ballNo, "Reverted last action")
   564→  }

   566→  function startInnings(
   567→    battingFirstId: string,
   568→    strikerId: string,
   569→    nonStrikerId: string,
   570→    bowlerId: string
   571→  ) {
   572→    const battingPlayers = playersByTeam[battingFirstId] ?? []
   573→    const bowlingPlayers =
   574→      playersByTeam[battingFirstId === match!.team1Id ? match!.team2Id : match!.team1Id] ?? []
   575→    const striker = battingPlayers.find((p) => p.id === strikerId)
   576→    const nonStriker = battingPlayers.find((p) => p.id === nonStrikerId)
   577→    const bowler = bowlingPlayers.find((p) => p.id === bowlerId)
   578→    if (!striker || !nonStriker || !bowler) {
   579→      toast.error("Select striker, non-striker and bowler")
   580→      return
   581→    }
   582→    let state = initState(match!, battingFirstId)
   583→    // Set the scorer's ID on the state so every commentary entry records
   584→    // who scored it (closes scoring data gap #10 — scorer identity per ball).
   585→    state.scorerId = session?.scorer?.id ?? null
   586→    // Pre-populate the squad with the full batting-team roster so the
   587→    // "Did not bat" section of the public scorecard lists bench players.
   588→    // Captain/keeper tags come from the explicit Player model fields
   589→    // (was previously a brittle regex on the role string).
   590→    for (const p of battingPlayers) {
   591→      state = addSquadMember(state, p.id, p.name, {
   592→        isCaptain: !!p.isCaptain,
   593→        isKeeper: !!p.isKeeper,
   594→      })
   595→    }
   596→    state = addBatsman(state, striker.id, striker.name, {
   597→      isCaptain: !!striker.isCaptain,
   598→      isKeeper: !!striker.isKeeper,
   599→    })
   600→    state = addBatsman(state, nonStriker.id, nonStriker.name, {
   601→      isCaptain: !!nonStriker.isCaptain,
   602→      isKeeper: !!nonStriker.isKeeper,
   603→    })
   604→    state = setBatsmen(state, striker.id, nonStriker.id)
   605→    state = setBowler(state, bowler.id, bowler.name)
   606→    setHistory([])
   607→    pushState(state)
   608→    persist(state)
   609→    toast.success("Innings started — status set to LIVE")
   610→    if (id) logAudit(id, state.scorerId ?? null, "innings_started", 0, `Innings ${state.currentInnings + 1}: ${battingFirstId === match!.team1Id ? t1?.shortName : t2?.shortName} batting first`)
   611→  }

   613→  function bringInBatsman(playerId: string) {
   614→    if (!liveState) return
   615→    const battingPlayers = playersByTeam[liveState.innings[liveState.currentInnings].battingTeamId] ?? []
   616→    const p = battingPlayers.find((x) => x.id === playerId)
   617→    if (!p) return
   618→    let state = addBatsman(liveState, p.id, p.name, {
   619→      isCaptain: !!p.isCaptain,
   620→      isKeeper: !!p.isKeeper,
   621→    })
   622→    // Fill whichever slot is currently empty — the striker's end if a normal
   623→    // wicket fell (strikerId === null), or the non-striker's end if a run-out
   624→    // happened at the non-striker's end (nonStrikerId === null). If both are
   625→    // empty (innings restart, edge case), default to striker.
   626→    const fillStriker = liveState.strikerId === null
   627→    const fillNonStriker = !fillStriker && liveState.nonStrikerId === null
   628→    state = {
   629→      ...state,
   630→      strikerId: fillStriker ? p.id : state.strikerId,
   631→      nonStrikerId: fillNonStriker ? p.id : state.nonStrikerId,
   632→    }
   633→    setHistory((h) => [...h, liveState].slice(-50))
   634→    pushState(state)
   635→    persist(state)
   636→    toast.success(`${p.name} in`)
   637→  }

   639→  function pickNextBowler(playerId: string) {
   640→    if (!liveState) return
   641→    // Cricket rule: a bowler cannot bowl two consecutive overs.
   642→    // The previous bowler is tracked in lastBowlerRef (updated whenever
   643→    // bowlerId changes from a value to null at over end).
   644→    if (playerId === lastBowlerRef.current) {
   645→      toast.error("A bowler cannot bowl two consecutive overs", {
   646→        description: "Pick a different bowler for this over.",
   647→      })
   648→      return
   649→    }
   650→    const bowlingPlayers = playersByTeam[liveState.innings[liveState.currentInnings].bowlingTeamId] ?? []
   651→    const p = bowlingPlayers.find((x) => x.id === playerId)
   652→    if (!p) return
   653→    setHistory((h) => [...h, liveState].slice(-50))
   654→    const state = setBowler(liveState, p.id, p.name)
   655→    pushState(state)
   656→    persist(state)
   657→    toast.success(`${p.name} to bowl`)
   658→  }

   693→  /** Swap the striker and non-striker ends. Used when the auto strike-rotation
   694→   *  logic disagrees with reality (e.g. batsmen crossed on a caught-and-
   695→   *  bowled, odd run on a wide, runner substituted, etc.). Allowed at any

   697→  function swapEnds() {
   698→    if (!liveState) return

   704→    const next = {
   705→      ...liveState,
   706→      strikerId: liveState.nonStrikerId,
   707→      nonStrikerId: liveState.strikerId,
   708→    }

   710→    persist(next)
   711→    toast.success("Ends swapped")

  1298→function SetupPanel({
  1299→  t1,
  1300→  t2,
  1301→  t1Players,
  1302→  t2Players,
  1303→  onStart,
  1304→  tokenReady,
  1305→  connected,
  1306→  matchId,
  1307→  token,
  1308→}: {
  1309→  t1: Player["team"] | undefined
  1310→  t2: Player["team"] | undefined
  1311→  t1Players: Player[]
  1312→  t2Players: Player[]
  1313→  onStart: (battingFirstId: string, strikerId: string, nonStrikerId: string, bowlerId: string) => void
  1314→  tokenReady: boolean
  1315→  connected: boolean
  1316→  matchId?: string
  1317→  token?: string | null
  1318→}) {
  1319→  const [battingFirstId, setBattingFirstId] = useState(t1?.id ?? "")
  1320→  const [strikerId, setStrikerId] = useState("")
  1321→  const [nonStrikerId, setNonStrikerId] = useState("")
  1322→  const [bowlerId, setBowlerId] = useState("")
  1323→  const [tossWinnerId, setTossWinnerId] = useState("")
  1324→  const [tossDecision, setTossDecision] = useState("")
  1325→  // Quick-add player state — lets the scorer create a player on-the-fly
  1326→  // if someone isn't in the team roster (late arrival, missing from CMS).
  1327→  const [showAddPlayer, setShowAddPlayer] = useState(false)
  1328→  const [addPlayerName, setAddPlayerName] = useState("")
  1329→  const [addPlayerTeam, setAddPlayerTeam] = useState("")
  1330→  const [addPlayerRole, setAddPlayerRole] = useState("Batsman")
  1331→  const [addingPlayer, setAddingPlayer] = useState(false)
  1332→  // Locally created players (merged with the prop data so dropdowns update
  1333→  // without needing a refetch from the parent).
  1334→  const [extraPlayers, setExtraPlayers] = useState<Player[]>([])
  1335→
  1336→  // Merge prop players with locally-created ones.
  1337→  const allT1Players = [...t1Players, ...extraPlayers.filter((p) => p.teamId === t1?.id)]
  1338→  const allT2Players = [...t2Players, ...extraPlayers.filter((p) => p.teamId === t2?.id)]
  1339→
  1340→  // When the toss winner + decision are both set, auto-fill the batting-first
  1341→  // team: if the winner chose "bat", they bat first; if "bowl", the other
  1342→  // team bats first. Also persist the toss to the match record.
  1343→  function applyToss(winnerId: string, decision: string) {
  1344→    if (!winnerId || !decision || !t1 || !t2) return
  1345→    const battingFirst = decision === "bat" ? winnerId : (winnerId === t1.id ? t2.id : t1.id)
  1346→    setBattingFirstId(battingFirst)
  1347→    setStrikerId("")
  1348→    setNonStrikerId("")
  1349→    setBowlerId("")
  1350→    // Persist the toss to the match record (best-effort — don't block start)
  1351→    if (matchId && token) {
  1352→      fetch(`/api/matches/${matchId}/toss`, {
  1353→        method: "PUT",
  1354→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
  1355→        body: JSON.stringify({ tossWinnerId: winnerId, tossDecision: decision }),
  1356→      }).catch(() => {})
  1357→    }
  1358→  }
  1359→
  1360→  const battingPlayers =
  1361→    battingFirstId === t1?.id ? allT1Players : battingFirstId === t2?.id ? allT2Players : []
  1362→  const bowlingPlayers =
  1363→    battingFirstId === t1?.id
  1364→      ? allT2Players
  1365→      : battingFirstId === t2?.id
  1366→        ? allT1Players
  1367→        : []
  1368→
  1369→  // Quick-add a player to a team — creates via API, adds to local state,
  1370→  // and auto-selects them in the specified slot (striker/non-striker/bowler).

  1372→    if (!name.trim() || !teamId || !token) return
  1373→    setAddingPlayer(true)
  1374→    try {
  1375→      const res = await fetch("/api/players", {
  1376→        method: "POST",
  1377→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
  1378→        body: JSON.stringify({ name: name.trim(), teamId, role, battingStyle: "Right-hand bat", bowlingStyle: role === "Bowler" ? "Right-arm medium" : "" }),
  1379→      })
  1380→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
  1381→      const player: Player = await res.json()
  1382→      setExtraPlayers((prev) => [...prev, player])
  1383→      if (slot === "striker") setStrikerId(player.id)
  1384→      else if (slot === "nonStriker") setNonStrikerId(player.id)
  1385→      else if (slot === "bowler") setBowlerId(player.id)
  1386→      setAddPlayerName("")
  1387→      setShowAddPlayer(false)
  1388→      toast.success(`${player.name} added`)
  1389→    } catch (e) {
  1390→      toast.error("Failed to add player", { description: String(e) })
  1391→    } finally {
  1392→      setAddingPlayer(false)
  1393→    }
  1394→  }
  1395→
  1396→  return (
  1397→    <div className="mt-5 space-y-4">
  1398→      <div className="rounded-lg border bg-muted/30 p-4">
  1399→        <div className="mb-3 flex items-center gap-2">
  1400→          <Radio className="h-4 w-4 text-red-600" />
  1401→          <h3 className="text-sm font-bold uppercase tracking-wide">
  1402→            Start live innings
  1403→          </h3>
  1404→        </div>
  1405→        {/* Toss section */}
  1406→        <div className="mb-4 rounded-md border border-amber-200 bg-amber-50 p-3 dark:border-amber-900/40 dark:bg-amber-900/10">
  1407→          <div className="mb-2 flex items-center gap-1.5">
  1408→            <span className="text-xs font-bold uppercase tracking-wide text-amber-700 dark:text-amber-400">Toss</span>
  1409→            <span className="text-[10px] text-muted-foreground">(optional)</span>
  1410→          </div>
  1411→          <div className="grid gap-2 sm:grid-cols-2">
  1412→            <Select
  1413→              value={tossWinnerId}
  1414→              onValueChange={(v) => {
  1415→                setTossWinnerId(v)
  1416→                if (tossDecision) applyToss(v, tossDecision)
  1417→              }}
  1418→            >
  1419→              <SelectTrigger className="w-full"><SelectValue placeholder="Won by" /></SelectTrigger>
  1420→              <SelectContent>
  1421→                {t1 && <SelectItem value={t1.id}>{t1.name}</SelectItem>}
  1422→                {t2 && <SelectItem value={t2.id}>{t2.name}</SelectItem>}
  1423→              </SelectContent>
  1424→            </Select>
  1425→            <Select
  1426→              value={tossDecision}
  1427→              onValueChange={(v) => {
  1428→                setTossDecision(v)
  1429→                if (tossWinnerId) applyToss(tossWinnerId, v)
  1430→              }}
  1431→            >
  1432→              <SelectTrigger className="w-full"><SelectValue placeholder="Elected to" /></SelectTrigger>
  1433→              <SelectContent>
  1434→                <SelectItem value="bat">Bat</SelectItem>
  1435→                <SelectItem value="bowl">Bowl</SelectItem>
  1436→              </SelectContent>
  1437→            </Select>
  1438→          </div>
  1439→        </div>
  1440→        <div className="grid gap-4 sm:grid-cols-2">
  1441→          <div className="space-y-1.5">
  1442→            <label className="text-xs font-semibold text-muted-foreground">
  1443→              Batting first

  1887→/** Build the dismissal summary string from the structured inputs (this is
  1888→ *  what the public scorecard renders next to the batsman's name). */
  1889→function buildDismissalText(args: {
  1890→  wicketType: WicketType
  1891→  bowlerName?: string
  1892→  fielderName?: string
  1893→}): string {
  1894→  const bowler = args.bowlerName ?? "?"
  1895→  const fielder = args.fielderName
  1896→  switch (args.wicketType) {
  1897→    case "bowled":
  1898→      return `b ${bowler}`
  1899→    case "caught":
  1900→      // Caught & bowled when no fielder is selected OR the fielder is the bowler.
  1901→      if (!fielder || fielder === bowler) return `c &b ${bowler}`
  1902→      return `c ${fielder} b ${bowler}`
  1903→    case "lbw":
  1904→      return `lbw b ${bowler}`
  1905→    case "stumped":
  1906→      return fielder ? `st ${fielder} b ${bowler}` : `st ${bowler}`
  1907→    case "runout":
  1908→      return fielder ? `run out (${fielder})` : `run out`
  1909→    case "hitwkt":
  1910→      return `hit wkt b ${bowler}`
  1911→    case "retiredHurt":
  1912→      return `retired hurt`
  1913→    case "retiredOut":
  1914→      return `retired out`
  1915→  }

  1918→function WicketDialog({
  1919→  open,

  1930→  striker: { id: string; name: string }
  1931→  nonStriker: { id: string; name: string }
  1932→  bowler: { id: string; name: string }
  1933→  fieldingPlayers: { id: string; name: string; role?: string }[]
  1934→  /** True when the current ball is a free hit — bowler-credited dismissals
  1935→   *  (bowled/caught/lbw/stumped/hit-wkt) will be REJECTED by the engine.
  1936→   *  The dialog shows a hint banner so the admin knows to pick Run Out or
  1937→   *  Retired Out instead. */
  1938→  freeHit?: boolean

  1940→    kind: "wicket"

  1947→  const [wicketType, setWicketType] = useState<WicketType>("bowled")
  1948→  const [outBatsmanId, setOutBatsmanId] = useState<string>(striker.id)
  1949→  const [fielderId, setFielderId] = useState<string>("")
  1950→  // Note: this component is conditionally rendered by the parent only when
  1951→  // `wicketOpen` is true, so it remounts fresh each time the user opens the
  1952→  // dialog — useState initial values act as the "reset on open".
  1953→
  1954→  // The wicket-keeper on the fielding side (auto-prefilled for stumped).
  1955→  // Uses the explicit `isKeeper` flag on the Player model — was previously
  1956→  // a brittle regex on the role string.
  1957→  const keeperId = useMemo(() => {
  1958→    const k = fieldingPlayers.find((p) => p.isKeeper)
  1959→    return k?.id ?? ""

  1963→  // selections during render. The picker's `value` reads from these, so the
  1964→  // UI stays in sync without needing an effect to "correct" the user's
