   120→function toFormState(b: Banner): BannerFormState {
   121→  return {
   122→    id: b.id,
   123→    type: b.type,
   124→    title: b.title,
   125→    placement: b.placement,
   126→    order: b.order,
   127→    published: b.published,
   128→    startDate: b.startDate ? b.startDate.slice(0, 10) : "",
   129→    endDate: b.endDate ? b.endDate.slice(0, 10) : "",
   130→    weight: b.weight,
   131→    imageUrl: b.imageUrl ?? "",
   132→    linkUrl: b.linkUrl ?? "",
   133→    altText: b.altText ?? "",
   134→    adsenseClient: b.adsenseClient ?? "",
   135→    adsenseSlot: b.adsenseSlot ?? "",
   136→    adsenseFormat: b.adsenseFormat ?? "auto",
   137→    responsive: b.responsive,
   138→    htmlContent: b.htmlContent ?? "",
   139→  }
   140→}

   142→function isDateActive(b: Banner): boolean {
   143→  const now = Date.now()
   144→  const start = b.startDate ? new Date(b.startDate).getTime() : -Infinity
   145→  const end = b.endDate ? new Date(b.endDate).getTime() : Infinity
   146→  return start <= now && now <= end
   147→}

   165→  async function handleSubmit() {

   172→      toast.error("Image banner needs an image URL — upload an image first")
   173→      return

   175→    if (form.type === "adsense" && (!form.adsenseClient || !form.adsenseSlot)) {
   176→      toast.error("AdSense banner needs both the publisher ID (ca-pub-...) and the slot ID")
   177→      return
   178→    }
   179→    if (form.type === "html" && !form.htmlContent.trim()) {
   180→      toast.error("HTML banner needs the HTML snippet")
   181→      return
   182→    }
   183→    setSaving(true)
   184→    try {
   185→      const payload: Partial<Banner> = {
   186→        type: form.type,
   187→        title: form.title.trim(),
   188→        placement: form.placement,
   189→        order: form.order,
   190→        published: form.published,
   191→        startDate: form.startDate ? new Date(form.startDate).toISOString() : null,
   192→        endDate: form.endDate ? new Date(form.endDate).toISOString() : null,
   193→        weight: form.weight,
   194→        imageUrl: form.imageUrl || null,
   195→        linkUrl: form.linkUrl || null,
   196→        altText: form.altText || null,
   197→        adsenseClient: form.adsenseClient || null,
   198→        adsenseSlot: form.adsenseSlot || null,
   199→        adsenseFormat: form.adsenseFormat || null,
   200→        responsive: form.responsive,
   201→        htmlContent: form.htmlContent || null,
   202→      }
   203→      if (form.id) {
   204→        await api.banners.update(form.id, payload)
   205→        toast.success("Banner updated")
   206→      } else {
   207→        await api.banners.create(payload)
   208→        toast.success("Banner created")
   209→      }
   210→      setOpen(false)
   211→      refetch()
   212→    } catch (e) {
   213→      toast.error("Failed to save banner", { description: String(e) })
   214→    } finally {
   215→      setSaving(false)
   216→    }
   217→  }

   577→                <div className="grid gap-1.5">
   578→                  <Label htmlFor="b-order">Order (lower shows first)</Label>
   579→                  <Input
   580→                    id="b-order"
   581→                    type="number"
   582→                    value={form.order}
   583→                    onChange={(e) => set("order", Number(e.target.value) || 0)}
   584→                  />
   585→                </div>
   586→              </div>
   587→              <div className="grid grid-cols-2 gap-3">
   588→                <div className="grid gap-1.5">
   589→                  <Label htmlFor="b-start">Start date (optional)</Label>
   590→                  <Input
   591→                    id="b-start"
   592→                    type="date"
   593→                    value={form.startDate}
   594→                    onChange={(e) => set("startDate", e.target.value)}
   595→                  />
   596→                </div>
   597→                <div className="grid gap-1.5">
   598→                  <Label htmlFor="b-end">End date (optional)</Label>
   599→                  <Input
   600→                    id="b-end"
   601→                    type="date"
   602→                    value={form.endDate}
   603→                    onChange={(e) => set("endDate", e.target.value)}
   604→                  />
   605→                </div>
   606→              </div>
   607→              <div className="grid grid-cols-2 gap-3">
   608→                <div className="grid gap-1.5">
   609→                  <Label htmlFor="b-weight">Rotation weight (higher = more often)</Label>
   610→                  <Input
   611→                    id="b-weight"
   612→                    type="number"
   613→                    min="1"
   614→                    value={form.weight}
   615→                    onChange={(e) => set("weight", Number(e.target.value) || 1)}
   616→                  />
   617→                  <p className="text-[10px] text-muted-foreground">
   618→                    Defaults to 1 (equal weighting). Set 5 to show 5× more
   619→                    often than weight-1 banners in the same slot.
   620→                  </p>
   621→                </div>
   622→                <div className="flex items-end gap-2 pb-1">
   623→                  <Switch
   624→                    checked={form.published}
   625→                    onCheckedChange={(v) => set("published", v)}
   626→                    id="b-published"
   627→                  />
   628→                  <Label htmlFor="b-published" className="cursor-pointer">
   629→                    Published
   630→                  </Label>
   631→                </div>
   632→              </div>
   633→            </div>
   634→          </div>
   635→
   636→          <DialogFooter>
   637→            <Button variant="outline" onClick={() => setOpen(false)}>
   638→              Cancel
   639→            </Button>
   640→            <Button onClick={handleSubmit} disabled={saving}>
   641→              {saving ? "Saving…" : form.id ? "Save Changes" : "Create Banner"}
   642→            </Button>
   643→          </DialogFooter>
   644→        </DialogContent>
   645→      </Dialog>
   646→
   647→      {/* Delete confirmation */}
   648→      <AlertDialog
   649→        open={!!deleteId}
   650→        onOpenChange={(o) => !o && setDeleteId(null)}
   651→      >
   652→        <AlertDialogContent>
   653→          <AlertDialogHeader>
   654→            <AlertDialogTitle className="flex items-center gap-2">
   655→              <AlertTriangle className="h-5 w-5 text-amber-500" /> Delete this banner?
   656→            </AlertDialogTitle>
   657→            <AlertDialogDescription>
   658→              The banner will be removed from the rotation pool immediately.
   659→              Stats (impressions + clicks) are lost. This cannot be undone.
   660→            </AlertDialogDescription>
   661→          </AlertDialogHeader>
   662→          <AlertDialogFooter>
   663→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   664→            <AlertDialogAction
   665→              className="bg-destructive text-white hover:bg-destructive/90"
   666→              onClick={() => deleteId && handleDelete(deleteId)}
   667→            >
   668→              Delete
   669→            </AlertDialogAction>
   670→          </AlertDialogFooter>
   671→        </AlertDialogContent>
   672→      </AlertDialog>
   673→    </AdminShell>
   674→  )
   675→}
