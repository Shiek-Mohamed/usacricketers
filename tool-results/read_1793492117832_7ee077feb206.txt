   117→  /* ------------------ Keyboard shortcuts (power-user scorers) ------------------
   118→   * 0,1,2,3,4,6 → score N runs | . / - → dot | w → wicket dialog |
   119→   * e → swap ends | u → undo | d → declare (Test) | s → save.
   120→   * No deps array → re-subscribes on every render with the latest closures.
   121→   * The function declarations (handleBall, swapEnds, etc.) are hoisted by
   122→   * JS so they're in scope at call-time even though defined later in the
   123→   * component body. The lint rule below is overly cautious about this
   124→   * pattern — it's safe because the effect re-runs every render. */
   125→  /* eslint-disable react-hooks/immutability */
   126→  useEffect(() => {
   127→    function isTyping(el: Element | null): boolean {
   128→      if (!el) return false
   129→      const tag = el.tagName?.toLowerCase()
   130→      return tag === "input" || tag === "textarea" || tag === "select" ||
   131→        el.getAttribute("role") === "combobox" ||
   132→        (el as HTMLElement)?.isContentEditable === true
   133→    }
   134→    function onKey(e: KeyboardEvent) {
   135→      if (e.metaKey || e.ctrlKey || e.altKey) return
   136→      if (isTyping(document.activeElement)) return
   137→      if (wicketOpen) return
   138→      if (!liveState || liveState.status !== "live") return
   139→      if (!liveState.strikerId || !liveState.nonStrikerId || !liveState.bowlerId) return
   140→      const k = e.key.toLowerCase()
   141→      if (["0", "1", "2", "3", "4", "6"].includes(k)) {
   142→        e.preventDefault()
   143→        handleBall({ kind: "run", runs: Number(k) })
   144→      } else if (k === "." || k === "-") {
   145→        e.preventDefault()
   146→        handleBall({ kind: "run", runs: 0 })
   147→      } else if (k === "w") {
   148→        e.preventDefault()
   149→        setWicketOpen(true)
   150→      } else if (k === "e") {
   151→        e.preventDefault()
   152→        swapEnds()
   153→      } else if (k === "u") {
   154→        e.preventDefault()
   155→        handleUndo()
   156→      } else if (k === "d") {
   157→        e.preventDefault()
   158→        handleDeclare()
   159→      } else if (k === "s") {
   160→        e.preventDefault()
   161→        if (liveState) {
   162→          persist(liveState)
   163→          toast.success("Scorecard saved")
   164→        }
   165→      }
   166→    }
   167→    window.addEventListener("keydown", onKey)
   168→    return () => window.removeEventListener("keydown", onKey)
   169→  })

   211→  function persist(state: LiveMatchState) {
   212→    if (saveTimer.current) clearTimeout(saveTimer.current)
   213→    saveTimer.current = setTimeout(async () => {
   214→      try {
   215→        await api.matches.update(match!.id, {
   216→          scorecard: stateToScorecard(state),
   217→          status: state.status,
   218→          result: state.result,
   219→        })
   220→        refetch()
   221→      } catch (e) {
   222→        // Surface persistence errors so the operator knows the DB write failed
   223→        // (was previously silent — caused confusion when "Innings started"
   224→        // toast showed but the match status didn't actually update).
   225→        // The live socket broadcast is still the priority — this toast
   226→        // doesn't block scoring, just informs the operator something's wrong.
   227→        const msg = e instanceof Error ? e.message : String(e).slice(0, 200)
   228→        toast.error("Persist failed — match state not saved", {
   229→          description: msg,
   230→        })
   231→      }
   232→    }, 800)
   233→  }

   235→  function handleBall(event: BallEvent) {
   236→    if (!liveState) return
   237→    setHistory((h) => [...h, liveState].slice(-50))
   238→    const next = applyBall(liveState, event)
   239→    pushState(next)
   240→    persist(next)

   243→  function handleUndo() {
   244→    if (history.length === 0) {
   245→      toast.info("Nothing to undo")
   246→      return
   247→    }
   248→    const prev = history[history.length - 1]
   249→    setHistory((h) => h.slice(0, -1))
   250→    pushState(prev)
   251→    persist(prev)
   252→    toast.success("Undid last action")
   253→  }

   255→  function startInnings(
   256→    battingFirstId: string,
   257→    strikerId: string,
   258→    nonStrikerId: string,
   259→    bowlerId: string
   260→  ) {
   261→    const battingPlayers = playersByTeam[battingFirstId] ?? []
   262→    const bowlingPlayers =
   263→      playersByTeam[battingFirstId === match!.team1Id ? match!.team2Id : match!.team1Id] ?? []
   264→    const striker = battingPlayers.find((p) => p.id === strikerId)
   265→    const nonStriker = battingPlayers.find((p) => p.id === nonStrikerId)
   266→    const bowler = bowlingPlayers.find((p) => p.id === bowlerId)
   267→    if (!striker || !nonStriker || !bowler) {
   268→      toast.error("Select striker, non-striker and bowler")
   269→      return
   270→    }
   271→    let state = initState(match!, battingFirstId)
   272→    // Pre-populate the squad with the full batting-team roster so the
   273→    // "Did not bat" section of the public scorecard lists bench players.
   274→    // Captain/keeper tags come from the explicit Player model fields
   275→    // (was previously a brittle regex on the role string).
   276→    for (const p of battingPlayers) {
   277→      state = addSquadMember(state, p.id, p.name, {
   278→        isCaptain: !!p.isCaptain,
   279→        isKeeper: !!p.isKeeper,
   280→      })
   281→    }
   282→    state = addBatsman(state, striker.id, striker.name, {
   283→      isCaptain: !!striker.isCaptain,
   284→      isKeeper: !!striker.isKeeper,
   285→    })
   286→    state = addBatsman(state, nonStriker.id, nonStriker.name, {
   287→      isCaptain: !!nonStriker.isCaptain,
   288→      isKeeper: !!nonStriker.isKeeper,
   289→    })
   290→    state = setBatsmen(state, striker.id, nonStriker.id)
   291→    state = setBowler(state, bowler.id, bowler.name)
   292→    setHistory([])
   293→    pushState(state)
   294→    persist(state)
   295→    toast.success("Innings started — status set to LIVE")
   296→  }

   298→  function bringInBatsman(playerId: string) {
   299→    if (!liveState) return
   300→    const battingPlayers = playersByTeam[liveState.innings[liveState.currentInnings].battingTeamId] ?? []
   301→    const p = battingPlayers.find((x) => x.id === playerId)
   302→    if (!p) return
   303→    let state = addBatsman(liveState, p.id, p.name, {
   304→      isCaptain: !!p.isCaptain,
   305→      isKeeper: !!p.isKeeper,
   306→    })
   307→    // Fill whichever slot is currently empty — the striker's end if a normal
   308→    // wicket fell (strikerId === null), or the non-striker's end if a run-out
   309→    // happened at the non-striker's end (nonStrikerId === null). If both are
   310→    // empty (innings restart, edge case), default to striker.
   311→    const fillStriker = liveState.strikerId === null
   312→    const fillNonStriker = !fillStriker && liveState.nonStrikerId === null
   313→    state = {
   314→      ...state,
   315→      strikerId: fillStriker ? p.id : state.strikerId,
   316→      nonStrikerId: fillNonStriker ? p.id : state.nonStrikerId,
   317→    }
   318→    setHistory((h) => [...h, liveState].slice(-50))
   319→    pushState(state)
   320→    persist(state)
   321→    toast.success(`${p.name} in`)
   322→  }

   324→  function pickNextBowler(playerId: string) {
   325→    if (!liveState) return
   326→    const bowlingPlayers = playersByTeam[liveState.innings[liveState.currentInnings].bowlingTeamId] ?? []
   327→    const p = bowlingPlayers.find((x) => x.id === playerId)
   328→    if (!p) return
   329→    setHistory((h) => [...h, liveState].slice(-50))
   330→    const state = setBowler(liveState, p.id, p.name)
   331→    pushState(state)
   332→    persist(state)
   333→    toast.success(`${p.name} to bowl`)
   334→  }

   336→  /** "I picked the wrong striker — let me re-pick." Pushes the current state
   337→   *  to the history stack (so Undo can recover it), then clears the strikerId
   338→   *  slot. The Picker re-shows automatically because `needStriker` becomes
   339→   *  true when strikerId === null. */
   340→  function changeStriker() {
   341→    if (!liveState || !liveState.strikerId) return
   342→    setHistory((h) => [...h, liveState].slice(-50))
   343→    const next = { ...liveState, strikerId: null as string | null }
   344→    pushState(next)
   345→    persist(next)
   346→    toast.info("Re-pick the striker")
   347→  }

   351→    if (!liveState || !liveState.nonStrikerId) return
   352→    setHistory((h) => [...h, liveState].slice(-50))
   353→    const next = { ...liveState, nonStrikerId: null as string | null }
   354→    pushState(next)
   355→    persist(next)
   356→    toast.info("Re-pick the non-striker")
   357→  }

   361→    if (!liveState || !liveState.bowlerId) return

   369→  /** Swap the striker and non-striker ends. Used when the auto strike-rotation
   370→   *  logic disagrees with reality (e.g. batsmen crossed on a caught-and-
   371→   *  bowled, odd run on a wide, runner substituted, etc.). Allowed at any
   372→   *  time — doesn't affect stats, just relabels who's on strike. */
   373→  function swapEnds() {
   374→    if (!liveState) return
   375→    if (!liveState.strikerId || !liveState.nonStrikerId) {
   376→      toast.error("Need both batsmen at the crease to swap ends")
   377→      return
   378→    }
   379→    setHistory((h) => [...h, liveState].slice(-50))
   380→    const next = {
   381→      ...liveState,
   382→      strikerId: liveState.nonStrikerId,
   383→      nonStrikerId: liveState.strikerId,
   384→    }
   385→    pushState(next)
   386→    persist(next)
   387→    toast.success("Ends swapped")
   388→  }

   390→  function reset() {
   391→    setHistory([])
   392→    // clear live cache
   393→    // (clearState not destructured here; use pushState(null) approach via a clear)
   394→    pushState({ ...liveState!, status: "live", result: "" })

   397→  /** Test-match declaration: captain ends the current innings without being
   398→   *  all out. No-op for limited-overs matches (those don't have declarations).
   399→   *  Pushes the prev state to history (so Undo can recover) then runs the
   400→   *  engine's declareInnings which marks `declared: true` + sets up the
   401→   *  next innings / computes the result. */
   402→  function handleDeclare() {
   403→    if (!liveState) return
   404→    if (liveState.maxOvers > 0) {
   405→      toast.error("Declarations are only available in Test matches")
   406→      return
   407→    }
   408→    if (liveState.status !== "live") {
   409→      toast.error("Can't declare — innings already ended")
   410→      return
   411→    }
   412→    setHistory((h) => [...h, liveState].slice(-50))
   413→    const next = declareInnings(liveState)
   414→    pushState(next)
   415→    persist(next)
   416→    toast.success("Innings declared")
   417→  }

   419→  /** Enforce the follow-on in a Test match. Only valid when Team A leads by
   420→   *  200+ runs after Team B's 1st innings. Team B bats again immediately. */
   421→  function handleFollowOn() {
   422→    if (!liveState) return
   423→    setHistory((h) => [...h, liveState].slice(-50))
   424→    const next = enforceFollowOn(liveState)
   425→    if (next === liveState) {
   426→      toast.error("Follow-on requires a 200+ run lead in a Test after innings 1")
   427→      return
   428→    }
   429→    pushState(next)
   430→    persist(next)
   431→    toast.success("Follow-on enforced")
   432→  }

   434→  /** Start a super-over after a tied limited-overs match. Creates 2 new
   435→   *  1-over innings (one per team). */
   436→  function handleSuperOver() {

   442→      return
   443→    }

   445→    persist(next)

   851→function LiveScoreboard({
   852→  state,
   853→  onChangeStriker,
   854→  onChangeNonStriker,
   855→  onChangeBowler,
   856→  onSwapEnds,
   857→}: {
   858→  state: LiveMatchState
   859→  onChangeStriker?: () => void
   860→  onChangeNonStriker?: () => void
   861→  onChangeBowler?: () => void
   862→  /** Swap striker ↔ non-striker — override the auto strike-rotation logic
   863→   *  when it disagrees with reality (caught-and-bowled crossing, etc.).
   864→   *  Allowed at any time (not over-boundary-gated) — doesn't affect stats. */
   865→  onSwapEnds?: () => void
   866→}) {
   867→  const inn = state.innings[state.currentInnings]
   868→  const striker = inn.batsmen.find((b) => b.id === state.strikerId)
   869→  const nonStriker = inn.batsmen.find((b) => b.id === state.nonStrikerId)
   870→  const bowler = inn.bowlers.find((b) => b.id === state.bowlerId)
   871→  const overs = oversString(inn.balls)
   872→  const runRate = inn.balls > 0 ? (inn.runs / inn.balls * 6).toFixed(2) : "0.00"
   873→  const target = inn.target
   874→  const ballsLeft = state.maxOvers > 0 ? state.maxOvers * 6 - inn.balls : null
   875→  const runsNeeded = target ? target - inn.runs : null
   876→  const reqRate =
   877→    target && ballsLeft && ballsLeft > 0
   878→      ? ((runsNeeded! / ballsLeft) * 6).toFixed(2)
   879→      : null
   880→
   881→  // A slot can be safely "Changed" only at over-boundaries — mid-over swaps
   882→  // would mess up the bowler's figures + the batsman's balls-faced count.
   883→  // Disable the Change button once any ball of the current over has been
   884→  // bowled (i.e. balls % 6 !== 0). The exception is the bowler slot after a
   885→  // wicket (still mid-over) — the bowler can be subbed off due to injury in
   886→  // real cricket, but for our MVP we keep the rule simple: only at over
   887→  // boundaries. Undo is still available mid-over for fixing mistakes.
   888→  const atOverBoundary = inn.balls % 6 === 0
   889→  const canChangeBatsmen = atOverBoundary && state.status === "live"
   890→  const canChangeBowler = atOverBoundary && state.status === "live"
   891→
   892→  return (
   893→    <div className="rounded-lg border bg-gradient-to-br from-slate-900 to-slate-800 p-4 text-white sm:p-5">
   894→      <div className="flex flex-wrap items-end justify-between gap-3">
   895→        <div>
   896→          <div className="text-xs uppercase tracking-wide text-slate-400">
   897→            {inn.battingTeamShortName} · Innings {state.currentInnings + 1}
   898→            {target ? ` · need ${runsNeeded} runs` : ""}
   899→          </div>
   900→          <div className="mt-1 text-4xl font-extrabold tabular-nums">
   901→            {inn.runs}/{inn.wickets}
   902→            <span className="ml-2 text-lg font-normal text-slate-300">
   903→              ({overs} ov)
   904→            </span>
   905→            {inn.freeHit && (
   906→              <span className="ml-3 inline-flex items-center gap-1 rounded-full bg-amber-500 px-2.5 py-0.5 text-xs font-extrabold uppercase tracking-wide text-amber-950 animate-pulse">
   907→                <span className="h-1.5 w-1.5 rounded-full bg-amber-950" />
   908→                Free hit
   909→              </span>
   910→            )}
   911→          </div>
   912→          <div className="mt-1 text-xs text-slate-300">
   913→            CRR {runRate}
   914→            {reqRate && ` · RRR ${reqRate}`}
   915→            {ballsLeft !== null && ` · ${ballsLeft} balls left`}
   916→          </div>
   917→        </div>
   918→      </div>
   919→      <div className="mt-3 grid grid-cols-2 gap-3 text-xs">
   920→        <div className="rounded-md bg-white/10 p-2.5">
   921→          <div className="flex items-center justify-between">
   922→            <div className="text-[10px] uppercase text-slate-300">Striker</div>
   923→            {striker && onChangeStriker && (
   924→              <ChangeButton
   925→                onClick={onChangeStriker}
   926→                disabled={!canChangeBatsmen}
   927→                hint={
   928→                  canChangeBatsmen
   929→                    ? "Re-pick striker"
   930→                    : "Wait for end of over (or use Undo)"
   931→                }
   932→              />
   933→            )}
   934→          </div>
   935→          <div className="font-bold">{striker?.name ?? "—"}</div>
   936→          <div className="text-slate-300">
   937→            {striker ? `${striker.runs} (${striker.balls})` : "—"}
   938→          </div>
   939→        </div>
   940→        <div className="relative rounded-md bg-white/10 p-2.5">
   941→          <div className="flex items-center justify-between">
   942→            <div className="text-[10px] uppercase text-slate-300">Non-striker</div>
   943→            {nonStriker && onChangeNonStriker && (
   944→              <ChangeButton
   945→                onClick={onChangeNonStriker}
   946→                disabled={!canChangeBatsmen}
   947→                hint={
   948→                  canChangeBatsmen
   949→                    ? "Re-pick non-striker"
   950→                    : "Wait for end of over (or use Undo)"
   951→                }
   952→              />
   953→            )}
   954→          </div>
   955→          <div className="font-bold">{nonStriker?.name ?? "—"}</div>
   956→          <div className="text-slate-300">
   957→            {nonStriker ? `${nonStriker.runs} (${nonStriker.balls})` : "—"}
   958→          </div>
   959→          {/* Swap-ends button — pinned to the inner edge between the two
   960→              batsman cells. Visible only when both batsmen are at the crease.
   961→              Always enabled (not over-boundary-gated) because swapping ends
   962→              doesn't affect any stats — it's just a relabel. */}
   963→          {striker && nonStriker && onSwapEnds && (
   964→            <button
   965→              type="button"
   966→              onClick={onSwapEnds}
   967→              title="Swap striker ↔ non-striker (override the auto strike-rotation)"
   968→              aria-label="Swap striker and non-striker ends"
   969→              className="absolute -left-[14px] top-1/2 grid h-7 w-7 -translate-y-1/2 place-items-center rounded-full border border-white/20 bg-slate-700 text-white shadow-md transition-colors hover:bg-slate-600"
   970→            >
   971→              <ArrowLeftRight className="h-3.5 w-3.5" />
   972→            </button>
   973→          )}
   974→        </div>
   975→        <div className="col-span-2 rounded-md bg-white/10 p-2.5">
   976→          <div className="flex items-center justify-between">
   977→            <div className="text-[10px] uppercase text-slate-300">Bowler</div>
   978→            {bowler && onChangeBowler && (
   979→              <ChangeButton
   980→                onClick={onChangeBowler}
   981→                disabled={!canChangeBowler}
   982→                hint={
   983→                  canChangeBowler
   984→                    ? "Re-pick bowler"
   985→                    : "Wait for end of over (or use Undo)"
   986→                }
   987→              />
   988→            )}
   989→          </div>
   990→          <div className="flex items-center justify-between">
   991→            <span className="font-bold">{bowler?.name ?? "—"}</span>
   992→            <span className="text-slate-300">
   993→              {bowler
   994→                ? `${oversString(bowler.balls)} - ${bowler.maidens} - ${bowler.runsConceded} - ${bowler.wickets}`
   995→                : "—"}
   996→            </span>
   997→          </div>
   998→        </div>
   999→      </div>
  1000→    </div>
  1001→  )
  1002→}

  1103→
  1104→const WICKET_TYPE_OPTIONS: { value: WicketType; label: string; hint: string }[] = [
  1105→  { value: "bowled", label: "Bowled", hint: "b Bowler" },
  1106→  { value: "caught", label: "Caught", hint: "c Fielder b Bowler" },
  1107→  { value: "lbw", label: "LBW", hint: "lbw b Bowler" },
  1108→  { value: "runout", label: "Run Out", hint: "run out (Fielder)" },
  1109→  { value: "stumped", label: "Stumped", hint: "st Keeper b Bowler" },
  1110→  { value: "hitwkt", label: "Hit Wicket", hint: "hit wkt b Bowler" },
  1111→  { value: "retiredHurt", label: "Retired Hurt", hint: "not out — may return" },
  1112→  { value: "retiredOut", label: "Retired Out", hint: "out — no ball" },

  1115→/** Build the dismissal summary string from the structured inputs (this is
  1116→ *  what the public scorecard renders next to the batsman's name). */
  1117→function buildDismissalText(args: {
  1118→  wicketType: WicketType
  1119→  bowlerName?: string
  1120→  fielderName?: string
  1121→}): string {
  1122→  const bowler = args.bowlerName ?? "?"
  1123→  const fielder = args.fielderName
  1124→  switch (args.wicketType) {
  1125→    case "bowled":
  1126→      return `b ${bowler}`
  1127→    case "caught":
  1128→      // Caught & bowled when no fielder is selected OR the fielder is the bowler.
  1129→      if (!fielder || fielder === bowler) return `c &b ${bowler}`
  1130→      return `c ${fielder} b ${bowler}`
  1131→    case "lbw":
  1132→      return `lbw b ${bowler}`
  1133→    case "stumped":
  1134→      return fielder ? `st ${fielder} b ${bowler}` : `st ${bowler}`
  1135→    case "runout":
  1136→      return fielder ? `run out (${fielder})` : `run out`
  1137→    case "hitwkt":
  1138→      return `hit wkt b ${bowler}`
  1139→    case "retiredHurt":
  1140→      return `retired hurt`
  1141→    case "retiredOut":
  1142→      return `retired out`
  1143→  }
  1144→}

  1146→function WicketDialog({
  1147→  open,
  1148→  onOpenChange,
  1149→  striker,
  1150→  nonStriker,
  1151→  bowler,
  1152→  fieldingPlayers,
  1153→  freeHit,
  1154→  onConfirm,
  1155→}: {
  1156→  open: boolean
  1157→  onOpenChange: (v: boolean) => void
  1158→  striker: { id: string; name: string }
  1159→  nonStriker: { id: string; name: string }
  1160→  bowler: { id: string; name: string }
  1161→  fieldingPlayers: { id: string; name: string; role?: string }[]
  1162→  /** True when the current ball is a free hit — bowler-credited dismissals
  1163→   *  (bowled/caught/lbw/stumped/hit-wkt) will be REJECTED by the engine.
  1164→   *  The dialog shows a hint banner so the admin knows to pick Run Out or
  1165→   *  Retired Out instead. */
  1166→  freeHit?: boolean
  1167→  onConfirm: (ev: {
  1168→    kind: "wicket"
  1169→    wicketType: WicketType
  1170→    outBatsmanId: string
  1171→    fielderIds: string[]
  1172→    dismissal: string
  1173→  }) => void
  1174→}) {
  1175→  const [wicketType, setWicketType] = useState<WicketType>("bowled")
  1176→  const [outBatsmanId, setOutBatsmanId] = useState<string>(striker.id)
  1177→  const [fielderId, setFielderId] = useState<string>("")
  1178→  // Note: this component is conditionally rendered by the parent only when
  1179→  // `wicketOpen` is true, so it remounts fresh each time the user opens the
  1180→  // dialog — useState initial values act as the "reset on open".
  1181→
  1182→  // The wicket-keeper on the fielding side (auto-prefilled for stumped).
  1183→  // Uses the explicit `isKeeper` flag on the Player model — was previously
  1184→  // a brittle regex on the role string.
  1185→  const keeperId = useMemo(() => {
  1186→    const k = fieldingPlayers.find((p) => p.isKeeper)
  1187→    return k?.id ?? ""
  1188→  }, [fieldingPlayers])
  1189→
  1190→  // Derived state — avoid setState-in-effect by computing the effective
  1191→  // selections during render. The picker's `value` reads from these, so the
  1192→  // UI stays in sync without needing an effect to "correct" the user's
  1193→  // last selection when the dismissal type changes.
  1194→  // • For any dismissal type other than run-out, the batsman out is always
  1195→  //   the striker (run-out can happen at the non-striker's end).
  1196→  // • For stumped with no fielder selected, auto-prefill the keeper.
  1197→  const effectiveOutBatsmanId =
  1198→    wicketType === "runout" ? outBatsmanId : striker.id
  1199→  const effectiveFielderId =
  1200→    wicketType === "stumped" && !fielderId && keeperId ? keeperId : fielderId
  1201→
  1202→  const needsFielder =
  1203→    wicketType === "caught" ||
  1204→    wicketType === "stumped" ||
  1205→    wicketType === "runout"
  1206→  const showBatsmanSelect = wicketType === "runout"
  1207→
  1208→  const fielderName = fieldingPlayers.find(
  1209→    (p) => p.id === effectiveFielderId
  1210→  )?.name
  1211→  const dismissal = buildDismissalText({
  1212→    wicketType,
  1213→    bowlerName: bowler.name,
  1214→    fielderName,
  1215→  })
  1216→
  1217→  const outBatsmanName =
  1218→    effectiveOutBatsmanId === striker.id
  1219→      ? striker.name
  1220→      : effectiveOutBatsmanId === nonStriker.id
  1221→        ? nonStriker.name
  1222→        : "?"
  1223→
  1224→  // Free-hit-derived warning: when the current ball is a free hit AND the
  1225→  // selected dismissal type is bowler-credited, the engine will REJECT the
  1226→  // wicket (batsman survives). Show a warning so the admin knows.
  1227→  const freeHitWillReject =
  1228→    !!freeHit &&
  1229→    BOWLER_CREDITED_WICKETS.has(wicketType) &&
  1230→    wicketType !== "retired"
  1231→
  1232→  function handleConfirm() {
  1233→    onConfirm({
  1234→      kind: "wicket",
  1235→      wicketType,
  1236→      outBatsmanId: effectiveOutBatsmanId,
  1237→      fielderIds: effectiveFielderId ? [effectiveFielderId] : [],
  1238→      dismissal,
  1239→    })
  1240→  }
  1241→
  1242→  return (
  1243→    <Dialog open={open} onOpenChange={onOpenChange}>
  1244→      <DialogContent className="sm:max-w-lg">
  1245→        <DialogHeader>
  1246→          <DialogTitle className="flex items-center gap-2">
  1247→            <span className="grid h-7 w-7 place-items-center rounded-md bg-red-600 text-white">
  1248→              <Activity className="h-4 w-4" />
  1249→            </span>
  1250→            Dismiss {outBatsmanName}
  1251→          </DialogTitle>
  1252→          <DialogDescription>
  1253→            Pick the dismissal type. If caught, stumped or run out, choose the
  1254→            fielder(s) involved. The dismissal text updates live in the preview.
  1255→          </DialogDescription>
  1256→        </DialogHeader>
  1257→
  1258→        {/* Free-hit banner — appears when the current ball is a free hit. */}
  1259→        {freeHit && (
  1260→          <div className="rounded-md border border-amber-400 bg-amber-50 px-3 py-2 text-xs dark:bg-amber-950/40">
  1261→            <div className="flex items-center gap-1.5 font-bold uppercase tracking-wide text-amber-800 dark:text-amber-300">
  1262→              <span className="h-1.5 w-1.5 animate-pulse rounded-full bg-amber-600" />
  1263→              Free hit in effect
  1264→            </div>
  1265→            <p className="mt-1 text-amber-700 dark:text-amber-400">
  1266→              Bowler-credited dismissals (bowled, caught, lbw, stumped, hit
  1267→              wicket) are <strong>not allowed</strong> on a free hit — the
  1268→              batsman survives and only run-out / retired-out stand.
  1269→              {freeHitWillReject
  1270→                ? " Picking this dismissal will be rejected by the engine — pick Run Out instead."
  1271→                : ""}
  1272→            </p>
  1273→          </div>
  1274→        )}
  1275→
  1276→        <div className="grid gap-3 py-2">
  1277→          {/* Dismissal type */}
  1278→          <div className="grid gap-1.5">
  1279→            <Label>Dismissal type</Label>
  1280→            <Select value={wicketType} onValueChange={(v) => setWicketType(v as WicketType)}>
  1281→              <SelectTrigger className="w-full">
  1282→                <SelectValue />
  1283→              </SelectTrigger>
  1284→              <SelectContent>
  1285→                {WICKET_TYPE_OPTIONS.map((opt) => (
  1286→                  <SelectItem key={opt.value} value={opt.value}>
  1287→                    <span className="font-medium">{opt.label}</span>
  1288→                    <span className="ml-2 text-[10px] text-muted-foreground">{opt.hint}</span>
  1289→                  </SelectItem>
  1290→                ))}
  1291→              </SelectContent>
  1292→            </Select>
  1293→          </div>
  1294→
  1295→          {/* Batsman out — only shown for run-out */}
  1296→          {showBatsmanSelect && (
  1297→            <div className="grid gap-1.5">
  1298→              <Label>Which batsman is out?</Label>
  1299→              <Select
  1300→                value={effectiveOutBatsmanId}
  1301→                onValueChange={setOutBatsmanId}
  1302→              >
  1303→                <SelectTrigger className="w-full">
  1304→                  <SelectValue />
  1305→                </SelectTrigger>
  1306→                <SelectContent>
  1307→                  <SelectItem value={striker.id}>
  1308→                    Striker — {striker.name}
  1309→                  </SelectItem>
  1310→                  <SelectItem value={nonStriker.id}>
  1311→                    Non-striker — {nonStriker.name}
  1312→                  </SelectItem>
  1313→                </SelectContent>
  1314→              </Select>
  1315→            </div>
  1316→          )}
  1317→
  1318→          {/* Fielder(s) involved — for caught, stumped, run out */}
  1319→          {needsFielder && (
  1320→            <div className="grid gap-1.5">
  1321→              <Label>
  1322→                {wicketType === "caught"
  1323→                  ? "Catcher"
  1324→                  : wicketType === "stumped"
  1325→                    ? "Wicket-keeper"
  1326→                    : "Run-out fielder"}
  1327→              </Label>
  1328→              <Select value={effectiveFielderId} onValueChange={setFielderId}>
  1329→                <SelectTrigger className="w-full">
  1330→                  <SelectValue placeholder="(direct hit / no fielder)" />
  1331→                </SelectTrigger>
  1332→                <SelectContent>
  1333→                  <SelectItem value="(direct-hit)">(direct hit / no fielder)</SelectItem>
  1334→                  {fieldingPlayers.map((p) => (
  1335→                    <SelectItem key={p.id} value={p.id}>
  1336→                      {p.name}
  1337→                      {p.isKeeper && " †"}
  1338→                      {p.isCaptain && " (c)"}
  1339→                    </SelectItem>
  1340→                  ))}
  1341→                </SelectContent>
  1342→              </Select>
  1343→              {wicketType === "stumped" && keeperId && !fielderId && (
  1344→                <p className="text-[11px] text-muted-foreground">
  1345→                  Wicket-keeper auto-prefilled (you can change it).
  1346→                </p>

  1348→            </div>
  1349→          )}

  1351→          {/* Live dismissal-text preview */}
  1352→          <div className="rounded-md border bg-muted/40 p-3 text-sm">

  1355→            </div>
  1356→            <div className="mt-1 font-mono text-sm">
  1357→              <span className="font-semibold">{outBatsmanName}</span>{" "}
  1358→              <span className="text-muted-foreground">{dismissal}</span>
  1359→            </div>
  1360→          </div>
  1361→        </div>
  1362→
  1363→        <DialogFooter>
  1364→          <Button variant="outline" onClick={() => onOpenChange(false)}>

  1366→          </Button>

  1368→            onClick={handleConfirm}

  1373→          </Button>
  1374→        </DialogFooter>

  1380→function Picker({
  1381→  title,
  1382→  players,
  1383→  placeholder,
  1384→  onPick,
  1385→  confirmLabel = "Confirm",
  1386→}: {
  1387→  title: string
  1388→  players: { id: string; name: string }[]
  1389→  placeholder: string
  1390→  onPick: (id: string) => void
  1391→  /** Label for the commit button. Use "Add batsman" / "Set bowler" etc.
  1392→   *  Defaults to "Confirm". */
  1393→  confirmLabel?: string
  1394→}) {
  1395→  const [val, setVal] = useState("")
  1396→  const selected = players.find((p) => p.id === val)
  1397→  // Decouple selection from commit: changing the dropdown only updates the
  1398→  // local `val` state. The commit (calling onPick) requires an explicit
  1399→  // button click — so a misclick in the dropdown doesn't write to the live
  1400→  // state. If the user picks the wrong name in the dropdown, they can just
  1401→  // pick a different one before pressing Confirm.
  1402→  return (
  1403→    <div className="rounded-lg border-2 border-amber-400 bg-amber-50 p-4 dark:bg-amber-950/40">
  1404→      <div className="mb-2 flex items-center gap-2 text-amber-800 dark:text-amber-300">
  1405→        <Plus className="h-4 w-4" />
  1406→        <h3 className="text-sm font-bold uppercase tracking-wide">{title}</h3>
  1407→      </div>
  1408→      {players.length === 0 ? (
  1409→        // Empty-state — the team has no bench players available (rare, but
  1410→        // happens if a team was created without enough players seeded, or
  1411→        // everyone has already batted). Tell the admin how to fix it instead
  1412→        // of opening a broken dropdown.
  1413→        <div className="rounded-md border border-dashed bg-amber-100/60 px-3 py-4 text-center text-xs text-amber-800 dark:bg-amber-950/60 dark:text-amber-300">
  1414→          No eligible players left in the squad.
  1415→          <br />
  1416→          Add more players to the team via{" "}
  1417→          <span className="font-semibold">Admin → Teams → bulk-add players</span>{" "}
  1418→          (the green Users icon on the team row), then come back here.
  1419→        </div>
  1420→      ) : (
  1421→        <>
  1422→          <Select value={val} onValueChange={setVal}>
  1423→            <SelectTrigger className="w-full">
  1424→              <SelectValue placeholder={placeholder} />
  1425→            </SelectTrigger>
  1426→            <SelectContent>
  1427→              {players.map((p) => (
  1428→                <SelectItem key={p.id} value={p.id}>
  1429→                  {p.name}
  1430→                </SelectItem>
  1431→              ))}
  1432→            </SelectContent>
  1433→          </Select>
  1434→          <div className="mt-2 flex items-center justify-between gap-2">
  1435→            <span className="text-xs text-amber-700 dark:text-amber-400">
  1436→              {selected ? (
  1437→                <>
  1438→                  Selected: <span className="font-semibold">{selected.name}</span>
  1439→                </>
  1440→              ) : (
  1441→                "Pick a player above, then confirm."
  1442→              )}
  1443→            </span>
  1444→            <Button
  1445→              size="sm"
  1446→              disabled={!val}
  1447→              onClick={() => val && onPick(val)}
  1448→              className="gap-1.5"
  1449→            >
  1450→              <Plus className="h-3.5 w-3.5" />
  1451→              {confirmLabel}
  1452→            </Button>
  1453→          </div>
  1454→        </>
  1455→      )}
  1456→    </div>
  1457→  )
  1458→}
