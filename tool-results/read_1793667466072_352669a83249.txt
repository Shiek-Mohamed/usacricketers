    43→/** Format an ISO date as "Mon DD" (e.g. "Mar 10"). */
    44→function formatShortDate(iso: string): string {
    45→  try {
    46→    const d = new Date(iso)
    47→    return d.toLocaleDateString("en-US", { month: "short", day: "numeric" })
    48→  } catch {
    49→    return ""
    50→  }
    51→}

   100→/** Last innings total for a team in a match, e.g. "217/4". */
   101→function inningsTotal(match: Match, teamId: string): string {
   102→  const sc = match.scorecard
   103→  if (!sc?.innings) return ""
   104→  const inn = sc.innings.filter((i) => i.teamId === teamId)
   105→  if (!inn.length) return ""
   106→  const last = inn[inn.length - 1]
   107→  return `${last.totalRuns}/${last.totalWickets}`
   108→}

   110→/** Compact horizontal live-match bar (cricket.com.au-style live strip).
   111→ *  Mobile-first: everything fits in one row on a 375px (iPhone SE) viewport.
   112→ *  - Badges + scores are `shrink-0` (never compressed).
   113→ *  - Team short-name spans are `min-w-0 truncate` so they clip instead of
   114→ *    pushing siblings past the card boundary.
   115→ *  - Gaps + font sizes scale up at the `sm:` breakpoint.
   116→ */
   117→function LiveMatchStrip({ match }: { match: Match }) {
   118→  const navigate = useAppStore((s) => s.navigate)
   119→  const t1 = match.team1
   120→  const t2 = match.team2
   121→  const t1Score = inningsTotal(match, match.team1Id)
   122→  const t2Score = inningsTotal(match, match.team2Id)
   123→  return (
   124→    <button
   125→      onClick={() => navigate("match-detail", match.id)}
   126→      className="group flex w-full items-center gap-1.5 overflow-hidden rounded-lg border bg-card px-2.5 py-2 text-left shadow-sm transition-shadow hover:shadow-md sm:gap-2 sm:px-4 sm:py-2.5"
   127→    >
   128→      {/* Live indicator */}
   129→      <span className="flex shrink-0 items-center gap-1">
   130→        <span className="relative flex h-2 w-2">
   131→          <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-red-500 opacity-75" />
   132→          <span className="relative inline-flex h-2 w-2 rounded-full bg-red-600" />
   133→        </span>
   134→        <span className="text-[10px] font-bold uppercase tracking-wide text-red-600">
   135→          Live
   136→        </span>
   137→      </span>
   138→      <FormatPill format={match.format} />
   139→      {/* Match metadata — hidden below `sm` to save horizontal real estate */}
   140→      <span className="hidden min-w-0 truncate text-xs text-muted-foreground sm:inline">
   141→        {match.matchNo}
   142→        {match.series?.shortName ? ` · ${match.series.shortName}` : ""}
   143→      </span>
   144→      {/* Teams + scores — pushed right; truncates names instead of overflowing */}
   145→      <div className="ml-auto flex min-w-0 items-center justify-end gap-1.5 sm:gap-3">
   146→        <div className="flex min-w-0 items-center gap-1 sm:gap-1.5">
   147→          {t1 && <TeamBadge team={t1} size="sm" />}
   148→          <span className="min-w-0 truncate text-xs font-bold sm:text-sm">
   149→            {t1?.shortName}
   150→          </span>
   151→          {t1Score && (
   152→            <span className="shrink-0 text-xs font-extrabold tabular-nums sm:text-sm">
   153→              {t1Score}
   154→            </span>
   155→          )}
   156→        </div>
   157→        <span className="shrink-0 text-[10px] font-medium text-muted-foreground sm:text-xs">
   158→          v
   159→        </span>
   160→        <div className="flex min-w-0 items-center gap-1 sm:gap-1.5">
   161→          {t2 && <TeamBadge team={t2} size="sm" />}
   162→          <span className="min-w-0 truncate text-xs font-bold sm:text-sm">
   163→            {t2?.shortName}
   164→          </span>
   165→          {t2Score && (
   166→            <span className="shrink-0 text-xs font-extrabold tabular-nums sm:text-sm">
   167→              {t2Score}
   168→            </span>

   296→

   299→
   300→  const completedMatches = useMemo(
   301→    () => matches.filter((m) => m.status === "completed").slice(0, 6),

   303→  )
   304→  const upcomingMatches = useMemo(
   305→    () => matches.filter((m) => m.status === "upcoming").slice(0, 6),
   306→    [matches]
   307→  )
   308→
   309→  const featuredItems = featuredArticles?.items ?? []
   310→  const latestItems = allArticles?.items ?? []
   311→
   312→  // Featured hero = most recent article that has a hero image; secondary = next 3 with images.
   313→  const withImages = useMemo(
   314→    () => latestItems.filter((a) => a.imageUrl),
   315→    [latestItems]
   316→  )
   317→  const heroStory = withImages[0] ?? featuredItems[0] ?? latestItems[0] ?? null
   318→  const secondaryStories = withImages.slice(1, 4)
   319→
   320→  // Exclude the hero + secondary stories from Latest News so they don't
   321→  // appear twice on the homepage.

   323→    () =>
   324→      new Set(
   325→        [heroStory?.id, ...secondaryStories.map((a) => a.id)].filter(
   326→          Boolean,
   327→        ) as string[],
   328→      ),
   329→    [heroStory, secondaryStories],
   330→  )

   332→    () => latestItems.filter((a) => !featuredIds.has(a.id)).slice(0, 8),

   338→  return (

   346→          <LiveMatchStrip match={liveMatch} />
   347→        ) : null}

   762→/** Resolve a display thumbnail for a video (custom → YouTube hqdefault → none). */
   763→function videoThumb(v: Video): string {
   764→  if (v.thumbnailUrl) return v.thumbnailUrl
   765→  if (v.source === "youtube" && v.youtubeId)
   766→    return `https://img.youtube.com/vi/${v.youtubeId}/hqdefault.jpg`
   767→  return ""
   768→}
