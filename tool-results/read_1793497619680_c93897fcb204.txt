    76→    76→async function jsonOrThrow<T>(res: Response): Promise<T> {
    77→    77→  if (!res.ok) {
    78→    78→    let msg = `Request failed: ${res.status}`
    79→    79→    try {
    80→    80→      const body = await res.json()
    81→    81→      if (body?.error) msg = body.error
    82→    82→    } catch {
    83→    83→      // ignore JSON parse errors
    84→    84→    }
    85→    85→    throw new Error(msg)
    86→    86→  }
    87→    87→  return (await res.json()) as T
    88→    88→}

   101→   101→function emptyMatchForm(): MatchForm {
   102→   102→  const now = new Date()
   103→   103→  // local datetime string YYYY-MM-DDTHH:MM
   104→   104→  const pad = (n: number) => String(n).padStart(2, "0")
   105→   105→  const local = `${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(
   106→   106→    now.getDate()
   107→   107→  )}T${pad(now.getHours())}:${pad(now.getMinutes())}`
   108→   108→  return {
   109→   109→    team1Id: "",
   110→   110→    team2Id: "",
   111→   111→    format: "T20",
   112→   112→    matchNo: "",
   113→   113→    status: "upcoming",
   114→   114→    venue: "",
   115→   115→    city: "",
   116→   116→    startDate: local,
   117→   117→  }
   118→   118→}

   120→   120→function toDatetimeLocal(iso: string): string {
   121→   121→  try {
   122→   122→    const d = new Date(iso)
   123→   123→    const pad = (n: number) => String(n).padStart(2, "0")
   124→   124→    return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(
   125→   125→      d.getDate()
   126→   126→    )}T${pad(d.getHours())}:${pad(d.getMinutes())}`
   127→   127→  } catch {
   128→   128→    return ""
   129→   129→  }
   130→   130→}
