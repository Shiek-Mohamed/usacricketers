    42→export function AdminSeriesView() {
    43→  const { data, loading, refetch } = useSeries()
    44→  const [editingId, setEditingId] = useState<string | null>(null)
    45→  const [creating, setCreating] = useState(false)
    46→  const [deleteId, setDeleteId] = useState<string | null>(null)
    47→  const navigate = useAppStore((s) => s.navigate)
    48→
    49→  function openCreate() { setCreating(true); setEditingId(null) }
    50→  function openEdit(id: string) { setEditingId(id); setCreating(false) }
    51→  function close() { setEditingId(null); setCreating(false) }
    52→
    53→  const editing = data?.find((s) => s.id === editingId)
    54→
    55→  if (creating || editing) {
    56→    return (
    57→      <TournamentEditor
    58→        tournament={editing ?? null}
    59→        onSave={async () => { await refetch(); close() }}
    60→        onRefresh={async () => { await refetch() }}
    61→        onCancel={close}
    62→        onScoreMatch={(matchId: string) => navigate("live-scorer", matchId)}
    63→      />
    64→    )
    65→  }
    66→
    67→  return (
    68→    <AdminShell title="Tournaments" count={data?.length} actionLabel="+ New Tournament" onAction={openCreate}>
    69→      <Card>
    70→        <CardContent className="p-0">
    71→          <div className={cn("max-h-[60vh] overflow-y-auto", SCROLLBAR)}>
    72→            <Table>
    73→              <TableHeader className="sticky top-0 bg-card z-10">
    74→                <TableRow>
    75→                  <TableHead>Tournament</TableHead>
    76→                  <TableHead>Format</TableHead>
    77→                  <TableHead className="text-center">Teams</TableHead>
    78→                  <TableHead>Status</TableHead>
    79→                  <TableHead className="text-right">Actions</TableHead>
    80→                </TableRow>
    81→              </TableHeader>
    82→              <TableBody>
    83→                {loading ? (
    84→                  Array.from({ length: 3 }).map((_, i) => (
    85→                    <TableRow key={i}><TableCell colSpan={5}><Skeleton className="h-6 w-full" /></TableCell></TableRow>
    86→                  ))
    87→                ) : (data ?? []).length === 0 ? (
    88→                  <TableRow><TableCell colSpan={5} className="text-center text-muted-foreground py-8">No tournaments yet. Click “New Tournament” to create one.</TableCell></TableRow>
    89→                ) : (
    90→                  (data ?? []).map((s) => (
    91→                    <TableRow key={s.id}>
    92→                      <TableCell className="font-medium">{s.name}</TableCell>
    93→                      <TableCell><Badge variant="secondary">{s.format}</Badge></TableCell>
    94→                      <TableCell className="text-center">{s.teams?.length ?? 0}</TableCell>
    95→                      <TableCell className="capitalize text-xs">{s.status}</TableCell>
    96→                      <TableCell className="text-right whitespace-nowrap">
    97→                        <Button size="icon" variant="ghost" onClick={() => openEdit(s.id)} aria-label="Edit"><Pencil className="h-3.5 w-3.5" /></Button>
    98→                        <Button size="icon" variant="ghost" onClick={() => setDeleteId(s.id)} className="text-destructive" aria-label="Delete"><Trash2 className="h-3.5 w-3.5" /></Button>
    99→                      </TableCell>
   100→                    </TableRow>
   101→                  ))
   102→                )}
   103→              </TableBody>
   104→            </Table>
   105→          </div>
   106→        </CardContent>
   107→      </Card>
   108→
   109→      <AlertDialog open={!!deleteId} onOpenChange={(o) => !o && setDeleteId(null)}>
   110→        <AlertDialogContent>
   111→          <AlertDialogHeader><AlertDialogTitle>Delete this tournament?</AlertDialogTitle><AlertDialogDescription>All matches in this tournament will have their series cleared (not deleted).</AlertDialogDescription></AlertDialogHeader>
   112→          <AlertDialogFooter><AlertDialogCancel>Cancel</AlertDialogCancel><AlertDialogAction className="bg-destructive text-white hover:bg-destructive/90" onClick={async () => { if (deleteId) { await api.series.remove(deleteId); toast.success("Tournament deleted"); refetch(); } setDeleteId(null); }}>Delete</AlertDialogAction></AlertDialogFooter>
   113→        </AlertDialogContent>
   114→      </AlertDialog>
   115→    </AdminShell>
   116→  )
   117→}

   120→
   121→function TournamentEditor({
   122→  tournament, onSave, onRefresh, onCancel, onScoreMatch,
   123→}: {
   124→  tournament: Series | null
   125→  onSave: () => void
   126→  onRefresh: () => void
   127→  onCancel: () => void
   128→  onScoreMatch: (matchId: string) => void
   129→}) {
   130→  const teamsApi = useTeams()
   131→  const matchesApi = useMatches()
   132→  const navigate = useAppStore((s) => s.navigate)
   133→
   134→  const [name, setName] = useState(tournament?.name ?? "")
   135→  const [shortName, setShortName] = useState(tournament?.shortName ?? "")
   136→  const [format, setFormat] = useState(tournament?.format ?? "T20I")
   137→  const [customOvers, setCustomOvers] = useState(() => {
   138→    // If editing a custom format like "T8", extract the number.
   139→    const match = tournament?.format?.match(/^T(\d+)$/i)
   140→    return match ? parseInt(match[1], 10) : 12
   141→  })
   142→  const [season, setSeason] = useState(tournament?.season ?? "2025")
   143→  const [status, setStatus] = useState(tournament?.status ?? "upcoming")
   144→  const [description, setDescription] = useState(tournament?.description ?? "")
   145→  const [startDate, setStartDate] = useState(tournament?.startDate?.slice(0, 10) ?? new Date().toISOString().slice(0, 10))
   146→  const [endDate, setEndDate] = useState(tournament?.endDate?.slice(0, 10) ?? new Date().toISOString().slice(0, 10))
   147→  // Classification & playing-rules metadata (CricClub "Create Series Info")
   148→  const [ballType, setBallType] = useState(tournament?.ballType ?? "Leather")
   149→  const [category, setCategory] = useState(tournament?.category ?? "Men")
   150→  const [level, setLevel] = useState(tournament?.level ?? "Club")
   151→  const [playersPerTeam, setPlayersPerTeam] = useState(tournament?.playersPerTeam ?? 11)
   152→  const [squadLimit, setSquadLimit] = useState(tournament?.squadLimit ?? 0)
   153→  const [logoUrl, setLogoUrl] = useState(tournament?.logoUrl ?? "")
   154→  // Points system + ranking criteria (CricClub Points Settings + How to Rank)
   155→  const [winPoints, setWinPoints] = useState(tournament?.winPoints ?? 2)
   156→  const [lossPoints, setLossPoints] = useState(tournament?.lossPoints ?? 0)
   157→  const [tiePoints, setTiePoints] = useState(tournament?.tiePoints ?? 1)
   158→  const [abandonedPoints, setAbandonedPoints] = useState(tournament?.abandonedPoints ?? 1)
   159→  const [bonusPoints, setBonusPoints] = useState(tournament?.bonusPoints ?? 0)
   160→  const [rankCriteria, setRankCriteria] = useState<string[]>((tournament?.rankCriteria ?? "points,nrr,wins").split(",").map((s) => s.trim()).filter(Boolean))
   161→  const [saving, setSaving] = useState(false)
   162→
   163→  // New-team quick-create form
   164→  const [newTeamName, setNewTeamName] = useState("")
   165→  const [newTeamShort, setNewTeamShort] = useState("")
   166→  const [newTeamColor, setNewTeamColor] = useState("#1e3a8a")
   167→  const [addingTeam, setAddingTeam] = useState(false)
   168→  // Optional: bulk players to add alongside the new team at creation time.
   169→  const [newTeamPlayersText, setNewTeamPlayersText] = useState("")
   170→
   171→  // Add-existing-team select
   172→  const [addExistingId, setAddExistingId] = useState("")
   173→
   174→  // Bulk-players dialog target (a team that's already in the tournament)
   175→  const [bulkFor, setBulkFor] = useState<Team | null>(null)
   176→
   177→  const tournamentTeams = tournament?.teams ?? []
   178→  const allTeams = teamsApi.data ?? []
   179→  const availableTeams = allTeams.filter((t) => !tournamentTeams.some((tt) => tt.id === t.id))
   180→  const tournamentMatches = (matchesApi.data ?? []).filter((m) => m.seriesId === tournament?.id)
   181→
   182→  async function handleSave() {
   183→    if (!name.trim()) { toast.error("Name is required"); return }
   184→    setSaving(true)
   185→    try {
   186→      // When "Custom" is selected, store as "T{N}" (e.g. "T12", "T8") so
   187→      // maxOversForFormat() can parse it.
   188→      const actualFormat = format === "Custom" ? `T${customOvers}` : format
   189→      const payload = { name, shortName: shortName || name.slice(0, 20), format: actualFormat, season, status, description, startDate, endDate, ballType, category, level, playersPerTeam: Number(playersPerTeam), squadLimit: Number(squadLimit), logoUrl: logoUrl || null, winPoints: Number(winPoints), lossPoints: Number(lossPoints), tiePoints: Number(tiePoints), abandonedPoints: Number(abandonedPoints), bonusPoints: Number(bonusPoints), rankCriteria: rankCriteria.join(",") }
   190→      if (tournament) { await api.series.update(tournament.id, payload); toast.success("Tournament saved") }
   191→      else { await api.series.create(payload); toast.success("Tournament created") }
   192→      onSave()
   193→    } catch (e) { toast.error("Failed to save", { description: String(e) }) }
   194→    finally { setSaving(false) }
   195→  }
   196→
   197→  async function addExistingTeam() {
   198→    if (!addExistingId || !tournament) return
   199→    setAddingTeam(true)
   200→    try {
   201→      await api.series.addTeam(tournament.id, { teamId: addExistingId })
   202→      toast.success("Team added to tournament")
   203→      setAddExistingId("")
   204→      onRefresh()
   205→    } catch (e) { toast.error("Failed to add team") }
   206→    finally { setAddingTeam(false) }
   207→  }
   208→
   209→  async function createNewTeam() {
   210→    if (!newTeamName.trim() || !tournament) return
   211→    setAddingTeam(true)
   212→    try {
   213→      // 1. Create the team and add it to the tournament in a single call.
   214→      const createdTeam = await api.series.addTeam(tournament.id, {
   215→        name: newTeamName,
   216→        shortName: (newTeamShort || newTeamName).toUpperCase().slice(0, 4),
   217→        primaryColor: newTeamColor,
   218→      })
   219→      // 2. If the admin pasted player rows, bulk-add them to the new team.
   220→      const rows = parsePlayerRows(newTeamPlayersText)
   221→      if (rows.length > 0) {
   222→        try {
   223→          const result = await api.players.bulkCreate(
   224→            createdTeam.id,
   225→            rows.map((r) => ({
   226→              name: r.name,
   227→              role: r.role,
   228→              isCaptain: r.isCaptain,
   229→              isKeeper: r.isKeeper,
   230→              battingStyle: r.battingStyle,
   231→              bowlingStyle: r.bowlingStyle,
   232→            }))
   233→          )
   234→          toast.success(
   235→            `Team "${newTeamName}" created + added — ${result.created} player${result.created === 1 ? "" : "s"} imported` +
   236→              (result.skipped > 0 ? `, ${result.skipped} skipped` : "")
   237→          )
   238→        } catch (e) {
   239→          toast.warning(
   240→            `Team "${newTeamName}" created, but player import failed`,
   241→            { description: String(e) }
   242→          )
   243→        }
   244→      } else {
   245→        toast.success(`Team "${newTeamName}" created + added`)
   246→      }
   247→      setNewTeamName("")
   248→      setNewTeamShort("")
   249→      setNewTeamColor("#1e3a8a")
   250→      setNewTeamPlayersText("")
   251→      onRefresh()
   252→    } catch (e) {
   253→      toast.error("Failed to create team")
   254→    } finally {
   255→      setAddingTeam(false)
   256→    }
   257→  }
   258→
   259→  async function removeTeam(teamId: string) {
   260→    if (!tournament) return
   261→    await api.series.removeTeam(tournament.id, teamId)
   262→    toast.success("Team removed from tournament")
   263→    onRefresh()
   264→  }
   265→
   266→  async function createMatchAndScore() {
   267→    if (!tournament || tournamentTeams.length < 2) { toast.error("Add at least 2 teams first"); return }
   268→    try {
   269→      const match = await api.matches.create({
   270→        seriesId: tournament.id,
   271→        team1Id: tournamentTeams[0].id,
   272→        team2Id: tournamentTeams[1].id,
   273→        format: tournament.format,
   274→        matchNo: `${tournamentMatches.length + 1}${ordinal(tournamentMatches.length + 1)} Match`,
   275→        status: "upcoming",
   276→        venue: "TBD",
   277→        startDate: new Date().toISOString(),
   278→      })
   279→      toast.success("Match created — opening Live Scorer")
   280→      onScoreMatch(match.id)
   281→    } catch (e) { toast.error("Failed to create match") }
   282→  }
   283→
   284→  return (
   285→    <AdminShell title={tournament ? "Edit Tournament" : "New Tournament"} actionLabel={undefined} onAction={undefined}>
   286→      <div className="space-y-5">
   287→        <Button variant="ghost" size="sm" onClick={onCancel} className="-ml-2 gap-1"><ArrowLeft className="h-4 w-4" /> Back to tournaments</Button>
   288→
   289→        <Tabs defaultValue="details" className="w-full">
   290→          <TabsList className="flex w-full justify-start gap-0 rounded-none border-b border-border bg-transparent p-0">
   291→            <TabsTrigger value="details" className="gap-2 rounded-none border-b-[3px] border-transparent bg-transparent px-4 py-3 text-sm font-bold text-muted-foreground transition-colors data-[state=active]:border-emerald-600 data-[state=active]:bg-emerald-600 data-[state=active]:text-white data-[state=active]:shadow-none whitespace-nowrap">
   292→              <Trophy className="h-4 w-4" /> Details
   293→            </TabsTrigger>
   294→            <TabsTrigger value="points" className="gap-2 rounded-none border-b-[3px] border-transparent bg-transparent px-4 py-3 text-sm font-bold text-muted-foreground transition-colors data-[state=active]:border-emerald-600 data-[state=active]:bg-emerald-600 data-[state=active]:text-white data-[state=active]:shadow-none whitespace-nowrap">
   295→              <BarChart3 className="h-4 w-4" /> Points & Ranking
   296→            </TabsTrigger>
   297→            <TabsTrigger value="teams" className="gap-2 rounded-none border-b-[3px] border-transparent bg-transparent px-4 py-3 text-sm font-bold text-muted-foreground transition-colors data-[state=active]:border-emerald-600 data-[state=active]:bg-emerald-600 data-[state=active]:text-white data-[state=active]:shadow-none whitespace-nowrap">
   298→              <Users className="h-4 w-4" /> Teams
   299→            </TabsTrigger>
   300→            {tournament && (
   301→              <TabsTrigger value="matches" className="gap-2 rounded-none border-b-[3px] border-transparent bg-transparent px-4 py-3 text-sm font-bold text-muted-foreground transition-colors data-[state=active]:border-emerald-600 data-[state=active]:bg-emerald-600 data-[state=active]:text-white data-[state=active]:shadow-none whitespace-nowrap">
   302→                <Radio className="h-4 w-4" /> Matches
   303→              </TabsTrigger>
   304→            )}
   305→            {tournament?.id && (
   306→              <TabsTrigger value="penalties" className="gap-2 rounded-none border-b-[3px] border-transparent bg-transparent px-4 py-3 text-sm font-bold text-muted-foreground transition-colors data-[state=active]:border-emerald-600 data-[state=active]:bg-emerald-600 data-[state=active]:text-white data-[state=active]:shadow-none whitespace-nowrap">
   307→                <Scale className="h-4 w-4" /> Penalties
   308→              </TabsTrigger>
   309→            )}
   310→          </TabsList>
   311→
   312→          {/* Details */}
   313→          <TabsContent value="details">
   314→            <Card><CardContent className="grid gap-4 p-4 sm:p-6">
   315→              <div className="grid gap-1.5"><Label htmlFor="t-name">Tournament name</Label><Input id="t-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. Sunday T20 League 2025" /></div>
   316→              <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   317→                <div className="grid gap-1.5"><Label>Short name</Label><Input value={shortName} onChange={(e) => setShortName(e.target.value)} placeholder="STL 2025" /></div>
   318→                <div className="grid gap-1.5">
   319→                  <Label>Format</Label>
   320→                  <Select
   321→                    value={format === "Custom" || /^T\d+$/i.test(format) ? "Custom" : format}
   322→                    onValueChange={(v) => setFormat(v)}
   323→                  >
   324→                    <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   325→                    <SelectContent>{FORMATS.map((f) => <SelectItem key={f} value={f}>{f}</SelectItem>)}</SelectContent>
   326→                  </Select>
   327→                  {format === "Custom" && (
   328→                    <div className="flex items-center gap-2 mt-1">
   329→                      <Input
   330→                        type="number"
   331→                        min="1"
   332→                        max="50"
   333→                        value={customOvers}
   334→                        onChange={(e) => setCustomOvers(Math.max(1, Math.min(50, parseInt(e.target.value) || 1)))}
   335→                        className="w-20"
   336→                      />
   337→                      <span className="text-xs text-muted-foreground">overs per side</span>
   338→                    </div>
   339→                  )}
   340→                </div>
   341→                <div className="grid gap-1.5"><Label>Season</Label><Input value={season} onChange={(e) => setSeason(e.target.value)} placeholder="2025" /></div>
   342→                <div className="grid gap-1.5"><Label>Status</Label><Select value={status} onValueChange={setStatus}><SelectTrigger className="w-full"><SelectValue /></SelectTrigger><SelectContent>{STATUSES.map((s) => <SelectItem key={s} value={s} className="capitalize">{s}</SelectItem>)}</SelectContent></Select></div>
   343→              </div>
   344→              <div className="grid grid-cols-2 gap-3">
   345→                <div className="grid gap-1.5"><Label>Start date</Label><Input type="date" value={startDate} onChange={(e) => setStartDate(e.target.value)} /></div>
   346→                <div className="grid gap-1.5"><Label>End date</Label><Input type="date" value={endDate} onChange={(e) => setEndDate(e.target.value)} /></div>
   347→              </div>
   348→              {/* Classification & playing rules (CricClub Create Series Info) */}
   349→              <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   350→                <div className="grid gap-1.5"><Label>Ball type</Label><Select value={ballType} onValueChange={setBallType}><SelectTrigger className="w-full"><SelectValue /></SelectTrigger><SelectContent>{["Leather","Hard Tennis","Soft Tennis","Tape Ball"].map((o) => <SelectItem key={o} value={o}>{o}</SelectItem>)}</SelectContent></Select></div>
   351→                <div className="grid gap-1.5"><Label>Category</Label><Select value={category} onValueChange={setCategory}><SelectTrigger className="w-full"><SelectValue /></SelectTrigger><SelectContent>{["Men","Women","Mixed","Boys","Girls"].map((o) => <SelectItem key={o} value={o}>{o}</SelectItem>)}</SelectContent></Select></div>
   352→                <div className="grid gap-1.5"><Label>Level</Label><Select value={level} onValueChange={setLevel}><SelectTrigger className="w-full"><SelectValue /></SelectTrigger><SelectContent>{["International","State","District","Club","School","Corporate"].map((o) => <SelectItem key={o} value={o}>{o}</SelectItem>)}</SelectContent></Select></div>
   353→                <div className="grid gap-1.5"><Label>Players per team</Label><Input type="number" min={1} max={20} value={playersPerTeam} onChange={(e) => setPlayersPerTeam(Number(e.target.value) || 11)} /></div>
   354→                <div className="grid gap-1.5"><Label>Squad limit (0 = unlimited)</Label><Input type="number" min={0} value={squadLimit} onChange={(e) => setSquadLimit(Number(e.target.value) || 0)} /></div>
   355→              </div>
   356→              <div className="grid gap-1.5">
   357→                <Label>Series logo</Label>
   358→                <div className="flex flex-wrap items-center gap-3">
   359→                  {logoUrl ? <img src={logoUrl} alt="Series logo" className="h-12 w-12 rounded-lg border object-cover" /> : null}
   360→                  <UploadButton label="Upload logo" withBrowse onUploaded={(url) => setLogoUrl(url)} />
   361→                  {logoUrl && <Button type="button" variant="ghost" size="sm" onClick={() => setLogoUrl("")}>Remove</Button>}
   362→                </div>
   363→              </div>
   364→              <div className="grid gap-1.5"><Label htmlFor="t-desc">Description</Label><Textarea id="t-desc" value={description} onChange={(e) => setDescription(e.target.value)} rows={2} /></div>
   365→            </CardContent></Card>
   366→          </TabsContent>
   367→
   368→          {/* Points & Ranking */}
   369→          <TabsContent value="points">
   370→            <Card><CardContent className="grid gap-4 p-4 sm:p-6">
   371→              <div className="flex items-center gap-2"><BarChart3 className="h-5 w-5 text-emerald-600" /><h2 className="text-base font-bold">Points System</h2></div>
   372→              <p className="text-xs text-muted-foreground">Points awarded per match outcome. Applied to the auto-computed standings table.</p>
   373→              <div className="grid grid-cols-2 gap-3 sm:grid-cols-5">
   374→                <div className="grid gap-1.5"><Label>Win</Label><Input type="number" min={0} value={winPoints} onChange={(e) => setWinPoints(Number(e.target.value) || 0)} /></div>
   375→                <div className="grid gap-1.5"><Label>Loss</Label><Input type="number" min={0} value={lossPoints} onChange={(e) => setLossPoints(Number(e.target.value) || 0)} /></div>
