   154→  /** Begin a voice clip recording. Requests microphone permission — the
   155→   *  browser shows a permission prompt the first time, then remembers the
   156→   *  decision for subsequent recordings on this origin. */
   157→  async function startRecording() {
   158→    if (!liveState || !id) return
   159→    if (typeof navigator === "undefined" || !navigator.mediaDevices?.getUserMedia) {
   160→      toast.error("Voice recording is not supported in this browser")
   161→      return
   162→    }
   163→    try {
   164→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   165→      streamRef.current = stream
   166→      chunksRef.current = []
   167→      // Pick the most likely mime type the browser supports — Chrome/Firefox
   168→      // emit audio/webm, Safari emits audio/mp4. We pass this through to the
   169→      // upload so the API saves the file with the correct extension.
   170→      const mimeType = MediaRecorder.isTypeSupported("audio/webm")
   171→        ? "audio/webm"
   172→        : MediaRecorder.isTypeSupported("audio/mp4")
   173→          ? "audio/mp4"
   174→          : ""
   175→      const recorder = mimeType
   176→        ? new MediaRecorder(stream, { mimeType })
   177→        : new MediaRecorder(stream)
   178→      recorder.ondataavailable = (e) => {
   179→        if (e.data && e.data.size > 0) chunksRef.current.push(e.data)
   180→      }
   181→      recorder.onstop = async () => {
   182→        // Tear down the mic stream first so the browser's red dot disappears.
   183→        if (streamRef.current) {
   184→          streamRef.current.getTracks().forEach((t) => t.stop())
   185→          streamRef.current = null
   186→        }
   187→        const chunks = chunksRef.current
   188→        chunksRef.current = []
   189→        if (chunks.length === 0) return
   190→        const blob = new Blob(chunks, {
   191→          type: recorder.mimeType || "audio/webm",
   192→        })
   193→        // Upload the blob.
   194→        setIsUploading(true)
   195→        try {
   196→          const formData = new FormData()
   197→          const ext = (recorder.mimeType || "").includes("mp4")
   198→            ? "mp4"
   199→            : (recorder.mimeType || "").includes("ogg")
   200→              ? "ogg"
   201→              : "webm"
   202→          formData.append("audio", blob, `voice-${Date.now()}.${ext}`)
   203→          const headers: Record<string, string> = {}
   204→          // Always send the bearer scorer token if we have one (real scorer
   205→          // session OR the god-mode token minted for site admins). The API
   206→          // also accepts site-admin cookie auth as a fallback.
   207→          if (token) headers["Authorization"] = `Bearer ${token}`
   208→          // CSRF double-submit for the cookie-auth path (site admin).
   209→          if (typeof document !== "undefined") {
   210→            const csrf = document.cookie.match(/csrf-token=([^;]+)/)?.[1] || ""
   211→            if (csrf) headers["X-CSRF-Token"] = csrf
   212→          }
   213→          const res = await fetch(
   214→            `/api/voice-clip?matchId=${encodeURIComponent(id)}`,
   215→            {
   216→              method: "POST",
   217→              headers,
   218→              credentials: "include",
   219→              body: formData,
   220→            }
   221→          )
   222→          if (!res.ok) {
   223→            const err = await res.json().catch(() => ({ error: "upload failed" }))
   224→            toast.error(`Voice clip upload failed: ${err.error || res.statusText}`)
   225→            return
   226→          }
   227→          const body = (await res.json()) as { url?: string; id?: string }
   228→          if (body?.url) {
   229→            // Insert as a commentary entry — the feed renders an audio
   230→            // player when the text starts with `[audio:`.
   231→            addCommentary(`[audio:${body.url}]`)
   232→            toast.success("Voice clip added to commentary")
   233→          }
   234→        } catch (e) {
   235→          toast.error(`Voice clip upload failed: ${String(e).slice(0, 100)}`)
   236→        } finally {
   237→          setIsUploading(false)
   238→        }
   239→      }
   240→      recorder.start()
   241→      mediaRecorderRef.current = recorder
   242→      setIsRecording(true)
   243→    } catch (e) {
   244→      // Most common: user denied the mic permission, or no mic attached.
   245→      toast.error(
   246→        `Microphone unavailable: ${String(e).slice(0, 120)}. Allow microphone access in your browser and try again.`
   247→      )
   248→      setIsRecording(false)
   249→    }
   250→  }

   269→  /* ------------------ Keyboard shortcuts (power-user scorers) ------------------
   270→   * 0,1,2,3,4,6 → score N runs | . / - → dot | w → wicket dialog |
   271→   * e → swap ends | u → undo | d → declare (Test) | s → save.
   272→   * No deps array → re-subscribes on every render with the latest closures.
   273→   * The function declarations (handleBall, swapEnds, etc.) are hoisted by
   274→   * JS so they're in scope at call-time even though defined later in the
   275→   * component body. The lint rule below is overly cautious about this
   276→   * pattern — it's safe because the effect re-runs every render. */
   277→  /* eslint-disable react-hooks/immutability */
   278→  useEffect(() => {
   279→    function isTyping(el: Element | null): boolean {
   280→      if (!el) return false
   281→      const tag = el.tagName?.toLowerCase()
   282→      return tag === "input" || tag === "textarea" || tag === "select" ||
   283→        el.getAttribute("role") === "combobox" ||
   284→        (el as HTMLElement)?.isContentEditable === true
   285→    }
   286→    function onKey(e: KeyboardEvent) {
   287→      if (e.metaKey || e.ctrlKey || e.altKey) return
   288→      if (isTyping(document.activeElement)) return
   289→      if (wicketOpen) return
   290→      if (!liveState || liveState.status !== "live") return
   291→      if (!liveState.strikerId || !liveState.nonStrikerId || !liveState.bowlerId) return
   292→      const k = e.key.toLowerCase()
   293→      if (["0", "1", "2", "3", "4", "6"].includes(k)) {
   294→        e.preventDefault()
   295→        handleBall({ kind: "run", runs: Number(k) })
   296→      } else if (k === "." || k === "-") {
   297→        e.preventDefault()
   298→        handleBall({ kind: "run", runs: 0 })
   299→      } else if (k === "w") {
   300→        e.preventDefault()
   301→        setWicketOpen(true)
   302→      } else if (k === "e") {
   303→        e.preventDefault()
   304→        swapEnds()
   305→      } else if (k === "u") {
   306→        e.preventDefault()
   307→        handleUndo()
   308→      } else if (k === "d") {
   309→        e.preventDefault()
   310→        handleDeclare()
   311→      } else if (k === "s") {
   312→        e.preventDefault()
   313→        if (liveState) {
   314→          persist(liveState)
   315→          toast.success("Scorecard saved")
   316→        }
   317→      }
   318→    }
   319→    window.addEventListener("keydown", onKey)
   320→    return () => window.removeEventListener("keydown", onKey)
   321→  })

   363→  function persist(state: LiveMatchState) {
   364→    if (saveTimer.current) clearTimeout(saveTimer.current)
   365→    saveTimer.current = setTimeout(async () => {
   366→      try {
   367→        await api.matches.update(match!.id, {
   368→          scorecard: stateToScorecard(state),
   369→          status: state.status,
   370→          result: state.result,
   371→        })
   372→        refetch()
   373→      } catch (e) {
   374→        // Surface persistence errors so the operator knows the DB write failed
   375→        // (was previously silent — caused confusion when "Innings started"
   376→        // toast showed but the match status didn't actually update).
   377→        // The live socket broadcast is still the priority — this toast
   378→        // doesn't block scoring, just informs the operator something's wrong.
   379→        const msg = e instanceof Error ? e.message : String(e).slice(0, 200)
   380→        toast.error("Persist failed — match state not saved", {
   381→          description: msg,
   382→        })
   383→      }
   384→    }, 800)
   385→  }

   387→  /** Add a manual / commentator note to the live commentary feed.
   388→   *  Pushes the updated state through the socket (so subscribers like the
   389→   *  public match-detail view see it instantly) and schedules a DB persist.
   390→   *  Manual entries get kind="manual" + a blue accent in the feed so they're
   391→   *  visually distinct from the auto-generated ball-by-ball commentary. */
   392→  function addCommentary(text: string) {
   393→    const trimmed = text.trim()
   394→    if (!trimmed || !liveState) return
   395→    const inn = liveState.innings[liveState.currentInnings]
   396→    const overLabel = inn ? oversString(inn.balls) : "0.0"
   397→    const entry = {
   398→      id: Math.random().toString(36).slice(2),
   399→      ballNo: liveState.ballNo,
   400→      over: overLabel,
   401→      text: trimmed,
   402→      kind: "manual" as const,
   403→    }
   404→    const next: LiveMatchState = {
   405→      ...liveState,
   406→      // Newest-first (matches the engine's pushCommentary pattern).
   407→      commentary: [entry, ...liveState.commentary].slice(0, 120),
   408→    }
   409→    setHistory((h) => [...h, liveState].slice(-50))
   410→    pushState(next)
   411→    persist(next)
   412→    setCommentaryText("")
   413→  }

   423→  function handleUndo() {
   424→    if (history.length === 0) {
   425→      toast.info("Nothing to undo")
   426→      return
   427→    }
   428→    const prev = history[history.length - 1]
   429→    setHistory((h) => h.slice(0, -1))
   430→    pushState(prev)
   431→    persist(prev)
   432→    toast.success("Undid last action")
   433→  }

   435→  function startInnings(
   436→    battingFirstId: string,
   437→    strikerId: string,
   438→    nonStrikerId: string,
   439→    bowlerId: string
   440→  ) {
   441→    const battingPlayers = playersByTeam[battingFirstId] ?? []
   442→    const bowlingPlayers =
   443→      playersByTeam[battingFirstId === match!.team1Id ? match!.team2Id : match!.team1Id] ?? []
   444→    const striker = battingPlayers.find((p) => p.id === strikerId)
   445→    const nonStriker = battingPlayers.find((p) => p.id === nonStrikerId)
   446→    const bowler = bowlingPlayers.find((p) => p.id === bowlerId)
   447→    if (!striker || !nonStriker || !bowler) {
   448→      toast.error("Select striker, non-striker and bowler")
   449→      return
   450→    }
   451→    let state = initState(match!, battingFirstId)
   452→    // Pre-populate the squad with the full batting-team roster so the
   453→    // "Did not bat" section of the public scorecard lists bench players.
   454→    // Captain/keeper tags come from the explicit Player model fields
   455→    // (was previously a brittle regex on the role string).
   456→    for (const p of battingPlayers) {
   457→      state = addSquadMember(state, p.id, p.name, {
   458→        isCaptain: !!p.isCaptain,
   459→        isKeeper: !!p.isKeeper,
   460→      })
   461→    }
   462→    state = addBatsman(state, striker.id, striker.name, {
   463→      isCaptain: !!striker.isCaptain,
   464→      isKeeper: !!striker.isKeeper,
   465→    })
   466→    state = addBatsman(state, nonStriker.id, nonStriker.name, {
   467→      isCaptain: !!nonStriker.isCaptain,
   468→      isKeeper: !!nonStriker.isKeeper,
   469→    })
   470→    state = setBatsmen(state, striker.id, nonStriker.id)
   471→    state = setBowler(state, bowler.id, bowler.name)
   472→    setHistory([])
   473→    pushState(state)
   474→    persist(state)
   475→    toast.success("Innings started — status set to LIVE")
   476→  }

   478→  function bringInBatsman(playerId: string) {
   479→    if (!liveState) return
   480→    const battingPlayers = playersByTeam[liveState.innings[liveState.currentInnings].battingTeamId] ?? []
   481→    const p = battingPlayers.find((x) => x.id === playerId)
   482→    if (!p) return
   483→    let state = addBatsman(liveState, p.id, p.name, {
   484→      isCaptain: !!p.isCaptain,
   485→      isKeeper: !!p.isKeeper,
   486→    })
   487→    // Fill whichever slot is currently empty — the striker's end if a normal
   488→    // wicket fell (strikerId === null), or the non-striker's end if a run-out
   489→    // happened at the non-striker's end (nonStrikerId === null). If both are
   490→    // empty (innings restart, edge case), default to striker.
   491→    const fillStriker = liveState.strikerId === null
   492→    const fillNonStriker = !fillStriker && liveState.nonStrikerId === null
   493→    state = {
   494→      ...state,
   495→      strikerId: fillStriker ? p.id : state.strikerId,
   496→      nonStrikerId: fillNonStriker ? p.id : state.nonStrikerId,
   497→    }
   498→    setHistory((h) => [...h, liveState].slice(-50))
   499→    pushState(state)
   500→    persist(state)
   501→    toast.success(`${p.name} in`)
   502→  }

   504→  function pickNextBowler(playerId: string) {

   516→  /** "I picked the wrong striker — let me re-pick." Pushes the current state

   529→  /** Same as changeStriker but for the non-striker slot. */
   530→  function changeNonStriker() {
   531→    if (!liveState || !liveState.nonStrikerId) return
   532→    setHistory((h) => [...h, liveState].slice(-50))
   533→    const next = { ...liveState, nonStrikerId: null as string | null }
   534→    pushState(next)
   535→    persist(next)
   536→    toast.info("Re-pick the non-striker")
   537→  }
   538→
   539→  /** Same as changeStriker but for the bowler slot. */
   540→  function changeBowler() {
   541→    if (!liveState || !liveState.bowlerId) return
   542→    setHistory((h) => [...h, liveState].slice(-50))
   543→    const next = { ...liveState, bowlerId: null as string | null }
   544→    pushState(next)
   545→    persist(next)
   546→    toast.info("Re-pick the bowler")
   547→  }

   549→  /** Swap the striker and non-striker ends. Used when the auto strike-rotation
   550→   *  logic disagrees with reality (e.g. batsmen crossed on a caught-and-
   551→   *  bowled, odd run on a wide, runner substituted, etc.). Allowed at any
   552→   *  time — doesn't affect stats, just relabels who's on strike. */
   553→  function swapEnds() {
   554→    if (!liveState) return
   555→    if (!liveState.strikerId || !liveState.nonStrikerId) {
   556→      toast.error("Need both batsmen at the crease to swap ends")
   557→      return
   558→    }
   559→    setHistory((h) => [...h, liveState].slice(-50))
   560→    const next = {
   561→      ...liveState,
   562→      strikerId: liveState.nonStrikerId,
   563→      nonStrikerId: liveState.strikerId,
   564→    }
   565→    pushState(next)
   566→    persist(next)
   567→    toast.success("Ends swapped")
   568→  }

   570→  function reset() {
   571→    setHistory([])
   572→    // clear live cache
   573→    // (clearState not destructured here; use pushState(null) approach via a clear)
   574→    pushState({ ...liveState!, status: "live", result: "" })
   575→  }
   576→
   577→  /** Test-match declaration: captain ends the current innings without being
   578→   *  all out. No-op for limited-overs matches (those don't have declarations).
   579→   *  Pushes the prev state to history (so Undo can recover) then runs the
   580→   *  engine's declareInnings which marks `declared: true` + sets up the
   581→   *  next innings / computes the result. */
   582→  function handleDeclare() {
   583→    if (!liveState) return
   584→    if (liveState.maxOvers > 0) {
   585→      toast.error("Declarations are only available in Test matches")
   586→      return
   587→    }
   588→    if (liveState.status !== "live") {
   589→      toast.error("Can't declare — innings already ended")
   590→      return
   591→    }
   592→    setHistory((h) => [...h, liveState].slice(-50))
   593→    const next = declareInnings(liveState)
   594→    pushState(next)
   595→    persist(next)
   596→    toast.success("Innings declared")
   597→  }
   598→
   599→  /** Enforce the follow-on in a Test match. Only valid when Team A leads by
   600→   *  200+ runs after Team B's 1st innings. Team B bats again immediately. */
   601→  function handleFollowOn() {
   602→    if (!liveState) return
   603→    setHistory((h) => [...h, liveState].slice(-50))
   604→    const next = enforceFollowOn(liveState)
   605→    if (next === liveState) {
   606→      toast.error("Follow-on requires a 200+ run lead in a Test after innings 1")
   607→      return
   608→    }
   609→    pushState(next)
   610→    persist(next)
   611→    toast.success("Follow-on enforced")
   612→  }
   613→
   614→  /** Start a super-over after a tied limited-overs match. Creates 2 new
   615→   *  1-over innings (one per team). */
   616→  function handleSuperOver() {
   617→    if (!liveState) return
   618→    setHistory((h) => [...h, liveState].slice(-50))
   619→    const next = startSuperOver(liveState)
   620→    if (next === liveState) {
   621→      toast.error("Super-over only available for tied limited-overs matches")
   622→      return
   623→    }
   624→    pushState(next)
   625→    persist(next)
   626→    toast.success("Super Over started")
   627→  }
   628→
   629→  const showSetup = !liveState || liveState.innings.length === 0
   630→  const currentInnings = liveState?.innings[liveState.currentInnings]
   631→  // Need a new batsman if either end is empty after a wicket (a run-out can
   632→  // empty the non-striker slot while the striker stays in).
   633→  const needStriker = !!liveState && liveState.strikerId === null && liveState.status === "live"
   634→  const needNonStriker =
   635→    !!liveState &&
   636→    liveState.nonStrikerId === null &&
   637→    liveState.status === "live" &&
   638→    !!currentInnings &&
   639→    currentInnings.batsmen.length > 0 // not the very first pickup of an innings
   640→  const needBatsman = needStriker || needNonStriker
   641→  const needBowler = !!liveState && liveState.bowlerId === null && liveState.status === "live"
   642→  const isComplete = liveState?.status === "completed"
   643→
   644→  return (

   646→      title="Live Scorer"
   647→    >
   648→      {/* Match header */}
   649→      <Card className="overflow-hidden p-0">

   651→          <div className="flex flex-wrap items-center gap-2 min-w-0">
   652→            <FormatBadge format={match.format} />
   653→            <span className="truncate text-sm font-semibold">{match.matchNo}</span>
   654→            <span className="text-xs text-muted-foreground">
   655→              {match.series?.shortName}
   656→            </span>

   664→                  isComplete ? "bg-emerald-600" : "bg-red-600 animate-pulse"

   666→              >
   667→                {isComplete ? "Completed" : "Live"}

   669→            )}
   670→          </div>
   671→        </div>
   672→        <CardContent className="p-4 sm:p-6">
   673→          <div className="flex flex-wrap items-center justify-between gap-4">
   674→            <div className="flex items-center gap-2">
   675→              <Button variant="ghost" size="sm" onClick={() => navigate("admin-matches")} className="-ml-2 gap-1">
   676→                <ArrowLeft className="h-4 w-4" /> Matches
   677→              </Button>
   678→              <span className="text-xs text-muted-foreground">
   679→                {t1?.shortName} v {t2?.shortName} · {match.venue}
   680→              </span>
   681→            </div>
   682→            {liveState && (
   683→              <div className="flex items-center gap-2">
   684→                <Button variant="outline" size="sm" onClick={handleUndo} className="gap-1" disabled={history.length === 0}>
   685→                  <Undo2 className="h-3.5 w-3.5" /> Undo
   686→                </Button>
   687→                {/* Test-only: captain can declare the current innings closed
   688→                    without being all out. Hidden for limited-overs matches. */}

   690→                  <Button
   691→                    variant="outline"
   692→                    size="sm"
   693→                    onClick={handleDeclare}
   694→                    className="gap-1 border-amber-400 text-amber-700 hover:bg-amber-50 dark:text-amber-300 dark:hover:bg-amber-950/40"

   700→                {/* Follow-on: Test only, after innings 1 (innings.length === 2),
   701→                    when Team A leads by 200+. */}
   702→                {liveState.maxOvers === 0 &&
   703→                  liveState.innings.length === 2 &&

   713→                      <Repeat className="h-3.5 w-3.5" /> Follow-on

   716→                {/* Super-over: limited-overs, match tied + completed. */}
   717→                {liveState.maxOvers > 0 &&
   718→                  liveState.status === "completed" &&
   719→                  liveState.result === "Match tied" && (
   720→                    <Button
   721→                      variant="outline"
   722→                      size="sm"
   723→                      onClick={handleSuperOver}
   724→                      className="gap-1 border-purple-500 text-purple-700 hover:bg-purple-50 dark:text-purple-300 dark:hover:bg-purple-950/40"

   727→                      <Zap className="h-3.5 w-3.5" /> Super Over

   729→                  )}
   730→                <Button
   731→                  variant="outline"
   732→                  size="sm"
   733→                  onClick={() => {
   734→                    persist(liveState)
   735→                    toast.success("Scorecard saved")
   736→                  }}
   737→                  className="gap-1"
   738→                >
   739→                  <Save className="h-3.5 w-3.5" /> Save

   741→              </div>
   742→            )}
   743→          </div>
   744→
   745→          {showSetup ? (
   746→            <SetupPanel
   747→              t1={t1}
   748→              t2={t2}
   749→              t1Players={t1Players ?? []}
   750→              t2Players={t2Players ?? []}
   751→              onStart={startInnings}
   752→              tokenReady={!!token}
   753→              connected={connected}
   754→            />
   755→          ) : (
   756→            <div className="mt-4 space-y-5">
   757→              {/* Live scoreboard summary */}
   758→              {currentInnings && (
   759→                <LiveScoreboard

   761→                  onChangeStriker={changeStriker}
   762→                  onChangeNonStriker={changeNonStriker}

   771→                  title={needStriker ? "New batsman (striker's end)" : "New batsman (non-striker's end)"}
   772→                  players={(playersByTeam[currentInnings!.battingTeamId] ?? []).filter(
   773→                    (p) => !currentInnings!.batsmen.some((b) => b.id === p.id)
   774→                  )}
   775→                  placeholder="Select new batsman"
   776→                  confirmLabel="Add batsman"
   777→                  onPick={bringInBatsman}
   778→                />
   779→              )}
   780→              {needBowler && (
   781→                <Picker
   782→                  title="Next bowler"
   783→                  players={playersByTeam[currentInnings!.bowlingTeamId] ?? []}
   784→                  placeholder="Select bowler for next over"

   788→              )}
   789→

   793→              )}
   794→
   795→              {/* Human / manual commentary input — only shown when there's an
   796→                  active live innings (the setup phase has no live commentary
   797→                  feed to attach the note to). Enter or the Send button
   798→                  submits; Esc or empty input is a no-op. */}
   799→              {!isComplete && !showSetup && liveState && (
   800→                <CommentaryInput
   801→                  value={commentaryText}
   802→                  onChange={setCommentaryText}
   803→                  onSubmit={() => addCommentary(commentaryText)}
   804→                  disabled={!liveState}
   805→                />
   806→              )}
   807→
   808→              {/* Voice clip (🎙) recorder — captures mic audio via the
   809→                  MediaRecorder API, uploads it to /api/voice-clip, and
   810→                  inserts a `[audio:URL]` entry into the commentary feed.
   811→                  The public match-detail Live tab renders an inline audio
   812→                  player for those entries. */}
   813→              {!isComplete && !showSetup && liveState && (
   814→                <div className="rounded-xl border-2 border-emerald-200 bg-emerald-50/50 p-3 dark:border-emerald-900 dark:bg-emerald-950/30">
   815→                  <div className="mb-2 flex items-center gap-2">
   816→                    <div className="grid h-7 w-7 place-items-center rounded-lg bg-emerald-600 text-white">
   817→                      <Mic className="h-4 w-4" />
   818→                    </div>
   819→                    <span className="text-sm font-extrabold uppercase tracking-wide text-emerald-700 dark:text-emerald-300">
   820→                      Record Voice Clip
   821→                    </span>
   822→                  </div>
   823→                  <div className="flex items-center gap-2">
   824→                    <Button
   825→                      size="sm"
   826→                      variant={isRecording ? "destructive" : "default"}

   829→                      className="gap-2 bg-emerald-600 text-sm font-bold text-white hover:bg-emerald-700"
   830→                    >
   831→                      {isRecording ? (
   832→                        <>
   833→                          <span className="h-2.5 w-2.5 animate-pulse rounded-full bg-white" />
   834→                          Stop Recording
   835→                        </>
   836→                      ) : isUploading ? (
   837→                        <>
   838→                          <span className="h-2.5 w-2.5 animate-pulse rounded-full bg-white" />
   839→                          Uploading Clip…
   840→                        </>
   841→                      ) : (
   842→                        <>
   843→                          <Mic className="h-4 w-4" />
   844→                          Start Recording
   845→                        </>
   846→                      )}
   847→                    </Button>
   848→                    {isRecording && (
   849→                      <span className="flex items-center gap-1 text-sm font-bold text-red-600">
   850→                        <span className="h-2 w-2 animate-pulse rounded-full bg-red-500" />
   851→                        Recording live…
   852→                      </span>
   853→                    )}
   854→                  </div>
   855→                  <p className="mt-1.5 text-[11px] font-medium text-emerald-600/70 dark:text-emerald-400/70">
   856→                    Click to record your voice → broadcasts to all viewers as a
   857→                    playable audio clip. Browser will ask for mic permission
   858→                    the first time.
   859→                    play inline on the public match detail Live tab.
   860→                  </p>
   861→                </div>
   862→              )}
   863→
   864→              {/* Wicket dialog — dismissal type + fielder picker */}
   865→              {wicketOpen && liveState && currentInnings && (
   866→                <WicketDialog
   867→                  open={wicketOpen}
   868→                  onOpenChange={setWicketOpen}
   869→                  striker={{
   870→                    id: liveState.strikerId!,
   871→                    name: currentInnings.batsmen.find((b) => b.id === liveState.strikerId)?.name ?? "Striker",
   872→                  }}
   873→                  nonStriker={{
   874→                    id: liveState.nonStrikerId!,
   875→                    name: currentInnings.batsmen.find((b) => b.id === liveState.nonStrikerId)?.name ?? "Non-striker",
   876→                  }}
   877→                  bowler={{
   878→                    id: liveState.bowlerId!,
   879→                    name: currentInnings.bowlers.find((b) => b.id === liveState.bowlerId)?.name ?? "Bowler",
   880→                  }}
   881→                  fieldingPlayers={playersByTeam[currentInnings.bowlingTeamId] ?? []}
   882→                  freeHit={currentInnings.freeHit}
   883→                  onConfirm={(ev) => {
   884→                    handleBall(ev)
   885→                    setWicketOpen(false)
   886→                  }}
   887→                />
   888→              )}
   889→
   890→              {isComplete && (
   891→                <div className="rounded-lg border border-emerald-300 bg-emerald-50 px-4 py-4 text-center">
   892→                  <p className="text-sm font-bold text-emerald-800">
   893→                    Match Result
   894→                  </p>
   895→                  <p className="mt-1 text-lg font-extrabold text-emerald-900">
   896→                    {liveState!.result}
   897→                  </p>
   898→                  <Button
   899→                    variant="outline"
   900→                    size="sm"
   901→                    className="mt-3 gap-1"
   902→                    onClick={() => navigate("match-detail", match.id)}
   903→                  >
   904→                    View public scorecard
   905→                  </Button>
   906→                </div>
   907→              )}
   908→
   909→              {/* Commentary */}
   910→              {liveState && liveState.commentary.length > 0 && (
   911→                <CommentaryFeed state={liveState} />
   912→              )}
   913→

   915→              {liveState && (
   916→                <div>
   917→                  <div className="mb-3 flex items-center gap-2">
   918→                    <span className="h-5 w-1 rounded-full bg-red-600" />
   919→                    <h2 className="text-sm font-extrabold uppercase tracking-tight">
   920→                      Live Scorecard (public view)
   921→                    </h2>
   922→                  </div>
   923→                  <ScorecardView scorecard={stateToScorecard(liveState)} />
   924→                </div>
   925→              )}
   926→            </div>
   927→          )}
   928→        </CardContent>
   929→      </Card>
   930→    </Shell>
   931→  )
   932→}

  1376→/** Build the dismissal summary string from the structured inputs (this is
  1377→ *  what the public scorecard renders next to the batsman's name). */
  1378→function buildDismissalText(args: {
  1379→  wicketType: WicketType
  1380→  bowlerName?: string
  1381→  fielderName?: string
  1382→}): string {
  1383→  const bowler = args.bowlerName ?? "?"
  1384→  const fielder = args.fielderName
  1385→  switch (args.wicketType) {
  1386→    case "bowled":
  1387→      return `b ${bowler}`
  1388→    case "caught":
  1389→      // Caught & bowled when no fielder is selected OR the fielder is the bowler.
  1390→      if (!fielder || fielder === bowler) return `c &b ${bowler}`
  1391→      return `c ${fielder} b ${bowler}`
  1392→    case "lbw":
  1393→      return `lbw b ${bowler}`
  1394→    case "stumped":
  1395→      return fielder ? `st ${fielder} b ${bowler}` : `st ${bowler}`
  1396→    case "runout":
  1397→      return fielder ? `run out (${fielder})` : `run out`
  1398→    case "hitwkt":
  1399→      return `hit wkt b ${bowler}`
  1400→    case "retiredHurt":
  1401→      return `retired hurt`
  1402→    case "retiredOut":
  1403→      return `retired out`
  1404→  }

  1641→function Picker({
  1642→  title,
  1643→  players,
  1644→  placeholder,
  1645→  onPick,
  1646→  confirmLabel = "Confirm",
  1647→}: {
  1648→  title: string
  1649→  players: { id: string; name: string }[]
  1650→  placeholder: string
  1651→  onPick: (id: string) => void
  1652→  /** Label for the commit button. Use "Add batsman" / "Set bowler" etc.
  1653→   *  Defaults to "Confirm". */
  1654→  confirmLabel?: string
  1655→}) {
  1656→  const [val, setVal] = useState("")
  1657→  const selected = players.find((p) => p.id === val)
  1658→  // Decouple selection from commit: changing the dropdown only updates the
  1659→  // local `val` state. The commit (calling onPick) requires an explicit
  1660→  // button click — so a misclick in the dropdown doesn't write to the live
  1661→  // state. If the user picks the wrong name in the dropdown, they can just
  1662→  // pick a different one before pressing Confirm.
  1663→  return (
  1664→    <div className="rounded-lg border-2 border-amber-400 bg-amber-50 p-4 dark:bg-amber-950/40">
  1665→      <div className="mb-2 flex items-center gap-2 text-amber-800 dark:text-amber-300">
  1666→        <Plus className="h-4 w-4" />
  1667→        <h3 className="text-sm font-bold uppercase tracking-wide">{title}</h3>
  1668→      </div>
  1669→      {players.length === 0 ? (
  1670→        // Empty-state — the team has no bench players available (rare, but
  1671→        // happens if a team was created without enough players seeded, or
  1672→        // everyone has already batted). Tell the admin how to fix it instead
  1673→        // of opening a broken dropdown.
  1674→        <div className="rounded-md border border-dashed bg-amber-100/60 px-3 py-4 text-center text-xs text-amber-800 dark:bg-amber-950/60 dark:text-amber-300">
  1675→          No eligible players left in the squad.
  1676→          <br />
  1677→          Add more players to the team via{" "}
  1678→          <span className="font-semibold">Admin → Teams → bulk-add players</span>{" "}
  1679→          (the green Users icon on the team row), then come back here.
  1680→        </div>
  1681→      ) : (
  1682→        <>
  1683→          <Select value={val} onValueChange={setVal}>
  1684→            <SelectTrigger className="w-full">
  1685→              <SelectValue placeholder={placeholder} />
  1686→            </SelectTrigger>
  1687→            <SelectContent>
  1688→              {players.map((p) => (
  1689→                <SelectItem key={p.id} value={p.id}>
  1690→                  {p.name}
  1691→                </SelectItem>
  1692→              ))}
  1693→            </SelectContent>
  1694→          </Select>
  1695→          <div className="mt-2 flex items-center justify-between gap-2">
  1696→            <span className="text-xs text-amber-700 dark:text-amber-400">
  1697→              {selected ? (
  1698→                <>
  1699→                  Selected: <span className="font-semibold">{selected.name}</span>
  1700→                </>
  1701→              ) : (
  1702→                "Pick a player above, then confirm."
  1703→              )}
  1704→            </span>
  1705→            <Button
  1706→              size="sm"
  1707→              disabled={!val}
  1708→              onClick={() => val && onPick(val)}
  1709→              className="gap-1.5"
  1710→            >
  1711→              <Plus className="h-3.5 w-3.5" />
  1712→              {confirmLabel}
  1713→            </Button>
  1714→          </div>
  1715→        </>
  1716→      )}
  1717→    </div>
  1718→  )
