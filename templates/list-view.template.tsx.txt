import { useVideos } from "@/lib/use-data"

import { InnerPageLayout } from "@/components/site/inner-layout"
import { NewsletterWidget } from "@/components/site/sidebar-widgets"

export function FeaturedVideosView() {
  // The hook params do the filtering — no new API route needed
  const { data, loading } = useVideos({ featured: true })

  return (
    <InnerPageLayout
      header={
        <div>
          <h1 className="flex items-center gap-2 text-2xl font-extrabold tracking-tight">
            <Film className="h-6 w-6 text-red-600" aria-hidden="true" />
            Featured Videos
          </h1>
          <p className="mt-1 text-sm text-muted-foreground">
            Hand-picked highlights and interviews from USA cricket.
          </p>
        </div>
      }
      sidebar={
        <>
          {/* For cricket-content pages, use RecentResultsWidget instead of
              (or in addition to) NewsletterWidget. Import useMatches and pass
              the data down. */}
          <NewsletterWidget />
          <CustomSidebarBlocks placement="video" />
          {/* ^^^ placement value — admin can create sidebar blocks targeted
              to this page type. Pick a sensible value: video, gallery, article,
              match, series, team, player, search, or a new one you introduce. */}
        </>
      }
    >
      {loading ? (
        // Loading skeleton — match the grid layout below
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
          {Array.from({ length: 6 }).map((_, i) => (
            <Skeleton key={i} className="aspect-video w-full rounded-lg" />
          ))}
        </div>
      ) : (data ?? []).length === 0 ? (
        // Empty state
        <Card className="grid min-h-[200px] place-items-center p-6">
          <div className="text-center">
            <div className="mb-2 text-3xl" aria-hidden="true">
              📭
            </div>
            <p className="text-sm text-muted-foreground">
              No featured videos yet. Check back soon.
            </p>

      ) : (
        // Main grid — adjust grid-cols count per your card size
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
          {(data ?? []).map((item) => (
            <VideoCard key={item.id} video={item} />
            // ^^^ rename `video={item}` to match your card component's prop
          ))}
        </div>
      )}
    </InnerPageLayout>
  )
}
