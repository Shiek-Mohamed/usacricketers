   136→  /* ------------------ Keyboard shortcuts (power-user scorers) ------------------
   137→   * 0,1,2,3,4,6 → score N runs | . / - → dot | w → wicket dialog |
   138→   * e → swap ends | u → undo | d → declare (Test) | s → save.
   139→   * No deps array → re-subscribes on every render with the latest closures.
   140→   * The function declarations (handleBall, swapEnds, etc.) are hoisted by
   141→   * JS so they're in scope at call-time even though defined later in the
   142→   * component body. The lint rule below is overly cautious about this

   144→  /* eslint-disable react-hooks/immutability */
   145→  useEffect(() => {
   146→    function isTyping(el: Element | null): boolean {
   147→      if (!el) return false
   148→      const tag = el.tagName?.toLowerCase()
   149→      return tag === "input" || tag === "textarea" || tag === "select" ||
   150→        el.getAttribute("role") === "combobox" ||
   151→        (el as HTMLElement)?.isContentEditable === true
   152→    }
   153→    function onKey(e: KeyboardEvent) {
   154→      if (e.metaKey || e.ctrlKey || e.altKey) return
   155→      if (isTyping(document.activeElement)) return
   156→      if (wicketOpen) return
   157→      if (!liveState || liveState.status !== "live") return
   158→      if (!liveState.strikerId || !liveState.nonStrikerId || !liveState.bowlerId) return
   159→      const k = e.key.toLowerCase()
   160→      if (["0", "1", "2", "3", "4", "6"].includes(k)) {
   161→        e.preventDefault()
   162→        handleBall({ kind: "run", runs: Number(k) })
   163→      } else if (k === "." || k === "-") {
   164→        e.preventDefault()
   165→        handleBall({ kind: "run", runs: 0 })
   166→      } else if (k === "w") {
   167→        e.preventDefault()
   168→        setWicketOpen(true)
   169→      } else if (k === "e") {
   170→        e.preventDefault()
   171→        swapEnds()
   172→      } else if (k === "u") {
   173→        e.preventDefault()
   174→        handleUndo()
   175→      } else if (k === "d") {
   176→        e.preventDefault()
   177→        handleDeclare()
   178→      } else if (k === "s") {
   179→        e.preventDefault()
   180→        if (liveState) {
   181→          persist(liveState)
   182→          toast.success("Scorecard saved")
   183→        }
   184→      }
   185→    }
   186→    window.addEventListener("keydown", onKey)
   187→    return () => window.removeEventListener("keydown", onKey)
   188→  })

   230→  function persist(state: LiveMatchState) {
   231→    if (saveTimer.current) clearTimeout(saveTimer.current)
   232→    saveTimer.current = setTimeout(async () => {
   233→      try {
   234→        await api.matches.update(match!.id, {
   235→          scorecard: stateToScorecard(state),
   236→          status: state.status,
   237→          result: state.result,
   238→        })
   239→        refetch()
   240→      } catch (e) {
   241→        // Surface persistence errors so the operator knows the DB write failed
   242→        // (was previously silent — caused confusion when "Innings started"
   243→        // toast showed but the match status didn't actually update).
   244→        // The live socket broadcast is still the priority — this toast
   245→        // doesn't block scoring, just informs the operator something's wrong.
   246→        const msg = e instanceof Error ? e.message : String(e).slice(0, 200)
   247→        toast.error("Persist failed — match state not saved", {
   248→          description: msg,
   249→        })
   250→      }
   251→    }, 800)
   252→  }

   254→  /** Add a manual / commentator note to the live commentary feed.
   255→   *  Pushes the updated state through the socket (so subscribers like the
   256→   *  public match-detail view see it instantly) and schedules a DB persist.
   257→   *  Manual entries get kind="manual" + a blue accent in the feed so they're
   258→   *  visually distinct from the auto-generated ball-by-ball commentary. */
   259→  function addCommentary(text: string) {
   260→    const trimmed = text.trim()
   261→    if (!trimmed || !liveState) return
   262→    const inn = liveState.innings[liveState.currentInnings]
   263→    const overLabel = inn ? oversString(inn.balls) : "0.0"
   264→    const entry = {
   265→      id: Math.random().toString(36).slice(2),
   266→      ballNo: liveState.ballNo,
   267→      over: overLabel,
   268→      text: trimmed,
   269→      kind: "manual" as const,
   270→    }
   271→    const next: LiveMatchState = {
   272→      ...liveState,
   273→      // Newest-first (matches the engine's pushCommentary pattern).
   274→      commentary: [entry, ...liveState.commentary].slice(0, 120),
   275→    }
   276→    setHistory((h) => [...h, liveState].slice(-50))
   277→    pushState(next)
   278→    persist(next)
   279→    setCommentaryText("")
   280→  }

   290→  function handleUndo() {
   291→    if (history.length === 0) {
   292→      toast.info("Nothing to undo")
   293→      return
   294→    }
   295→    const prev = history[history.length - 1]
   296→    setHistory((h) => h.slice(0, -1))
   297→    pushState(prev)
   298→    persist(prev)
   299→    toast.success("Undid last action")
   300→  }

   302→  function startInnings(
   303→    battingFirstId: string,
   304→    strikerId: string,
   305→    nonStrikerId: string,
   306→    bowlerId: string
   307→  ) {
   308→    const battingPlayers = playersByTeam[battingFirstId] ?? []
   309→    const bowlingPlayers =
   310→      playersByTeam[battingFirstId === match!.team1Id ? match!.team2Id : match!.team1Id] ?? []
   311→    const striker = battingPlayers.find((p) => p.id === strikerId)
   312→    const nonStriker = battingPlayers.find((p) => p.id === nonStrikerId)
   313→    const bowler = bowlingPlayers.find((p) => p.id === bowlerId)
   314→    if (!striker || !nonStriker || !bowler) {
   315→      toast.error("Select striker, non-striker and bowler")
   316→      return
   317→    }
   318→    let state = initState(match!, battingFirstId)
   319→    // Pre-populate the squad with the full batting-team roster so the
   320→    // "Did not bat" section of the public scorecard lists bench players.
   321→    // Captain/keeper tags come from the explicit Player model fields
   322→    // (was previously a brittle regex on the role string).
   323→    for (const p of battingPlayers) {
   324→      state = addSquadMember(state, p.id, p.name, {
   325→        isCaptain: !!p.isCaptain,
   326→        isKeeper: !!p.isKeeper,
   327→      })
   328→    }
   329→    state = addBatsman(state, striker.id, striker.name, {
   330→      isCaptain: !!striker.isCaptain,
   331→      isKeeper: !!striker.isKeeper,
   332→    })
   333→    state = addBatsman(state, nonStriker.id, nonStriker.name, {
   334→      isCaptain: !!nonStriker.isCaptain,
   335→      isKeeper: !!nonStriker.isKeeper,
   336→    })
   337→    state = setBatsmen(state, striker.id, nonStriker.id)
   338→    state = setBowler(state, bowler.id, bowler.name)
   339→    setHistory([])
   340→    pushState(state)
   341→    persist(state)
   342→    toast.success("Innings started — status set to LIVE")
   343→  }

   345→  function bringInBatsman(playerId: string) {
   346→    if (!liveState) return
   347→    const battingPlayers = playersByTeam[liveState.innings[liveState.currentInnings].battingTeamId] ?? []
   348→    const p = battingPlayers.find((x) => x.id === playerId)
   349→    if (!p) return
   350→    let state = addBatsman(liveState, p.id, p.name, {
   351→      isCaptain: !!p.isCaptain,
   352→      isKeeper: !!p.isKeeper,
   353→    })
   354→    // Fill whichever slot is currently empty — the striker's end if a normal
   355→    // wicket fell (strikerId === null), or the non-striker's end if a run-out
   356→    // happened at the non-striker's end (nonStrikerId === null). If both are
   357→    // empty (innings restart, edge case), default to striker.
   358→    const fillStriker = liveState.strikerId === null
   359→    const fillNonStriker = !fillStriker && liveState.nonStrikerId === null
   360→    state = {
   361→      ...state,
   362→      strikerId: fillStriker ? p.id : state.strikerId,
   363→      nonStrikerId: fillNonStriker ? p.id : state.nonStrikerId,
   364→    }
   365→    setHistory((h) => [...h, liveState].slice(-50))
   366→    pushState(state)
   367→    persist(state)
   368→    toast.success(`${p.name} in`)
   369→  }

   371→  function pickNextBowler(playerId: string) {
   372→    if (!liveState) return

   383→  /** "I picked the wrong striker — let me re-pick." Pushes the current state

   416→  /** Swap the striker and non-striker ends. Used when the auto strike-rotation
   417→   *  logic disagrees with reality (e.g. batsmen crossed on a caught-and-
   418→   *  bowled, odd run on a wide, runner substituted, etc.). Allowed at any
   419→   *  time — doesn't affect stats, just relabels who's on strike. */
   420→  function swapEnds() {
   421→    if (!liveState) return
   422→    if (!liveState.strikerId || !liveState.nonStrikerId) {
   423→      toast.error("Need both batsmen at the crease to swap ends")
   424→      return
   425→    }
   426→    setHistory((h) => [...h, liveState].slice(-50))
   427→    const next = {
   428→      ...liveState,
   429→      strikerId: liveState.nonStrikerId,
   430→      nonStrikerId: liveState.strikerId,
   431→    }
   432→    pushState(next)
   433→    persist(next)
   434→    toast.success("Ends swapped")
   435→  }

   438→    setHistory([])

   444→  /** Test-match declaration: captain ends the current innings without being
   445→   *  all out. No-op for limited-overs matches (those don't have declarations).
   446→   *  Pushes the prev state to history (so Undo can recover) then runs the
   447→   *  engine's declareInnings which marks `declared: true` + sets up the
   448→   *  next innings / computes the result. */
   449→  function handleDeclare() {
   450→    if (!liveState) return
   451→    if (liveState.maxOvers > 0) {
   452→      toast.error("Declarations are only available in Test matches")
   453→      return
   454→    }
   455→    if (liveState.status !== "live") {
   456→      toast.error("Can't declare — innings already ended")
   457→      return
   458→    }
   459→    setHistory((h) => [...h, liveState].slice(-50))
   460→    const next = declareInnings(liveState)
   461→    pushState(next)
   462→    persist(next)
   463→    toast.success("Innings declared")
   464→  }

   466→  /** Enforce the follow-on in a Test match. Only valid when Team A leads by
   467→   *  200+ runs after Team B's 1st innings. Team B bats again immediately. */
   468→  function handleFollowOn() {
   469→    if (!liveState) return
   470→    setHistory((h) => [...h, liveState].slice(-50))
   471→    const next = enforceFollowOn(liveState)
   472→    if (next === liveState) {
   473→      toast.error("Follow-on requires a 200+ run lead in a Test after innings 1")
   474→      return
   475→    }
   476→    pushState(next)
   477→    persist(next)
   478→    toast.success("Follow-on enforced")
   479→  }

   481→  /** Start a super-over after a tied limited-overs match. Creates 2 new
   482→   *  1-over innings (one per team). */
   483→  function handleSuperOver() {
   484→    if (!liveState) return
   485→    setHistory((h) => [...h, liveState].slice(-50))
   486→    const next = startSuperOver(liveState)
   487→    if (next === liveState) {
   488→      toast.error("Super-over only available for tied limited-overs matches")
   489→      return
   490→    }
   491→    pushState(next)
   492→    persist(next)
   493→    toast.success("Super Over started")
   494→  }

   497→  const currentInnings = liveState?.innings[liveState.currentInnings]
   498→  // Need a new batsman if either end is empty after a wicket (a run-out can
   499→  // empty the non-striker slot while the striker stays in).
   500→  const needStriker = !!liveState && liveState.strikerId === null && liveState.status === "live"
   501→  const needNonStriker =
   502→    !!liveState &&
   503→    liveState.nonStrikerId === null &&
   504→    liveState.status === "live" &&
   505→    !!currentInnings &&
   506→    currentInnings.batsmen.length > 0 // not the very first pickup of an innings
   507→  const needBatsman = needStriker || needNonStriker
   508→  const needBowler = !!liveState && liveState.bowlerId === null && liveState.status === "live"
   509→  const isComplete = liveState?.status === "completed"

   923→function LiveScoreboard({
   924→  state,
   925→  onChangeStriker,
   926→  onChangeNonStriker,
   927→  onChangeBowler,
   928→  onSwapEnds,
   929→}: {
   930→  state: LiveMatchState
   931→  onChangeStriker?: () => void
   932→  onChangeNonStriker?: () => void
   933→  onChangeBowler?: () => void
   934→  /** Swap striker ↔ non-striker — override the auto strike-rotation logic
   935→   *  when it disagrees with reality (caught-and-bowled crossing, etc.).
   936→   *  Allowed at any time (not over-boundary-gated) — doesn't affect stats. */
   937→  onSwapEnds?: () => void
   938→}) {
   939→  const inn = state.innings[state.currentInnings]
   940→  const striker = inn.batsmen.find((b) => b.id === state.strikerId)
   941→  const nonStriker = inn.batsmen.find((b) => b.id === state.nonStrikerId)
   942→  const bowler = inn.bowlers.find((b) => b.id === state.bowlerId)
   943→  const overs = oversString(inn.balls)
   944→  const runRate = inn.balls > 0 ? (inn.runs / inn.balls * 6).toFixed(2) : "0.00"
   945→  const target = inn.target
   946→  const ballsLeft = state.maxOvers > 0 ? state.maxOvers * 6 - inn.balls : null
   947→  const runsNeeded = target ? target - inn.runs : null
   948→  const reqRate =
   949→    target && ballsLeft && ballsLeft > 0
   950→      ? ((runsNeeded! / ballsLeft) * 6).toFixed(2)
   951→      : null
   952→
   953→  // A slot can be safely "Changed" only at over-boundaries — mid-over swaps
   954→  // would mess up the bowler's figures + the batsman's balls-faced count.
   955→  // Disable the Change button once any ball of the current over has been
   956→  // bowled (i.e. balls % 6 !== 0). The exception is the bowler slot after a
   957→  // wicket (still mid-over) — the bowler can be subbed off due to injury in
   958→  // real cricket, but for our MVP we keep the rule simple: only at over
   959→  // boundaries. Undo is still available mid-over for fixing mistakes.
   960→  const atOverBoundary = inn.balls % 6 === 0
   961→  const canChangeBatsmen = atOverBoundary && state.status === "live"
   962→  const canChangeBowler = atOverBoundary && state.status === "live"
   963→
   964→  return (
   965→    <div className="rounded-lg border bg-gradient-to-br from-slate-900 to-slate-800 p-4 text-white sm:p-5">
   966→      <div className="flex flex-wrap items-end justify-between gap-3">
   967→        <div>
   968→          <div className="text-xs uppercase tracking-wide text-slate-400">
   969→            {inn.battingTeamShortName} · Innings {state.currentInnings + 1}
   970→            {target ? ` · need ${runsNeeded} runs` : ""}
   971→          </div>
   972→          <div className="mt-1 text-4xl font-extrabold tabular-nums">
   973→            {inn.runs}/{inn.wickets}
   974→            <span className="ml-2 text-lg font-normal text-slate-300">
   975→              ({overs} ov)
   976→            </span>
   977→            {inn.freeHit && (
   978→              <span className="ml-3 inline-flex items-center gap-1 rounded-full bg-amber-500 px-2.5 py-0.5 text-xs font-extrabold uppercase tracking-wide text-amber-950 animate-pulse">
   979→                <span className="h-1.5 w-1.5 rounded-full bg-amber-950" />
   980→                Free hit
   981→              </span>
   982→            )}
   983→          </div>
   984→          <div className="mt-1 text-xs text-slate-300">
   985→            CRR {runRate}
   986→            {reqRate && ` · RRR ${reqRate}`}
   987→            {ballsLeft !== null && ` · ${ballsLeft} balls left`}
   988→          </div>
   989→        </div>
   990→      </div>
   991→      <div className="mt-3 grid grid-cols-2 gap-3 text-xs">
   992→        <div className="rounded-md bg-white/10 p-2.5">
   993→          <div className="flex items-center justify-between">
   994→            <div className="text-[10px] uppercase text-slate-300">Striker</div>
   995→            {striker && onChangeStriker && (
   996→              <ChangeButton
   997→                onClick={onChangeStriker}
   998→                disabled={!canChangeBatsmen}
   999→                hint={
  1000→                  canChangeBatsmen
  1001→                    ? "Re-pick striker"
  1002→                    : "Wait for end of over (or use Undo)"
  1003→                }
  1004→              />
  1005→            )}
  1006→          </div>
  1007→          <div className="font-bold">{striker?.name ?? "—"}</div>
  1008→          <div className="text-slate-300">
  1009→            {striker ? `${striker.runs} (${striker.balls})` : "—"}
  1010→          </div>
  1011→        </div>
  1012→        <div className="relative rounded-md bg-white/10 p-2.5">
  1013→          <div className="flex items-center justify-between">
  1014→            <div className="text-[10px] uppercase text-slate-300">Non-striker</div>
  1015→            {nonStriker && onChangeNonStriker && (
  1016→              <ChangeButton
  1017→                onClick={onChangeNonStriker}
  1018→                disabled={!canChangeBatsmen}
  1019→                hint={
  1020→                  canChangeBatsmen
  1021→                    ? "Re-pick non-striker"
  1022→                    : "Wait for end of over (or use Undo)"
  1023→                }
  1024→              />
  1025→            )}
  1026→          </div>
  1027→          <div className="font-bold">{nonStriker?.name ?? "—"}</div>
  1028→          <div className="text-slate-300">
  1029→            {nonStriker ? `${nonStriker.runs} (${nonStriker.balls})` : "—"}
  1030→          </div>
  1031→          {/* Swap-ends button — pinned to the inner edge between the two
  1032→              batsman cells. Visible only when both batsmen are at the crease.
  1033→              Always enabled (not over-boundary-gated) because swapping ends
  1034→              doesn't affect any stats — it's just a relabel. */}
  1035→          {striker && nonStriker && onSwapEnds && (
  1036→            <button
  1037→              type="button"
  1038→              onClick={onSwapEnds}
  1039→              title="Swap striker ↔ non-striker (override the auto strike-rotation)"
  1040→              aria-label="Swap striker and non-striker ends"
  1041→              className="absolute -left-[14px] top-1/2 grid h-7 w-7 -translate-y-1/2 place-items-center rounded-full border border-white/20 bg-slate-700 text-white shadow-md transition-colors hover:bg-slate-600"
  1042→            >
  1043→              <ArrowLeftRight className="h-3.5 w-3.5" />
  1044→            </button>
  1045→          )}
  1046→        </div>
  1047→        <div className="col-span-2 rounded-md bg-white/10 p-2.5">
  1048→          <div className="flex items-center justify-between">
  1049→            <div className="text-[10px] uppercase text-slate-300">Bowler</div>
  1050→            {bowler && onChangeBowler && (
  1051→              <ChangeButton
  1052→                onClick={onChangeBowler}
  1053→                disabled={!canChangeBowler}
  1054→                hint={
  1055→                  canChangeBowler
  1056→                    ? "Re-pick bowler"
  1057→                    : "Wait for end of over (or use Undo)"
  1058→                }
  1059→              />
  1060→            )}
  1061→          </div>
  1062→          <div className="flex items-center justify-between">
  1063→            <span className="font-bold">{bowler?.name ?? "—"}</span>

  1187→/** Build the dismissal summary string from the structured inputs (this is
  1188→ *  what the public scorecard renders next to the batsman's name). */
  1189→function buildDismissalText(args: {
  1190→  wicketType: WicketType
  1191→  bowlerName?: string
  1192→  fielderName?: string
  1193→}): string {
  1194→  const bowler = args.bowlerName ?? "?"
  1195→  const fielder = args.fielderName
  1196→  switch (args.wicketType) {
  1197→    case "bowled":
  1198→      return `b ${bowler}`
  1199→    case "caught":
  1200→      // Caught & bowled when no fielder is selected OR the fielder is the bowler.
  1201→      if (!fielder || fielder === bowler) return `c &b ${bowler}`
  1202→      return `c ${fielder} b ${bowler}`
  1203→    case "lbw":
  1204→      return `lbw b ${bowler}`
  1205→    case "stumped":
  1206→      return fielder ? `st ${fielder} b ${bowler}` : `st ${bowler}`
  1207→    case "runout":
  1208→      return fielder ? `run out (${fielder})` : `run out`
  1209→    case "hitwkt":
  1210→      return `hit wkt b ${bowler}`
  1211→    case "retiredHurt":
  1212→      return `retired hurt`
  1213→    case "retiredOut":
  1214→      return `retired out`
  1215→  }
  1216→}

  1218→function WicketDialog({
  1219→  open,
  1220→  onOpenChange,
  1221→  striker,
  1222→  nonStriker,
  1223→  bowler,
  1224→  fieldingPlayers,
  1225→  freeHit,
  1226→  onConfirm,
  1227→}: {
  1228→  open: boolean
  1229→  onOpenChange: (v: boolean) => void
  1230→  striker: { id: string; name: string }
  1231→  nonStriker: { id: string; name: string }
  1232→  bowler: { id: string; name: string }
  1233→  fieldingPlayers: { id: string; name: string; role?: string }[]
  1234→  /** True when the current ball is a free hit — bowler-credited dismissals
  1235→   *  (bowled/caught/lbw/stumped/hit-wkt) will be REJECTED by the engine.
  1236→   *  The dialog shows a hint banner so the admin knows to pick Run Out or
  1237→   *  Retired Out instead. */
  1238→  freeHit?: boolean
  1239→  onConfirm: (ev: {
  1240→    kind: "wicket"
  1241→    wicketType: WicketType
  1242→    outBatsmanId: string
  1243→    fielderIds: string[]
  1244→    dismissal: string
  1245→  }) => void
  1246→}) {
  1247→  const [wicketType, setWicketType] = useState<WicketType>("bowled")
  1248→  const [outBatsmanId, setOutBatsmanId] = useState<string>(striker.id)
  1249→  const [fielderId, setFielderId] = useState<string>("")
  1250→  // Note: this component is conditionally rendered by the parent only when
  1251→  // `wicketOpen` is true, so it remounts fresh each time the user opens the
  1252→  // dialog — useState initial values act as the "reset on open".
  1253→
  1254→  // The wicket-keeper on the fielding side (auto-prefilled for stumped).
  1255→  // Uses the explicit `isKeeper` flag on the Player model — was previously
  1256→  // a brittle regex on the role string.
  1257→  const keeperId = useMemo(() => {
  1258→    const k = fieldingPlayers.find((p) => p.isKeeper)
  1259→    return k?.id ?? ""
  1260→  }, [fieldingPlayers])
  1261→
  1262→  // Derived state — avoid setState-in-effect by computing the effective
  1263→  // selections during render. The picker's `value` reads from these, so the
  1264→  // UI stays in sync without needing an effect to "correct" the user's
  1265→  // last selection when the dismissal type changes.
  1266→  // • For any dismissal type other than run-out, the batsman out is always
  1267→  //   the striker (run-out can happen at the non-striker's end).
  1268→  // • For stumped with no fielder selected, auto-prefill the keeper.
  1269→  const effectiveOutBatsmanId =
  1270→    wicketType === "runout" ? outBatsmanId : striker.id
  1271→  const effectiveFielderId =
  1272→    wicketType === "stumped" && !fielderId && keeperId ? keeperId : fielderId
  1273→
  1274→  const needsFielder =
  1275→    wicketType === "caught" ||
  1276→    wicketType === "stumped" ||
  1277→    wicketType === "runout"
  1278→  const showBatsmanSelect = wicketType === "runout"
  1279→
  1280→  const fielderName = fieldingPlayers.find(
  1281→    (p) => p.id === effectiveFielderId
  1282→  )?.name
  1283→  const dismissal = buildDismissalText({
  1284→    wicketType,
  1285→    bowlerName: bowler.name,
  1286→    fielderName,
  1287→  })
  1288→
  1289→  const outBatsmanName =
  1290→    effectiveOutBatsmanId === striker.id
  1291→      ? striker.name
  1292→      : effectiveOutBatsmanId === nonStriker.id
  1293→        ? nonStriker.name
  1294→        : "?"
  1295→
  1296→  // Free-hit-derived warning: when the current ball is a free hit AND the
  1297→  // selected dismissal type is bowler-credited, the engine will REJECT the
  1298→  // wicket (batsman survives). Show a warning so the admin knows.
  1299→  const freeHitWillReject =
  1300→    !!freeHit &&
  1301→    BOWLER_CREDITED_WICKETS.has(wicketType) &&
  1302→    wicketType !== "retired"
  1303→
  1304→  function handleConfirm() {
  1305→    onConfirm({
  1306→      kind: "wicket",
  1307→      wicketType,
  1308→      outBatsmanId: effectiveOutBatsmanId,
  1309→      fielderIds: effectiveFielderId ? [effectiveFielderId] : [],
  1310→      dismissal,
  1311→    })
  1312→  }
  1313→
  1314→  return (
  1315→    <Dialog open={open} onOpenChange={onOpenChange}>
  1316→      <DialogContent className="sm:max-w-lg">
  1317→        <DialogHeader>
  1318→          <DialogTitle className="flex items-center gap-2">
  1319→            <span className="grid h-7 w-7 place-items-center rounded-md bg-red-600 text-white">
  1320→              <Activity className="h-4 w-4" />
  1321→            </span>
  1322→            Dismiss {outBatsmanName}
  1323→          </DialogTitle>
  1324→          <DialogDescription>
  1325→            Pick the dismissal type. If caught, stumped or run out, choose the
  1326→            fielder(s) involved. The dismissal text updates live in the preview.
  1327→          </DialogDescription>
  1328→        </DialogHeader>
  1329→
  1330→        {/* Free-hit banner — appears when the current ball is a free hit. */}
  1331→        {freeHit && (
  1332→          <div className="rounded-md border border-amber-400 bg-amber-50 px-3 py-2 text-xs dark:bg-amber-950/40">
  1333→            <div className="flex items-center gap-1.5 font-bold uppercase tracking-wide text-amber-800 dark:text-amber-300">
  1334→              <span className="h-1.5 w-1.5 animate-pulse rounded-full bg-amber-600" />
  1335→              Free hit in effect
  1336→            </div>
  1337→            <p className="mt-1 text-amber-700 dark:text-amber-400">
  1338→              Bowler-credited dismissals (bowled, caught, lbw, stumped, hit
  1339→              wicket) are <strong>not allowed</strong> on a free hit — the
  1340→              batsman survives and only run-out / retired-out stand.
  1341→              {freeHitWillReject
  1342→                ? " Picking this dismissal will be rejected by the engine — pick Run Out instead."
  1343→                : ""}
  1344→            </p>
  1345→          </div>
  1346→        )}
  1347→
  1348→        <div className="grid gap-3 py-2">
  1349→          {/* Dismissal type */}
  1350→          <div className="grid gap-1.5">
  1351→            <Label>Dismissal type</Label>
  1352→            <Select value={wicketType} onValueChange={(v) => setWicketType(v as WicketType)}>
  1353→              <SelectTrigger className="w-full">
  1354→                <SelectValue />
  1355→              </SelectTrigger>
  1356→              <SelectContent>
  1357→                {WICKET_TYPE_OPTIONS.map((opt) => (
  1358→                  <SelectItem key={opt.value} value={opt.value}>
  1359→                    <span className="font-medium">{opt.label}</span>
  1360→                    <span className="ml-2 text-[10px] text-muted-foreground">{opt.hint}</span>
  1361→                  </SelectItem>
  1362→                ))}
  1363→              </SelectContent>
  1364→            </Select>
  1365→          </div>
  1366→
  1367→          {/* Batsman out — only shown for run-out */}
  1368→          {showBatsmanSelect && (
  1369→            <div className="grid gap-1.5">
  1370→              <Label>Which batsman is out?</Label>
  1371→              <Select
  1372→                value={effectiveOutBatsmanId}
  1373→                onValueChange={setOutBatsmanId}
  1374→              >
  1375→                <SelectTrigger className="w-full">
  1376→                  <SelectValue />
  1377→                </SelectTrigger>
  1378→                <SelectContent>
  1379→                  <SelectItem value={striker.id}>
  1380→                    Striker — {striker.name}
  1381→                  </SelectItem>
  1382→                  <SelectItem value={nonStriker.id}>
  1383→                    Non-striker — {nonStriker.name}
  1384→                  </SelectItem>
  1385→                </SelectContent>
  1386→              </Select>
  1387→            </div>
  1388→          )}
  1389→
  1390→          {/* Fielder(s) involved — for caught, stumped, run out */}
  1391→          {needsFielder && (
  1392→            <div className="grid gap-1.5">
  1393→              <Label>
  1394→                {wicketType === "caught"
  1395→                  ? "Catcher"
  1396→                  : wicketType === "stumped"
  1397→                    ? "Wicket-keeper"
  1398→                    : "Run-out fielder"}
  1399→              </Label>
  1400→              <Select value={effectiveFielderId} onValueChange={setFielderId}>
  1401→                <SelectTrigger className="w-full">
  1402→                  <SelectValue placeholder="(direct hit / no fielder)" />
  1403→                </SelectTrigger>
  1404→                <SelectContent>
  1405→                  <SelectItem value="(direct-hit)">(direct hit / no fielder)</SelectItem>
  1406→                  {fieldingPlayers.map((p) => (
  1407→                    <SelectItem key={p.id} value={p.id}>
  1408→                      {p.name}
  1409→                      {p.isKeeper && " †"}
  1410→                      {p.isCaptain && " (c)"}
  1411→                    </SelectItem>
  1412→                  ))}
  1413→                </SelectContent>
  1414→              </Select>
  1415→              {wicketType === "stumped" && keeperId && !fielderId && (
  1416→                <p className="text-[11px] text-muted-foreground">
  1417→                  Wicket-keeper auto-prefilled (you can change it).
  1418→                </p>
  1419→              )}
  1420→            </div>
  1421→          )}
  1422→
  1423→          {/* Live dismissal-text preview */}
  1424→          <div className="rounded-md border bg-muted/40 p-3 text-sm">
  1425→            <div className="text-[10px] font-bold uppercase tracking-wide text-muted-foreground">

  1427→            </div>
  1428→            <div className="mt-1 font-mono text-sm">
  1429→              <span className="font-semibold">{outBatsmanName}</span>{" "}
  1430→              <span className="text-muted-foreground">{dismissal}</span>
  1431→            </div>
  1432→          </div>
  1433→        </div>
  1434→
  1435→        <DialogFooter>
  1436→          <Button variant="outline" onClick={() => onOpenChange(false)}>
  1437→            Cancel
  1438→          </Button>
  1439→          <Button
  1440→            onClick={handleConfirm}

  1452→function Picker({

  1461→  placeholder: string

  1515→            </span>
  1516→            <Button
  1517→              size="sm"
  1518→              disabled={!val}
  1519→              onClick={() => val && onPick(val)}
  1520→              className="gap-1.5"

  1524→            </Button>
